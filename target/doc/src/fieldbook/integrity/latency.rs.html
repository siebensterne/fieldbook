1//! Measured discovery and websocket timestamp-latency experiment.
2//!
3//! If no timestamped frames arrive, percentiles are absent. Exchange
4//! timestamps are compared with the local receive clock, so the result
5//! includes any exchange/local clock offset. When more than one replica is
6//! configured, identical events are matched across connections and their
7//! arrival skew is reported separately. The methodology is documented in
8//! The methodology is recorded in `docs/performance-methodology.md`.

35/// Message from a replica task to the measurement loop.
36enum ReplicaEvent {
37    Frame(usize, Result<SessionFrame, String>),
38    /// Wall-clock milliseconds from the replica task start (immediately after
39    /// connect and subscribe) to the first frame; sent once per replica,
40    /// immediately before the frame itself.
41    FirstFrame(usize, f64),
42}
43
44/// Startup period excluded from skew measurement. Initial snapshots are large,
45/// arrive at different times on each connection, and would otherwise dominate
46/// the distribution with connection-setup artefacts rather than steady-state
47/// feed behaviour.
48const SETTLE_SECS: u64 = 2;
49/// Bound on pending cross-replica matches. A replica that never sees an event
50/// must not grow the map without limit.
51const MAX_PENDING_MATCHES: usize = 100_000;
52/// Bound on recorded skew samples, matching the latency sample cap.
53const MAX_SKEW_SAMPLES: usize = 1_000_000;
54

99/// Measure discovery RTT and exchange-to-local timestamp latency for a short
100/// live session. When `settings.collector.replicas > 1`, every replica opens
101/// its own connection and the report includes cross-connection arrival skew.
102pub async fn run(
103    settings: Settings,
104    max_markets: usize,
105    duration_secs: u64,
106    output: Option<&Path>,
107) -> anyhow::Result<LatencyReport> {
108    settings.validate()?;
109    let network = settings.network.clone();
110    let replica_count = settings.collector.replicas;
111    let exchange = build_exchange(&settings, true)?;
112    let discovery_started = Instant::now();
113    let markets = exchange
114        .discover_markets(max_markets)
115        .await
116        .with_context(|| format!("discovering {} markets for latency run", exchange.name()))?;
117    let discovery_rtt_ms = discovery_started.elapsed().as_secs_f64() * 1_000.0;
118    if markets.is_empty() {
119        bail!("{} returned no markets for latency run", exchange.name());
120    }
121    let wanted = if max_markets == 0 {
122        markets.len()
123    } else {
124        max_markets.min(markets.len())
125    };
126    let desired: std::collections::BTreeSet<String> = markets
127        .iter()
128        .take(wanted)
129        .map(|market| market.native_symbol.clone())
130        .collect();
131    let factory = Arc::clone(&exchange).session_factory();
132
133    let mut sessions: Vec<Box<dyn MarketDataSession>> = Vec::with_capacity(replica_count);
134    let mut replica_connect_ms: Vec<f64> = Vec::with_capacity(replica_count);
135    for replica in 0..replica_count {
136        let mut session = factory
137            .create_session(SourceId::new(0, replica as u16))
138            .await
139            .with_context(|| format!("creating latency session {replica}"))?;
140        let connect_started = Instant::now();
141        connect_with_retry(&mut session, replica).await?;
142        replica_connect_ms.push(connect_started.elapsed().as_secs_f64() * 1_000.0);
143        session
144            .apply_subscriptions(&desired)
145            .await
146            .map_err(|failure| {
147                anyhow::anyhow!(
148                    "subscribing latency session {replica}: {}",
149                    failure_text(&failure)
150                )
151            })?;
152        sessions.push(session);
153    }
154    info!(
155        exchange = exchange.name(),
156        markets = wanted,
157        replicas = replica_count,
158        duration_secs,
159        "latency sessions started"
160    );
161
162    let (frame_tx, mut frame_rx) = mpsc::channel::<ReplicaEvent>(4096);
163    let (stop_tx, _) = broadcast::channel::<()>(1);
164    let mut handles = Vec::with_capacity(replica_count);
165    for (replica, session) in sessions.into_iter().enumerate() {
166        handles.push(tokio::spawn(run_replica(
167            replica,
168            session,
169            frame_tx.clone(),
170            stop_tx.subscribe(),
171        )));
172    }
173    drop(frame_tx);
174
175    let mut frames = 0u64;
176    let mut events = 0u64;
177    let mut bytes = 0u64;
178    let mut negative_latency_samples = 0u64;
179    let mut samples: Vec<i64> = Vec::new();
180    let mut replica_events = vec![0u64; replica_count];
181    let mut first_frames: Vec<Option<f64>> = vec![None; replica_count];
182    let mut errors: Vec<String> = Vec::new();
183    let mut skew = SkewTracker::new(replica_count);
184    let settle_secs = if replica_count > 1 { SETTLE_SECS } else { 0 };
185    // The settle window discards startup frames, but the
186    // first-frame timing is setup evidence and must survive it. Replica
187    // transport errors during the window are retained.
188    errors.extend(discard_startup_frames(&mut frame_rx, settle_secs, &mut first_frames).await);
189    let measurement_started_us =
190        now_us().context("reading wall clock at the start of latency measurement")?;
191    let deadline = tokio::time::Instant::now() + Duration::from_secs(duration_secs.max(1));
192
193    loop {
194        tokio::select! {
195            _ = sleep_until(deadline) => break,
196            received = frame_rx.recv() => {
197                match received {
198                    Some(ReplicaEvent::FirstFrame(replica, elapsed_ms)) => {
199                        if let Some(slot) = first_frames.get_mut(replica) {
200                            slot.get_or_insert(elapsed_ms);
201                        }
202                    }
203                    Some(ReplicaEvent::Frame(replica, Ok(frame))) => {
204                        if frame.bytes > 0 {
205                            frames = frames.saturating_add(1);
206                            bytes = bytes.saturating_add(frame.bytes as u64);
207                        }
208                        for event in &frame.events {
209                            let Some(meta) = event_meta(event) else {
210                                continue;
211                            };
212                            if meta.received_at_us < measurement_started_us {
213                                continue;
214                            }
215                            events = events.saturating_add(1);
216                            replica_events[replica] = replica_events[replica].saturating_add(1);
217                            let Some(exchange_ts_us) = meta.exchange_ts_us else {
218                                continue;
219                            };
220                            let latency = meta.received_at_us.saturating_sub(exchange_ts_us);
221                            if latency < 0 {
222                                negative_latency_samples =
223                                    negative_latency_samples.saturating_add(1);
224                            } else if latency <= 60_000_000 && samples.len() < 1_000_000 {
225                                samples.push(latency);
226                            }
227                            if let Some(key) = event_key(event) {
228                                skew.observe(key, replica, meta.received_at_us);
229                            }
230                        }
231                    }
232                    Some(ReplicaEvent::Frame(_, Err(error))) => {
233                        errors.push(error);
234                    }
235                    None => break,
236                }
237            }
238        }
239    }
240
241    let _ = stop_tx.send(());
242    for handle in handles {
243        if timeout(Duration::from_secs(5), handle).await.is_err() {
244            warn!("latency replica task did not stop within five seconds");
245        }
246    }
247
248    samples.sort_unstable();
249    let latency_samples = samples.len() as u64;
250    let mut skew_samples = std::mem::take(&mut skew.samples);
251    skew_samples.sort_unstable();
252    let skew_sample_count = skew_samples.len() as u64;
253    let abandoned = skew
254        .pending
255        .len()
256        .saturating_add(usize::try_from(skew.dropped).unwrap_or(usize::MAX));
257    let report = LatencyReport {
258        exchange: exchange.name().to_owned(),
259        environment: exchange.environment().to_owned(),
260        measured_at_us: now_us().unwrap_or(0),
261        discovery_rtt_ms,
262        discovered_markets: markets.len(),
263        subscribed_markets: wanted,
264        duration_secs: duration_secs.max(1),
265        settle_secs,
266        replicas: replica_count,
267        tcp_nodelay: network.tcp_nodelay,
268        source_ip: network.source_ip.map(|ip| ip.to_string()),
269        source_port: network.source_port,
270        connect_race: network.connect_race,
271        frames,
272        events,
273        bytes,
274        replica_connect_ms,
275        replica_first_frame_ms: first_frames,
276        latency_samples,
277        negative_latency_samples,
278        latency_p50_us: percentile(&samples, 50),
279        latency_p95_us: percentile(&samples, 95),
280        latency_p99_us: percentile(&samples, 99),
281        latency_min_us: samples.first().copied(),
282        latency_max_us: samples.last().copied(),
283        latency_mean_us: mean(&samples),
284        replica_events,
285        replica_first_arrivals: skew.first_arrivals,
286        replica_matched_events: skew.matched,
287        replica_abandoned_events: u64::try_from(abandoned).unwrap_or(u64::MAX),
288        replica_skew_samples: skew_sample_count,
289        replica_skew_p50_us: percentile(&skew_samples, 50),
290        replica_skew_p95_us: percentile(&skew_samples, 95),
291        replica_skew_p99_us: percentile(&skew_samples, 99),
292        replica_skew_min_us: skew_samples.first().copied(),
293        replica_skew_max_us: skew_samples.last().copied(),
294        replica_skew_mean_us: mean(&skew_samples),
295        error: errors.first().cloned(),
296        errors,
297        notes: vec![
298            "discovery_rtt_ms is a local wall-clock measurement around one discovery call".to_owned(),
299            "latency is local_receive_us - exchange_timestamp_us and includes exchange/local clock offset".to_owned(),
300            "negative values are counted separately and excluded from percentiles".to_owned(),
301            "no latency value is reported when no timestamped events arrived".to_owned(),
302            "replica_connect_ms and replica_first_frame_ms are local wall-clock measurements of session setup and subscription-to-first-frame latency".to_owned(),
303            format!(
304                "replica skew compares the same symbol, exchange timestamp, clock and event shape across {replica_count} connection(s); each delta is relative to the earliest arrival"
305            ),
306            format!(
307                "the first {settle_secs} second(s) after all subscriptions are established are excluded so initial snapshots do not dominate the skew distribution"
308            ),
309            "identical duplicate events at the same exchange timestamp and shape can be coalesced; unmatched events are counted as abandoned".to_owned(),
310            "a one-replica run reports no skew samples; skew requires at least two independent connections".to_owned(),
311        ],
312    };
313    if let Some(path) = output {
314        write_json_atomic(path, &report).context("writing latency report")?;
315    }
316    if let Some(error) = &report.error {
317        warn!(error, "latency run ended with a replica transport error");
318    }
319    Ok(report)
320}

322/// Connect one latency session with the same bounded retry the live collector
323/// uses, because the venue edge intermittently rejects individual handshakes
324/// (HTTP 525). A fatal failure such as an authentication rejection is never
325/// retried. The measurement window only starts after every session is
326/// connected, so a retried handshake cannot bias any reported value.
327async fn connect_with_retry(
328    session: &mut Box<dyn MarketDataSession>,
329    replica: usize,
330) -> anyhow::Result<()> {
331    let mut last_failure = None;
332    for attempt in 0..3u32 {
333        match session.connect().await {
334            Ok(()) => return Ok(()),
335            Err(SessionFailure::Fatal(error)) => {
336                return Err(anyhow::anyhow!(
337                    "connecting latency session {replica}: {error:#}"
338                ));
339            }
340            Err(failure) => {
341                warn!(
342                    replica,
343                    attempt = attempt + 1,
344                    error = %failure_text(&failure),
345                    "latency session connect failed; retrying"
346                );
347                last_failure = Some(failure_text(&failure));
348                if attempt < 2 {
349                    tokio::time::sleep(Duration::from_millis(1_000 << attempt)).await;
350                }
351            }
352        }
353    }
354    Err(anyhow::anyhow!(
355        "connecting latency session {replica} failed after three attempts: {}",
356        last_failure.unwrap_or_else(|| "unknown failure".to_owned())
357    ))
358}

360async fn discard_startup_frames(
361    frames: &mut mpsc::Receiver<ReplicaEvent>,
362    settle_secs: u64,
363    first_frames: &mut [Option<f64>],
364) -> Vec<String> {
365    let mut errors = Vec::new();
366    if settle_secs == 0 {
367        return errors;
368    }
369    let deadline = tokio::time::Instant::now() + Duration::from_secs(settle_secs);
370    while let Ok(Some(event)) = tokio::time::timeout_at(deadline, frames.recv()).await {
371        match event {
372            ReplicaEvent::FirstFrame(replica, elapsed_ms) => {
373                if let Some(slot) = first_frames.get_mut(replica) {
374                    slot.get_or_insert(elapsed_ms);
375                }
376            }
377            ReplicaEvent::Frame(_, Err(error)) => errors.push(error),
378            ReplicaEvent::Frame(_, Ok(_)) => {}
379        }
380    }
381    while let Ok(event) = frames.try_recv() {
382        match event {
383            ReplicaEvent::FirstFrame(replica, elapsed_ms) => {
384                if let Some(slot) = first_frames.get_mut(replica) {
385                    slot.get_or_insert(elapsed_ms);
386                }
387            }
388            ReplicaEvent::Frame(_, Err(error)) => errors.push(error),
389            ReplicaEvent::Frame(_, Ok(_)) => {}
390        }
391    }
392    errors
393}

395async fn send_frame(
396    frames: &mpsc::Sender<ReplicaEvent>,
397    stop: &mut broadcast::Receiver<()>,
398    value: ReplicaEvent,
399) -> bool {
400    tokio::select! {
401        _ = stop.recv() => false,
402        result = frames.send(value) => result.is_ok(),
403    }

406async fn run_replica(
407    replica: usize,
408    mut session: Box<dyn MarketDataSession>,
409    frames: mpsc::Sender<ReplicaEvent>,
410    mut stop: broadcast::Receiver<()>,
411) {
412    let started = Instant::now();
413    let mut first_frame_sent = false;
414    let mut keepalive = interval(Duration::from_secs(10));
415    keepalive.set_missed_tick_behavior(MissedTickBehavior::Delay);
416    keepalive.tick().await;
417    loop {
418        tokio::select! {
419            _ = stop.recv() => break,
420            _ = keepalive.tick() => {
421                if let Err(failure) = session.send_keepalive().await {
422                    let _ = send_frame(
423                        &frames,
424                        &mut stop,
425                        ReplicaEvent::Frame(
426                            replica,
427                            Err(format!("keepalive failed: {}", failure_text(&failure))),
428                        ),
429                    )
430                    .await;
431                    break;
432                }
433            }
434            frame = session.next_frame() => {
435                let message = match frame {
436                    Ok(frame) => Ok(frame),
437                    Err(failure) => Err(format!("websocket read failed: {}", failure_text(&failure))),
438                };
439                let terminal = message.is_err();
440                if !first_frame_sent && message.is_ok() {
441                    first_frame_sent = true;
442                    let elapsed_ms = started.elapsed().as_secs_f64() * 1_000.0;
443                    if !send_frame(&frames, &mut stop, ReplicaEvent::FirstFrame(replica, elapsed_ms))
444                        .await
445                    {
446                        break;
447                    }
448                }
449                let sent = send_frame(&frames, &mut stop, ReplicaEvent::Frame(replica, message)).await;
450                if terminal || !sent {
451                    break;
452                }
453            }
454        }
455    }
456    session.close().await;
457}

459#[derive(Clone, PartialEq, Eq, Hash)]
460struct EventKey {
461    symbol: String,
462    exchange_ts_us: i64,
463    clock: Option<i64>,
464    detail: EventDetail,
465}
466
467#[derive(Clone, PartialEq, Eq, Hash)]
468enum EventDetail {
469    Snapshot,
470    Level {
471        side: Side,
472        price: i64,
473        change: (u8, i64),
474    },
475    Lifecycle,
476}
477
478fn event_key(event: &ExchangeEvent) -> Option<EventKey> {
479    let (native_symbol, meta, detail) = match event {
480        ExchangeEvent::Snapshot {
481            native_symbol,
482            meta,
483            ..
484        } => (native_symbol, meta, EventDetail::Snapshot),
485        ExchangeEvent::LevelChange {
486            native_symbol,
487            side,
488            price,
489            change,
490            meta,
491        } => (
492            native_symbol,
493            meta,
494            EventDetail::Level {
495                side: *side,
496                price: price.raw(),
497                change: match change {
498                    QuantityChange::Absolute(quantity) => (0, quantity.raw()),
499                    QuantityChange::Delta(quantity) => (1, quantity.raw()),
500                },
501            },
502        ),
503        ExchangeEvent::Lifecycle {
504            native_symbol,
505            meta,
506            ..
507        } => (native_symbol, meta, EventDetail::Lifecycle),
508        ExchangeEvent::StreamReset { .. } => return None,
509    };
510    Some(EventKey {
511        symbol: native_symbol.clone(),
512        exchange_ts_us: meta.exchange_ts_us?,
513        clock: meta.clock,
514        detail,
515    })
516}

518struct SkewTracker {
519    replicas: usize,
520    pending: HashMap<EventKey, Vec<Option<i64>>>,
521    order: VecDeque<EventKey>,
522    first_arrivals: Vec<u64>,
523    matched: u64,
524    dropped: u64,
525    samples: Vec<i64>,
526}
527
528impl SkewTracker {
529    fn new(replicas: usize) -> Self {
530        Self {
531            replicas,
532            pending: HashMap::new(),
533            order: VecDeque::new(),
534            first_arrivals: vec![0; replicas],
535            matched: 0,
536            dropped: 0,
537            samples: Vec::new(),
538        }
539    }
540
541    fn observe(&mut self, key: EventKey, replica: usize, received_at_us: i64) {
542        if self.replicas < 2 || replica >= self.replicas {
543            return;
544        }
545        if !self.pending.contains_key(&key) {
546            while self.pending.len() >= MAX_PENDING_MATCHES {
547                let Some(oldest) = self.order.pop_front() else {
548                    break;
549                };
550                if self.pending.remove(&oldest).is_some() {
551                    self.dropped = self.dropped.saturating_add(1);
552                }
553            }
554            self.pending.insert(key.clone(), vec![None; self.replicas]);
555            self.order.push_back(key.clone());
556            if self.order.len() > MAX_PENDING_MATCHES.saturating_mul(2) {
557                self.order
558                    .retain(|candidate| self.pending.contains_key(candidate));
559            }
560        }
561        let Some(arrivals) = self.pending.get_mut(&key) else {
562            return;
563        };
564        if arrivals[replica].is_some() {
565            return;
566        }
567        arrivals[replica] = Some(received_at_us);
568        if !arrivals.iter().all(Option::is_some) {
569            return;
570        }
571        let times: Vec<i64> = arrivals.iter().filter_map(|value| *value).collect();
572        let earliest = times.iter().copied().min().unwrap_or(received_at_us);
573        let winner = times.iter().position(|value| *value == earliest);
574        if let Some(winner) = winner {
575            self.first_arrivals[winner] = self.first_arrivals[winner].saturating_add(1);
576        }
577        for (index, time) in times.iter().enumerate() {
578            if Some(index) == winner || self.samples.len() >= MAX_SKEW_SAMPLES {
579                continue;
580            }
581            self.samples.push(time.saturating_sub(earliest));
582        }
583        self.matched = self.matched.saturating_add(1);
584        self.pending.remove(&key);
585    }
586}

588fn event_meta(event: &ExchangeEvent) -> Option<&EventMeta> {
589    match event {
590        ExchangeEvent::Snapshot { meta, .. }
591        | ExchangeEvent::LevelChange { meta, .. }
592        | ExchangeEvent::Lifecycle { meta, .. } => Some(meta),
593        ExchangeEvent::StreamReset { .. } => None,
594    }
595}

597fn percentile(sorted: &[i64], percentile: u32) -> Option<i64> {
598    if sorted.is_empty() {
599        return None;
600    }
601    let len = sorted.len() as u64;
602    let rank = len.saturating_mul(u64::from(percentile)).saturating_add(99) / 100;
603    let index = rank.saturating_sub(1).min(len.saturating_sub(1));
604    sorted.get(usize::try_from(index).ok()?).copied()
605}
606
607fn mean(samples: &[i64]) -> Option<i64> {
608    if samples.is_empty() {
609        return None;
610    }
611    let sum: i128 = samples.iter().map(|value| i128::from(*value)).sum();
612    i64::try_from(sum / samples.len() as i128).ok()
613}

615fn failure_text(failure: &SessionFailure) -> String {
616    match failure {
617        SessionFailure::Reconnect(error)
618        | SessionFailure::Protocol { error, .. }
619        | SessionFailure::Fatal(error) => format!("{error:#}"),
620        SessionFailure::Decode { error, .. } => format!("{error:#}"),
621    }
622}
