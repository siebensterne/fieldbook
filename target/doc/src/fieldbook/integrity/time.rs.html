1//! Timestamp conversion helpers.
2//!
3//! Every serialized timestamp comes from the wall clock ([`now_us`]): an
4//! exchange frame's receive time is read when the frame arrives, and a
5//! generated snapshot is stamped when it is produced, so one timeline reaches
6//! the output file. Durations, retries, and timeouts use monotonic timers
7//! (`Instant`/Tokio) instead; they are never serialized. Replay feeds fixture
8//! timestamps through the same paths so the core stays deterministic.
9
10use std::time::{SystemTime, UNIX_EPOCH};
11
12use anyhow::{Context, bail};
13use chrono::{DateTime, Utc};
14
15/// Current wall-clock time as microseconds since the Unix epoch.
16pub fn now_us() -> anyhow::Result<i64> {
17    let elapsed = SystemTime::now()
18        .duration_since(UNIX_EPOCH)
19        .context("system clock predates the Unix epoch")?;
20    i64::try_from(elapsed.as_micros()).context("current timestamp exceeds i64 microseconds")
21}
22
23/// Current wall-clock time as a UTC datetime.
24pub fn now_utc() -> anyhow::Result<DateTime<Utc>> {
25    DateTime::from_timestamp_micros(now_us()?).context("current timestamp outside chrono range")
26}
27
28/// Convert microseconds since the epoch into a UTC datetime.
29pub fn utc_from_us(timestamp_us: i64) -> anyhow::Result<DateTime<Utc>> {
30    DateTime::from_timestamp_micros(timestamp_us)
31        .with_context(|| format!("timestamp {timestamp_us} outside chrono range"))
32}
33
34/// Parse an RFC 3339 timestamp into microseconds.
35pub fn parse_rfc3339_us(input: &str) -> anyhow::Result<i64> {
36    let parsed = DateTime::parse_from_rfc3339(input)
37        .with_context(|| format!("invalid RFC 3339 timestamp `{input}`"))?
38        .with_timezone(&Utc);
39    Ok(parsed.timestamp_micros())
40}
41
42/// Convert epoch milliseconds to microseconds with overflow checking.
43pub fn epoch_ms_to_us(milliseconds: i64) -> anyhow::Result<i64> {
44    milliseconds
45        .checked_mul(1_000)
46        .context("millisecond timestamp overflows microseconds")
47}
48
49/// Convert epoch seconds to microseconds with overflow checking.
50pub fn epoch_secs_to_us(seconds: i64) -> anyhow::Result<i64> {
51    seconds
52        .checked_mul(1_000_000)
53        .context("second timestamp overflows microseconds")
54}
55
56/// Parse a timestamp that may be either Unix milliseconds or RFC 3339.
57pub fn parse_millis_or_rfc3339_us(input: &str) -> anyhow::Result<i64> {
58    let text = input.trim();
59    if text.is_empty() {
60        bail!("empty timestamp");
61    }
62    if text.bytes().all(|byte| byte.is_ascii_digit()) {
63        let milliseconds: i64 = text
64            .parse()
65            .with_context(|| format!("invalid millisecond timestamp `{input}`"))?;
66        return epoch_ms_to_us(milliseconds);
67    }
68    parse_rfc3339_us(text)
69}
70
71/// Format microseconds as RFC 3339 in UTC with microsecond precision.
72#[must_use]
73pub fn format_rfc3339_us(timestamp_us: i64) -> String {
74    utc_from_us(timestamp_us)
75        .map(|timestamp| timestamp.to_rfc3339_opts(chrono::SecondsFormat::Micros, true))
76        .unwrap_or_else(|_| format!("invalid-timestamp({timestamp_us})"))
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    #[test]
84    fn parses_timestamp_forms() {
85        let expected = 1_782_753_357_257_000;
86        assert_eq!(
87            parse_millis_or_rfc3339_us("1782753357257").unwrap(),
88            expected
89        );
90        assert_eq!(
91            parse_millis_or_rfc3339_us("2026-06-29T17:15:57.257000Z").unwrap(),
92            expected
93        );
94    }
95
96    #[test]
97    fn formats_round_trip() {
98        let timestamp = 1_700_000_000_123_456;
99        let text = format_rfc3339_us(timestamp);
100        assert_eq!(parse_rfc3339_us(&text).unwrap(), timestamp);
101    }
