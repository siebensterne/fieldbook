1//! Per-symbol feed-completeness accounting for an aligned SBE capture.
2//!
3//! `verify` answers whether deltas reconstruct the next snapshots exactly.
4//! This report answers a question used by backtests:
5//! **which symbols never became anchored**, how much of the stream was
6//! captured before the first snapshot for each symbol, and whether any
7//! symbol's book state was ever accepted without a complete baseline.
8//!
9//! What this does **not** establish:
10//!
11//! - that the venue delivered every event (a checksum-valid capture can
12//!   still be missing events the exchange dropped upstream);
13//! - that observed receive buffers had no holes (reordering between sources
14//!   and across sockets is visible only through sequence evidence, which is
15//!   what the journal and `verify` record);
16//! - causation for silent intervals. Long quiet gaps are reported as
17//!   observations with their bounds, not as missing data.
18//!
19//! A capture is `valid` when no delta arrives before the symbol's first
20//! snapshot. Unanchored deltas are itemized per symbol in
21//! [`CompletenessSummary::worst_unanchored`] (bounded sample), and the full
22//! per-symbol table is available with `--include-symbols`.
23

34/// Cap on per-symbol entries in [`CompletenessSummary::per_symbol`] and on
35/// the unanchored sample list. Keeps the JSON bounded on universe-scale
36/// captures; the full population still contributes to the aggregate counts.
37const MAX_SAMPLED_SYMBOLS: usize = 1_000;
38/// Cap on the unanchored sample rendered under `worst_unanchored`.
39const MAX_WORST_UNANCHORED: usize = 25;
40/// Silent intervals at or above this length are listed as observations.
41const QUIET_INTERVAL_REPORT_US: i64 = 60_000_000;
42/// Cap on reported quiet intervals so the JSON stays bounded.
43const MAX_QUIET_INTERVALS: usize = 25;
44
45/// Per-symbol completeness ledger.
46#[derive(Debug, Clone, Serialize)]
47pub struct PerSymbolCompleteness {
48    pub symbol: String,
49    /// Receive/exchange micros of the first captured event for the symbol.
50    pub first_event_at_us: Option<i64>,
51    /// Receive/exchange micros of the last captured event for the symbol.
52    pub last_event_at_us: Option<i64>,
53    /// First event observed was a price-level update rather than a snapshot.
54    pub first_event_was_update: bool,
55    pub updates: u64,
56    pub snapshots: u64,
57    /// Deltas counted before the first snapshot arrived for this symbol.
58    pub deltas_before_snapshot: u64,
59    /// Receive micros of the first unanchored delta (null when anchored).
60    pub first_unanchored_at_us: Option<i64>,
61    /// Receive micros of the last unanchored delta (null when anchored).
62    pub last_unanchored_at_us: Option<i64>,
63    /// Longest silence between consecutive events for this symbol, with the
64    /// receive bounds of the interval. Observation only; not missing-data
65    /// evidence.
66    pub longest_quiet_interval_us: i64,
67    /// Receive micros where the longest quiet interval starts (after the
68    /// preceding event). Null when fewer than two events were captured.
69    pub longest_quiet_starts_at_us: Option<i64>,
70}

72/// Aggregate completeness accounting for one capture.
73#[derive(Debug, Clone, Serialize)]
74pub struct CompletenessSummary {
75    pub file: String,
76    pub chunk_size: usize,
77    pub messages: u64,
78    pub updates: u64,
79    pub snapshots: u64,
80    pub symbols: usize,
81    /// Symbols whose first captured event was a price-level update.
82    pub symbols_first_event_update: u64,
83    /// Deltas arriving before the symbol's first snapshot, across the capture.
84    pub deltas_before_snapshot: u64,
85    /// Longest receive-time span between the first and last unanchored delta
86    /// in any single symbol.
87    pub longest_unanchored_span_us: i64,
88    /// Number of symbols with at least one unanchored delta.
89    pub unanchored_symbols: u64,
90    /// Silent intervals of at least 60 s observed, bounded sample.
91    pub quiet_intervals: Vec<QuietInterval>,
92    /// The most severe unanchored symbols, capped at `MAX_WORST_UNANCHORED`.
93    pub worst_unanchored: Vec<PerSymbolCompleteness>,
94    /// Full per-symbol ledger, capped at `MAX_SAMPLED_SYMBOLS` by message
95    /// count, only when `include_symbols` is set.
96    #[serde(skip_serializing_if = "Option::is_none")]
97    pub per_symbol: Option<Vec<PerSymbolCompleteness>>,
98    /// True when `deltas_before_snapshot == 0`.
99    pub valid: bool,
100    pub notes: Vec<String>,
101}

103/// One observed silent interval between consecutive events of a symbol.
104#[derive(Debug, Clone, Serialize)]
105pub struct QuietInterval {
106    pub symbol: String,
107    pub starts_after_us: i64,
108    pub lasts_us: i64,
109}

111struct SymbolState {
112    first_event_at_us: i64,
113    last_event_at_us: i64,
114    last_event_us: i64,
115    longest_quiet_us: i64,
116    longest_quiet_starts_us: i64,
117    first_event_was_update: bool,
118    updates: u64,
119    snapshots: u64,
120    deltas_before_snapshot: u64,
121    anchored: bool,
122    first_unanchored_at_us: Option<i64>,
123    last_unanchored_at_us: Option<i64>,
124}
125
126impl SymbolState {
127    fn observe(&mut self, received_at_us: i64, is_snapshot: bool) {
128        if self.first_event_at_us == 0 {
129            self.first_event_at_us = received_at_us;
130        }
131        self.last_event_at_us = received_at_us;
132        if self.last_event_us != 0 {
133            let gap = received_at_us.saturating_sub(self.last_event_us);
134            if gap > self.longest_quiet_us {
135                self.longest_quiet_us = gap;
136                self.longest_quiet_starts_us = self.last_event_us;
137            }
138        }
139        self.last_event_us = received_at_us;
140        if is_snapshot {
141            self.snapshots += 1;
142            self.anchored = true;
143        } else {
144            self.updates += 1;
145            if !self.anchored {
146                self.deltas_before_snapshot += 1;
147                if self.first_unanchored_at_us.is_none() {
148                    self.first_unanchored_at_us = Some(received_at_us);
149                }
150                self.last_unanchored_at_us = Some(received_at_us);
151            }
152        }
153    }
154}
155
156impl SymbolState {
157    fn into_per_symbol(self, symbol: String) -> PerSymbolCompleteness {
158        PerSymbolCompleteness {
159            symbol,
160            first_event_at_us: (self.first_event_at_us != 0).then_some(self.first_event_at_us),
161            last_event_at_us: (self.last_event_at_us != 0).then_some(self.last_event_at_us),
162            first_event_was_update: self.first_event_was_update,
163            updates: self.updates,
164            snapshots: self.snapshots,
165            deltas_before_snapshot: self.deltas_before_snapshot,
166            first_unanchored_at_us: self.first_unanchored_at_us,
167            last_unanchored_at_us: self.last_unanchored_at_us,
168            longest_quiet_interval_us: self.longest_quiet_us,
169            longest_quiet_starts_at_us: (self.longest_quiet_us > 0)
170                .then_some(self.longest_quiet_starts_us),
171        }
172    }
173}

175/// Produce the completeness report for one capture.
176///
177/// `chunk_size` is either an integer or `auto`. `include_symbols` adds the
178/// full per-symbol ledger, capped at `MAX_SAMPLED_SYMBOLS` entries ranked
179/// by message count.
180pub fn run(
181    input: &Path,
182    chunk_size: &str,
183    include_symbols: bool,
184) -> anyhow::Result<CompletenessSummary> {
185    let mut reader = if chunk_size.eq_ignore_ascii_case("auto") {
186        AlignedSbeReader::<OrderbookSchema>::open_auto(input)
187            .with_context(|| format!("opening `{}` with auto chunk detection", input.display()))?
188    } else {
189        let chunk_size: usize = chunk_size
190            .parse()
191            .with_context(|| format!("invalid chunk size `{chunk_size}`"))?;
192        AlignedSbeReader::<OrderbookSchema>::open_with_chunk_size(input, chunk_size)
193            .with_context(|| format!("opening `{}`", input.display()))?
194    };
195
196    let mut messages: u64 = 0;
197    let mut updates: u64 = 0;
198    let mut snapshots: u64 = 0;
199    let mut states: BTreeMap<String, SymbolState> = BTreeMap::new();
200
201    while let Some(message) = reader.read_msg()? {
202        match message {
203            AnyMessage::Snapshot(decoder) => {
204                let snapshot =
205                    SnapshotDomain::try_from_decoder(decoder).context("decoding SBE snapshot")?;
206                messages += 1;
207                snapshots += 1;
208                let state = states
209                    .entry(snapshot.symbol.clone())
210                    .or_insert_with(|| SymbolState {
211                        first_event_at_us: 0,
212                        last_event_at_us: 0,
213                        last_event_us: 0,
214                        longest_quiet_us: 0,
215                        longest_quiet_starts_us: 0,
216                        first_event_was_update: false,
217                        updates: 0,
218                        snapshots: 0,
219                        deltas_before_snapshot: 0,
220                        anchored: false,
221                        first_unanchored_at_us: None,
222                        last_unanchored_at_us: None,
223                    });
224                state.observe(snapshot.received_at.timestamp_micros(), true);
225            }
226            AnyMessage::PriceLevelUpdate(decoder) => {
227                let update = PriceLevelUpdateDomain::try_from_decoder(decoder)
228                    .context("decoding SBE price-level update")?;
229                messages += 1;
230                updates += 1;
231                let state = states
232                    .entry(update.symbol.clone())
233                    .or_insert_with(|| SymbolState {
234                        first_event_at_us: 0,
235                        last_event_at_us: 0,
236                        last_event_us: 0,
237                        longest_quiet_us: 0,
238                        longest_quiet_starts_us: 0,
239                        first_event_was_update: true,
240                        updates: 0,
241                        snapshots: 0,
242                        deltas_before_snapshot: 0,
243                        anchored: false,
244                        first_unanchored_at_us: None,
245                        last_unanchored_at_us: None,
246                    });
247                state.observe(update.received_at.timestamp_micros(), false);
248            }
249            AnyMessage::Unknown { .. } => {
250                messages += 1;
251            }
252        }
253    }
254
255    // Aggregate and render. Ordering by message count keeps the bounded
256    // per-symbol sample honest and deterministic.
257    let mut per_symbol: Vec<PerSymbolCompleteness> = states
258        .into_iter()
259        .map(|(symbol, state)| state.into_per_symbol(symbol))
260        .collect();
261    per_symbol.sort_by(|left, right| {
262        (right.updates + right.snapshots)
263            .cmp(&(left.updates + left.snapshots))
264            .then_with(|| left.symbol.cmp(&right.symbol))
265    });
266
267    let symbols = per_symbol.len();
268    let symbols_first_event_update = per_symbol
269        .iter()
270        .filter(|entry| entry.first_event_was_update)
271        .count() as u64;
272    let deltas_before_snapshot: u64 = per_symbol
273        .iter()
274        .map(|entry| entry.deltas_before_snapshot)
275        .sum();
276    let unanchored_symbols = per_symbol
277        .iter()
278        .filter(|entry| entry.deltas_before_snapshot > 0)
279        .count() as u64;
280    let longest_unanchored_span_us = per_symbol
281        .iter()
282        .filter_map(
283            |entry| match (entry.first_unanchored_at_us, entry.last_unanchored_at_us) {
284                (Some(first), Some(last)) => Some(last.saturating_sub(first)),
285                _ => None,
286            },
287        )
288        .max()
289        .unwrap_or(0);
290
291    let mut quiet_intervals: Vec<QuietInterval> = per_symbol
292        .iter()
293        .filter(|entry| entry.longest_quiet_interval_us >= QUIET_INTERVAL_REPORT_US)
294        .filter_map(|entry| {
295            entry
296                .longest_quiet_starts_at_us
297                .map(|starts_after_us| QuietInterval {
298                    symbol: entry.symbol.clone(),
299                    starts_after_us,
300                    lasts_us: entry.longest_quiet_interval_us,
301                })
302        })
303        .collect();
304    quiet_intervals.sort_by_key(|interval| std::cmp::Reverse(interval.lasts_us));
305    quiet_intervals.truncate(MAX_QUIET_INTERVALS);
306
307    let mut worst_unanchored: Vec<PerSymbolCompleteness> = per_symbol
308        .iter()
309        .filter(|entry| entry.deltas_before_snapshot > 0)
310        .cloned()
311        .collect();
312    worst_unanchored.sort_by(|left, right| {
313        right
314            .deltas_before_snapshot
315            .cmp(&left.deltas_before_snapshot)
316            .then_with(|| left.symbol.cmp(&right.symbol))
317    });
318    worst_unanchored.truncate(MAX_WORST_UNANCHORED);
319
320    let per_symbol_output = include_symbols.then(|| {
321        per_symbol
322            .into_iter()
323            .take(MAX_SAMPLED_SYMBOLS)
324            .collect::<Vec<_>>()
325    });
326
327    Ok(CompletenessSummary {
328        file: input.display().to_string(),
329        chunk_size: reader.chunk_size(),
330        messages,
331        updates,
332        snapshots,
333        symbols,
334        symbols_first_event_update,
335        deltas_before_snapshot,
336        longest_unanchored_span_us,
337        unanchored_symbols,
338        quiet_intervals,
339        worst_unanchored,
340        per_symbol: per_symbol_output,
341        valid: deltas_before_snapshot == 0,
342        notes: vec![
343            "valid means every delta was anchored to a prior snapshot for its symbol".to_owned(),
344            "unanchored deltas are itemized in worst_unanchored (capped sample) and counted in full in deltas_before_snapshot".to_owned(),
345            "quiet intervals are observations with bounds; they are not, by themselves, evidence of missing data".to_owned(),
346            "captures prove what was recorded, not that the venue delivered every event; see verify for book reconstruction".to_owned(),
347        ],
348    })
349}

368    const SYMBOL_A: &str = "test:alpha";

371    fn timestamp(us: i64) -> DateTime<Utc> {
372        DateTime::from_timestamp_micros(us).expect("timestamp")
373    }
374
375    fn snapshot(symbol: &str, us: i64, bids: &[(i64, i64)], asks: &[(i64, i64)]) -> OutputMessage {
376        OutputMessage::Snapshot(SnapshotDomain {
377            received_at: timestamp(us),
378            exchange_timestamp: timestamp(us),
379            bids: bids
380                .iter()
381                .map(|(price, quantity)| SnapshotBidsEntryDomain {
382                    price_level: *price,
383                    quantity: *quantity,
384                })
385                .collect(),
386            asks: asks
387                .iter()
388                .map(|(price, quantity)| SnapshotAsksEntryDomain {
389                    price_level: *price,
390                    quantity: *quantity,
391                })
392                .collect(),
393            symbol: symbol.to_owned(),
394        })
395    }
396
397    fn update(symbol: &str, us: i64, side: Side, price: i64, quantity: i64) -> OutputMessage {
398        OutputMessage::PriceLevelUpdate(PriceLevelUpdateDomain {
399            received_at: timestamp(us),
400            exchange_timestamp: timestamp(us),
401            price_level: price,
402            quantity,
403            side,
404            symbol: symbol.to_owned(),
405        })
406    }

408    fn write_capture(messages: Vec<OutputMessage>) -> (tempfile::TempDir, PathBuf) {
409        let directory = tempfile::tempdir().expect("tempdir");
410        let path = directory.path().join("capture.sbe");
411        let mut writer = DirectWriter::create(crate::writer::WriterOptions {
412            path: path.to_str().expect("path").to_owned(),
413            chunk_size: 4096,
414            queue: 16,
415            sync_on_close: false,
416            overwrite: false,
417            rotate_bytes: 0,
418            rotate_secs: 0,
419            min_free_bytes: 0,
420            fsync_interval_secs: 0,
421        })
422        .expect("writer");
423        for message in messages {
424            writer.write(&message).expect("write");
425        }
426        writer.shutdown().expect("shutdown");
427        (directory, path)
428    }

430    #[test]
431    fn a_capture_with_every_delta_anchored_is_valid() {
432        let (_directory, path) = write_capture(vec![
433            snapshot(SYMBOL_A, 1, &[(400_000, 10)], &[(600_000, 20)]),
434            update(SYMBOL_A, 2, Side::Buy, 400_000, 15),
435            snapshot(SYMBOL_A, 3, &[(400_000, 15)], &[(600_000, 20)]),
436            update(SYMBOL_A, 4, Side::Sell, 600_000, 0),
437            snapshot(SYMBOL_B, 5, &[(500_000, 30)], &[(700_000, 40)]),
438            update(SYMBOL_B, 6, Side::Buy, 500_000, 35),
439        ]);
440        let summary = run(&path, "auto", true).expect("report");
441        assert!(summary.valid);
442        assert_eq!(summary.messages, 6);
443        assert_eq!(summary.updates, 3);
444        assert_eq!(summary.snapshots, 3);
445        assert_eq!(summary.symbols, 2);
446        assert_eq!(summary.deltas_before_snapshot, 0);
447        assert_eq!(summary.symbols_first_event_update, 0);
448        assert_eq!(summary.unanchored_symbols, 0);
449        let per_symbol = summary.per_symbol.expect("per-symbol ledger");
450        assert_eq!(per_symbol.len(), 2);
451        let beta = per_symbol
452            .iter()
453            .find(|entry| entry.symbol == SYMBOL_B)
454            .expect("beta present");
455        assert_eq!(beta.updates, 1);
456        assert_eq!(beta.snapshots, 1);
457        assert_eq!(beta.deltas_before_snapshot, 0);
458    }

460    #[test]
461    fn unanchored_deltas_are_itemized_per_symbol() {
462        let (_directory, path) = write_capture(vec![
463            update(SYMBOL_A, 10, Side::Buy, 400_000, 15),
464            update(SYMBOL_A, 20, Side::Buy, 400_000, 20),
465            update(SYMBOL_A, 30, Side::Buy, 400_000, 25),
466            update(SYMBOL_A, 40, Side::Sell, 600_000, 0),
467            snapshot(SYMBOL_A, 50, &[(400_000, 25)], &[]),
468            update(SYMBOL_A, 60, Side::Buy, 400_000, 30),
469            update(SYMBOL_B, 5, Side::Buy, 500_000, 1),
470        ]);
471        let summary = run(&path, "auto", true).expect("report");
472        assert!(!summary.valid);
473        assert_eq!(summary.deltas_before_snapshot, 5);
474        assert_eq!(summary.unanchored_symbols, 2);
475        assert_eq!(summary.symbols_first_event_update, 2);
476        assert_eq!(summary.longest_unanchored_span_us, 30);
477        assert_eq!(summary.worst_unanchored.len(), 2);
478        let worst = &summary.worst_unanchored[0];
479        assert_eq!(worst.symbol, SYMBOL_A);
480        assert_eq!(worst.deltas_before_snapshot, 4);
481        assert_eq!(worst.first_unanchored_at_us, Some(10));
482        assert_eq!(worst.last_unanchored_at_us, Some(40));
483        let per_symbol = summary.per_symbol.expect("per-symbol ledger");
484        let beta = per_symbol
485            .iter()
486            .find(|entry| entry.symbol == SYMBOL_B)
487            .expect("beta present");
488        assert_eq!(beta.deltas_before_snapshot, 1);
489        assert_eq!(beta.first_unanchored_at_us, Some(5));
490    }

507    #[test]
508    fn quiet_intervals_are_reported_as_observations_with_bounds() {
509        let (_directory, path) = write_capture(vec![
510            snapshot(SYMBOL_A, 1, &[(400_000, 10)], &[]),
511            update(SYMBOL_A, 2, Side::Buy, 400_000, 11),
