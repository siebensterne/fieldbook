2//! One integrated acquisition slice from reception to replay verification.
3//!
4//! The source declaration and fixture connector feed raw packets into the
5//! journal through the evidence pipeline. The arbiter orders them, the codec
6//! registry decodes them, typed normalization produces domain records, quality
7//! transitions are recorded, and replay from the journal verifies the typed
8//! result. Raw bytes stay in the evidence store; typed records only reference
9//! them.
10

51/// Canonical sink that retains records for replay reconciliation.
52#[derive(Debug, Default)]
53pub struct CollectingSink {
54    records: Mutex<Vec<CanonicalEvidenceRecord>>,
55}
56
57impl CollectingSink {
58    /// Return a clone of the records persisted during the run.
59    pub fn records(&self) -> Vec<CanonicalEvidenceRecord> {
60        self.records
61            .lock()
62            .unwrap_or_else(|poisoned| poisoned.into_inner())
63            .clone()
64    }
65}
66
67impl CanonicalSink for CollectingSink {
68    fn persist(&self, record: &CanonicalEvidenceRecord) -> anyhow::Result<()> {
69        self.records
70            .lock()
71            .unwrap_or_else(|poisoned| poisoned.into_inner())
72            .push(record.clone());
73        Ok(())
74    }
75}

89/// Processing disposition of one raw record.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
91#[serde(rename_all = "snake_case")]
92pub enum ProcessingDisposition {
93    Accepted,
94    Buffered,
95    DuplicateIdentical,
96    DuplicateConflict,
97    Stale,
98    GapRejected,
99    BufferRejected,
100    DecodeRejected,
101    NormalizationRejected,
102}
103
104impl ProcessingDisposition {
105    /// Return the stable contract string for this disposition.
106    pub const fn as_str(self) -> &'static str {
107        match self {
108            Self::Accepted => "accepted",
109            Self::Buffered => "buffered",
110            Self::DuplicateIdentical => "duplicate_identical",
111            Self::DuplicateConflict => "duplicate_conflict",
112            Self::Stale => "stale",
113            Self::GapRejected => "gap_rejected",
114            Self::BufferRejected => "buffer_rejected",
115            Self::DecodeRejected => "decode_rejected",
116            Self::NormalizationRejected => "normalization_rejected",
117        }
118    }
119}

144struct ReplayCollector<'a> {
145    codecs: &'a CodecRegistry,
146    source_id: &'a str,
147    schema_version: u32,
148    arbiter: FeedArbiter,
149    extension_index: BTreeMap<(u64, u64), (String, Option<u64>, FeedChannel)>,
150    evidence_ids: BTreeMap<u64, String>,
151    identities: Vec<(String, String)>,
152}
153
154impl<'a> JournalVisitor for ReplayCollector<'a> {
155    fn visit(&mut self, entry: &JournalEntry<'_>) -> anyhow::Result<()> {
156        let key = (
157            entry.position.segment_index,
158            entry.position.segment_record_ordinal,
159        );
160        let Some((evidence_id, sequence, channel)) = self.extension_index.get(&key).cloned() else {
161            return Ok(());
162        };
163        let sequence = sequence.unwrap_or_else(|| entry.position.segment_record_ordinal + 1);
164        self.evidence_ids.insert(sequence, evidence_id);
165        let packet = SequencedPacket {
166            channel,
167            sequence,
168            payload: entry.payload.to_vec(),
169            received_at_us: entry.received_at_us,
170        };
171        let in_order = packet.clone();
172        let verdict = self.arbiter.submit(packet);
173        let ready = if verdict == ArbiterVerdict::Accepted {
174            vec![in_order]
175        } else {
176            Vec::new()
177        };
178        for replayed in ready.into_iter().chain(self.arbiter.drain_available()) {
179            let replay_id = self
180                .evidence_ids
181                .get(&replayed.sequence)
182                .cloned()
183                .unwrap_or_else(|| format!("replay-{}", replayed.sequence));
184            let events = self.codecs.decode_normalized(
185                &replay_id,
186                self.source_id,
187                self.schema_version,
188                &replayed.payload,
189            )?;
190            for event in events {
191                if let Some(typed) = event.payload {
192                    let body = serde_json::to_string(&typed.payload)
193                        .map_err(|error| anyhow::anyhow!(error.to_string()))?;
194                    self.identities
195                        .push((event.packet_id, sha256_hex(body.as_bytes())));
196                }
197            }
198        }
199        Ok(())
200    }
201}

203/// Replay journal records and reconcile typed digests with canonical records.
204pub fn verify_replay(
205    journal_directory: &std::path::Path,
206    codecs: &CodecRegistry,
207    config: &SliceConfig,
208    canonical: &[CanonicalEvidenceRecord],
209    store: &EvidenceStore,
210    limits: ArbiterLimits,
211) -> anyhow::Result<SliceVerification> {
212    let mut extension_index = BTreeMap::new();
213    for extension in store.load()? {
214        if let Some(position) = extension.position {
215            let channel = if extension.channel == "secondary" {
216                FeedChannel::Secondary
217            } else {
218                FeedChannel::Primary
219            };
220            extension_index.insert(
221                (position.segment_index, position.segment_record_ordinal),
222                (extension.evidence_id, extension.sequence, channel),
223            );
224        }
225    }
226    let mut collector = ReplayCollector {
227        codecs,
228        source_id: &config.source_id,
229        schema_version: config.schema_version,
230        arbiter: FeedArbiter::new(
231            SequenceDomain::new(&config.source_id, 1),
232            1,
233            limits,
234            Box::new(BoundedMemorySink::new(limits.max_rejected_entries)),
235        )?,
236        extension_index,
237        evidence_ids: BTreeMap::new(),
238        identities: Vec::new(),
239    };
240    visit_observations(journal_directory, &mut collector)?;
241    let mut matched: usize = 0;
242    let mut order_mismatches: usize = 0;
243    let common = collector.identities.len().min(canonical.len());
244    for ((evidence_id, digest), expected) in collector.identities.iter().zip(canonical.iter()) {
245        if evidence_id == &expected.evidence_id && digest == &expected.payload_digest {
246            matched = matched.saturating_add(1);
247        } else {
248            order_mismatches = order_mismatches.saturating_add(1);
249        }
250    }
251    let missing = canonical.len().saturating_sub(common);
252    let unexpected = collector.identities.len().saturating_sub(common);
253    let valid = order_mismatches == 0 && missing == 0 && unexpected == 0;
254    Ok(SliceVerification {
255        journal_records: collector.identities.len(),
256        matched,
257        missing,
258        unexpected,
259        corrupt: order_mismatches,
260        order_mismatches,
261        valid,
262    })
263}

278/// Run one deterministic slice through every stage.
279pub fn run_stream_slice(
280    directory: &std::path::Path,
281    run_id: &str,
282    config: &SliceConfig,
283    packets: Vec<SequencedPacket>,
284    codecs: &CodecRegistry,
285    sink: &CollectingSink,
286    limits: ArbiterLimits,
287) -> anyhow::Result<SliceOutcome> {
288    let journal = JournalHandle::start(
289        journal_config(directory),
290        JournalMeta {
291            run_id: run_id.to_owned(),
292            venue: config.source_id.clone(),
293            code_commit: "fixture".to_owned(),
294            route: None,
295            config_sha256: sha256_hex(format!("{config:?}").as_bytes()),
296        },
297    )?;
298    let store = EvidenceStore::new(directory, "slice-evidence.jsonl")?;
299    let pipeline = EvidencePipeline::new(journal, store.clone(), config.profile)?;
300    let domain = SequenceDomain::new(&config.source_id, 1);
301    let mut arbiter = FeedArbiter::new(
302        domain,
303        1,
304        limits,
305        Box::new(BoundedMemorySink::new(limits.max_rejected_entries)),
306    )?;
307    let submitted = packets.len();
308    let mut accepted = 0;
309    let mut duplicates = 0;
310    let mut stale = 0;
311    let mut typed_events = 0;
312    let mut dispositions: BTreeMap<String, u64> = BTreeMap::new();
313    let mut disposition_records = Vec::new();
314    let mut evidence_ids: BTreeMap<u64, String> = BTreeMap::new();
315    let mut quality = Vec::new();
316
317    for (receive_sequence, packet) in packets.into_iter().enumerate() {
318        let context = crate::acquisition::evidence::EvidenceContext {
319            sequence_domain_id: config.source_id.clone(),
320            source_schema_id: "fixture-schema".to_owned(),
321            source_schema_version: config.schema_version,
322            receive_sequence: receive_sequence as u64 + 1,
323        };
324        let evidence = PacketEvidence::new(
325            &config.source_id,
326            packet.channel,
327            config.source.shard as u64,
328            Some(packet.sequence),
329            "fixture-slice",
330            packet.payload.clone(),
331            packet.received_at_us,
332            "lab",
333            "fixture-decoder",
334        );
335        let receipt = pipeline.ingest(&evidence, config.source, &context);
336        evidence_ids
337            .entry(packet.sequence)
338            .or_insert_with(|| receipt.evidence_id.clone());
339        *dispositions
340            .entry(receipt.disposition.as_str().to_owned())
341            .or_default() += 1;
342        let sequence = packet.sequence;
343        let in_order = packet.clone();
344        let verdict = arbiter.submit(packet);
345        let processing_disposition = match &verdict {
346            ArbiterVerdict::Accepted => ProcessingDisposition::Accepted,
347            ArbiterVerdict::OutOfOrder | ArbiterVerdict::Gap(_) => ProcessingDisposition::Buffered,
348            ArbiterVerdict::DuplicateIdentical => ProcessingDisposition::DuplicateIdentical,
349            ArbiterVerdict::DuplicateConflict { .. } => ProcessingDisposition::DuplicateConflict,
350            ArbiterVerdict::Stale => ProcessingDisposition::Stale,
351            ArbiterVerdict::GapExceeded { .. } => ProcessingDisposition::GapRejected,
352            _ => ProcessingDisposition::BufferRejected,
353        };
354        disposition_records.push(SliceDispositionRecord {
355            evidence_id: receipt.evidence_id.clone(),
356            sequence,
357            disposition: processing_disposition,
358        });
359        match &verdict {
360            ArbiterVerdict::Accepted => accepted += 1,
361            ArbiterVerdict::DuplicateIdentical | ArbiterVerdict::DuplicateConflict { .. } => {
362                duplicates += 1;
363            }
364            ArbiterVerdict::Stale => stale += 1,
365            _ => {}
366        }
367        let ready = if matches!(verdict, ArbiterVerdict::Accepted) {
368            vec![in_order]
369        } else {
370            Vec::new()
371        };
372        for drained in ready.into_iter().chain(arbiter.drain_available()) {
373            let drained_id = evidence_ids
374                .get(&drained.sequence)
375                .cloned()
376                .unwrap_or_else(|| receipt.evidence_id.clone());
377            let events = codecs.decode_normalized(
378                &drained_id,
379                &config.source_id,
380                config.schema_version,
381                &drained.payload,
382            )?;
383            for event in events {
384                if let Some(typed) = event.payload {
385                    let state = transition(QualityState::RawCaptured, QualityState::Decoded)?;
386                    let state = transition(state, QualityState::Normalized)?;
387                    let state = transition(state, QualityState::Validated)?;
388                    quality.push(QualityRecord {
389                        event_id: event.packet_id.clone(),
390                        state,
391                        reason: None,
392                        observed_at_us: drained.received_at_us,
393                        verifier: "stream-slice".to_owned(),
394                    });
395                    let body = serde_json::to_string(&typed.payload)
396                        .map_err(|error| anyhow::anyhow!(error.to_string()))?;
397                    sink.persist(&CanonicalEvidenceRecord {
398                        evidence_id: event.packet_id.clone(),
399                        source: config.source_id.clone(),
400                        schema_version: config.schema_version,
401                        kind: event.kind,
402                        payload_type: typed.payload_type,
403                        payload_digest: sha256_hex(body.as_bytes()),
404                    })?;
405                    typed_events += 1;
406                } else {
407                    quality.push(QualityRecord {
408                        event_id: event.packet_id.clone(),
409                        state: QualityState::Rejected,
410                        reason: Some(QualityReason::DecodeFailure),
411                        observed_at_us: drained.received_at_us,
412                        verifier: "stream-slice".to_owned(),
413                    });
414                }
415            }
416        }
417    }
418    pipeline.shutdown(Duration::from_secs(10))?;
419    let run_directory = directory.join(run_id);
420    let evidence = bind_and_verify(&run_directory, &store)?;
421    let verification = verify_replay(
422        &run_directory,
423        codecs,
424        config,
425        &sink.records(),
426        &store,
427        limits,
428    )?;
429    Ok(SliceOutcome {
430        submitted,
431        accepted,
432        duplicates,
433        stale,
434        typed_events,
435        dispositions,
436        disposition_records,
437        quality,
438        verification,
439        evidence_reconciled: evidence.reconciled(),
440    })
441}

484    #[test]
485    fn integrated_slice_reconciles_journal_typed_events_and_replay() {
486        let directory = tempfile::tempdir().expect("tempdir");
487        let config = SliceConfig {
488            source_id: "slice-source".to_owned(),
489            source: SourceId::new(1, 0),
490            schema_version: 1,
491            profile: DurabilityProfile::D1,
492        };
493        let mut codecs = CodecRegistry::new();
494        codecs
495            .register("slice-source", Box::new(SliceCodec))
496            .expect("codec registered");
497        let sink = CollectingSink::default();
498        let packets = vec![
499            packet(
500                1,
501                "{\"trade_id\":\"t1\",\"price_mantissa\":1,\"quantity_mantissa\":2}",
502            ),
503            packet(
504                2,
505                "{\"trade_id\":\"t2\",\"price_mantissa\":3,\"quantity_mantissa\":4}",
506            ),
507        ];
508        let outcome = run_stream_slice(
509            directory.path(),
510            "slice-run",
511            &config,
512            packets,
513            &codecs,
514            &sink,
515            ArbiterLimits::default(),
516        )
517        .expect("slice completed");
518        assert_eq!(outcome.submitted, 2, "both packets submitted");
519        assert_eq!(outcome.accepted, 2, "both packets accepted");
520        assert_eq!(outcome.typed_events, 2, "two typed events produced");
521        assert!(outcome.evidence_reconciled, "evidence extensions reconcile");
522        assert!(
523            outcome.verification.valid,
524            "replay matches canonical records"
525        );
526        assert_eq!(sink.records().len(), 2, "two canonical records persisted");
527    }

550                "{\"trade_id\":\"t3\",\"price_mantissa\":3,\"quantity_mantissa\":4}",

558                2,
559                1_700_000_000_000_004,
560                "{\"trade_id\":\"t2\",\"price_mantissa\":5,\"quantity_mantissa\":6}",
561            ),

563                2,

569                1_700_000_000_000_006,

571            ),
572        ];
573        let outcome = run_stream_slice(
574            directory.path(),
575            "reordered-run",
576            &config,

578            &codecs,

583        let dispositions: Vec<ProcessingDisposition> = outcome
584            .disposition_records
585            .iter()

587            .collect();
588        assert!(

604        assert_eq!(
605            outcome.typed_events, 3,
606            "sequences 1, 2 and 3 produced typed events"

608        let detail = format!("ordered replay mismatch: {:?}", outcome.verification);
609        assert!(outcome.verification.valid, "{detail}");

611            outcome.verification.matched, 3,
612            "every typed event matched in order"

615        let ids: std::collections::BTreeSet<&str> = canonical
616            .iter()
617            .map(|record| record.evidence_id.as_str())
618            .collect();
619        assert_eq!(
620            ids.len(),
621            3,
622            "each drained packet keeps its own raw evidence identity"
623        );
624    }
