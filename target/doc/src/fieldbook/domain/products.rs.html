1//! Typed data-product and source-capability contracts (Platform M3, Phase 1).
2//!
3//! These contracts coexist with `MarketDataSession::apply_subscriptions`
4//! still takes `&BTreeSet<String>`, and `ExchangeCapabilities::new(bool)` still
5//! reports the legacy sequence-clock flag. Nothing here changes the SBE,
6//! journal, replay or catalog formats, and no adapter behavior is altered.
7//!
8//! A [`DataProduct`] names *what* a source can deliver. A [`ProductKind`] is
9//! its stable discriminant and supplies deterministic subscription keys so a
10//! legacy native-symbol set can be converted to a [`SubscriptionSet`] without
11//! migrating `MarketInfo.symbol` or `native_symbol`. A [`SourceCapabilities`]
12//! declaration describes *how* a source delivers those products. A
13//! [`SubscriptionLease`] tracks the local lifecycle of one subscription and
14//! refuses to treat a socket write or a stale connection epoch as a venue
15//! acknowledgement.
16

25/// Scope token for products that are venue- or feed-global rather than bound

29/// Stable discriminant of a [`DataProduct`].
30///
31/// The `as_str` values are the serialized serde names and the key prefixes.
32/// They are part of the contract: changing one changes every subscription key.
33#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
34pub enum ProductKind {
35    #[serde(rename = "orderbook_l2")]
36    OrderBookL2,
37    #[serde(rename = "trades")]
38    Trades,
39    #[serde(rename = "best_bid_offer")]
40    BestBidOffer,
41    #[serde(rename = "asset_context")]
42    AssetContext,
43    #[serde(rename = "venue_lifecycle")]
44    VenueLifecycle,
45    #[serde(rename = "global_statistics")]
46    GlobalStatistics,
47}
48
49impl ProductKind {
50    /// Stable serde and key prefix for this kind.
51    #[must_use]
52    pub const fn as_str(self) -> &'static str {
53        match self {
54            Self::OrderBookL2 => "orderbook_l2",
55            Self::Trades => "trades",
56            Self::BestBidOffer => "best_bid_offer",
57            Self::AssetContext => "asset_context",
58            Self::VenueLifecycle => "venue_lifecycle",
59            Self::GlobalStatistics => "global_statistics",
60        }
61    }
62
63    /// Every declared kind, in a fixed order for deterministic iteration.
64    #[must_use]
65    pub const fn all() -> &'static [Self] {
66        &[
67            Self::OrderBookL2,
68            Self::Trades,
69            Self::BestBidOffer,
70            Self::AssetContext,
71            Self::VenueLifecycle,
72            Self::GlobalStatistics,
73        ]
74    }
75
76    /// Deterministic subscription key for one scope. `scope` may itself
77    /// contain separators (native symbols are namespaced); the fixed prefix
78    /// keeps different kinds distinct.
79    #[must_use]
80    pub fn subscription_key(self, scope: &str) -> String {
81        format!("{}:{}", self.as_str(), scope)
82    }
83}
84
85impl Display for ProductKind {
86    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
87        formatter.write_str(self.as_str())
88    }

91/// One typed product a source can deliver.
92///
93/// Instrument fields are plain `String`s so legacy native symbols remain the
94/// only migration seam; a future `InstrumentId` adapter can feed the same
95/// strings without changing any key. `Display` equals [`DataProduct::key`].
96#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
97pub enum DataProduct {
98    #[serde(rename = "orderbook_l2")]
99    OrderBookL2 { instrument: String },
100    #[serde(rename = "trades")]
101    Trades { instrument: String },
102    #[serde(rename = "best_bid_offer")]
103    BestBidOffer { instrument: String },
104    #[serde(rename = "asset_context")]
105    AssetContext { instrument: Option<String> },
106    #[serde(rename = "venue_lifecycle")]
107    VenueLifecycle { scope: String },
108    #[serde(rename = "global_statistics")]
109    GlobalStatistics { scope: String },
110}
111
112impl DataProduct {
113    /// Build the variant for `kind` with a single scope/instrument value. This
114    /// is the bridge used when converting legacy native symbols.
115    #[must_use]
116    pub fn for_scope(kind: ProductKind, scope: impl Into<String>) -> Self {
117        let scope = scope.into();
118        match kind {
119            ProductKind::OrderBookL2 => Self::OrderBookL2 { instrument: scope },
120            ProductKind::Trades => Self::Trades { instrument: scope },
121            ProductKind::BestBidOffer => Self::BestBidOffer { instrument: scope },
122            ProductKind::AssetContext => Self::AssetContext {
123                instrument: Some(scope),
124            },
125            ProductKind::VenueLifecycle => Self::VenueLifecycle { scope },
126            ProductKind::GlobalStatistics => Self::GlobalStatistics { scope },
127        }
128    }
129
130    #[must_use]
131    pub const fn kind(&self) -> ProductKind {
132        match self {
133            Self::OrderBookL2 { .. } => ProductKind::OrderBookL2,
134            Self::Trades { .. } => ProductKind::Trades,
135            Self::BestBidOffer { .. } => ProductKind::BestBidOffer,
136            Self::AssetContext { .. } => ProductKind::AssetContext,
137            Self::VenueLifecycle { .. } => ProductKind::VenueLifecycle,
138            Self::GlobalStatistics { .. } => ProductKind::GlobalStatistics,
139        }
140    }
141
142    /// Instrument for per-instrument products, `global` for a product without
143    /// an instrument, and the explicit scope for venue/feed-scoped products.
144    #[must_use]
145    pub fn scope(&self) -> &str {
146        match self {
147            Self::OrderBookL2 { instrument }
148            | Self::Trades { instrument }
149            | Self::BestBidOffer { instrument } => instrument.as_str(),
150            Self::AssetContext { instrument } => instrument.as_deref().unwrap_or(GLOBAL_SCOPE),
151            Self::VenueLifecycle { scope } | Self::GlobalStatistics { scope } => scope.as_str(),
152        }
153    }
154
155    /// Stable subscription key: `<kind>:<scope>`.
156    #[must_use]
157    pub fn key(&self) -> String {
158        self.kind().subscription_key(self.scope())
159    }
160}
161
162impl Display for DataProduct {
163    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
164        write!(formatter, "{}:{}", self.kind(), self.scope())
165    }
166}

168/// Deterministic set of subscription keys ready for the existing
169/// `apply_subscriptions(&BTreeSet<String>)` seam.
170#[derive(Debug, Clone, Default, PartialEq, Eq)]
171pub struct SubscriptionSet {
172    keys: BTreeSet<String>,
173}
174
175impl SubscriptionSet {
176    /// Convert a legacy native-symbol set into one product kind's keys.
177    /// Insertion order cannot affect the result, and the keys are directly
178    /// usable as the `BTreeSet<String>` the session trait already accepts.
179    #[must_use]
180    pub fn from_native_symbols(symbols: &BTreeSet<String>, kind: ProductKind) -> Self {
181        Self {
182            keys: symbols
183                .iter()
184                .map(|symbol| kind.subscription_key(symbol))
185                .collect(),
186        }
187    }
188
189    #[must_use]
190    pub fn from_products(products: impl IntoIterator<Item = DataProduct>) -> Self {
191        Self {
192            keys: products.into_iter().map(|product| product.key()).collect(),
193        }
194    }
195
196    #[must_use]
197    pub fn keys(&self) -> &BTreeSet<String> {
198        &self.keys
199    }
200
201    #[must_use]
202    pub fn is_empty(&self) -> bool {
203        self.keys.is_empty()
204    }
205
206    #[must_use]
207    pub fn len(&self) -> usize {
208        self.keys.len()
209    }
210
211    pub fn insert(&mut self, product: DataProduct) {
212        self.keys.insert(product.key());
213    }
214}

307/// Typed declaration of what and how a source delivers.
308///
309/// These are declarations consumed by callers, not instructions that change
310/// any adapter's runtime behavior. `has_sequence_clock` mirrors the legacy
311/// [`crate::exchange::ExchangeCapabilities`] flag; the core keeps reading that
312/// flag unchanged.
313#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
314pub struct SourceCapabilities {
315    /// Live products this source can deliver. Empty means no live feed.
316    pub products: BTreeSet<ProductKind>,
317    /// The source supplies reference data (discovery-time records).
318    pub provides_reference_data: bool,
319    pub update_model: UpdateModel,
320    pub ordering_scope: OrderingScope,
321    pub recovery_mode: RecoveryMode,
322    pub heartbeat: HeartbeatMechanism,
323    pub subscription_topology: SubscriptionTopology,
324    /// Topology of telemetry channels such as venue-wide statistics.
325    pub telemetry_topology: SubscriptionTopology,
326    pub precision: PrecisionModel,
327    pub backpressure: BackpressureBehavior,
328    pub has_sequence_clock: bool,
329}
330
331impl SourceCapabilities {
332    /// Conservative projection of the legacy boolean capability flag. It
333    /// declares no typed products: a legacy adapter made no product claim.
334    #[must_use]
335    pub fn from_legacy(has_sequence_clock: bool) -> Self {
336        Self {
337            products: BTreeSet::new(),
338            provides_reference_data: false,
339            update_model: UpdateModel::SnapshotAndDelta,
340            ordering_scope: if has_sequence_clock {
341                OrderingScope::GlobalSequence
342            } else {
343                OrderingScope::ReceiveFence
344            },
345            recovery_mode: if has_sequence_clock {
346                RecoveryMode::SequenceGapResubscribe
347            } else {
348                RecoveryMode::ResubscribeAndReanchor
349            },
350            heartbeat: HeartbeatMechanism::Disabled,
351            subscription_topology: SubscriptionTopology::PerInstrument,
352            telemetry_topology: SubscriptionTopology::NotApplicable,
353            precision: PrecisionModel::FixedPoint,
354            backpressure: BackpressureBehavior::BoundedFailClosed,
355            has_sequence_clock,
356        }
357    }
358
359    #[must_use]
360    pub fn supports(&self, kind: ProductKind) -> bool {
361        self.products.contains(&kind)
362    }

365/// Local lifecycle of one declared subscription.
366#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
367pub enum SubscriptionState {
368    #[serde(rename = "desired")]
369    Desired,
370    #[serde(rename = "sent")]
371    Sent,
372    #[serde(rename = "acknowledged")]
373    Acknowledged,
374    #[serde(rename = "active")]
375    Active,
376    #[serde(rename = "removing")]
377    Removing,
378    #[serde(rename = "removed")]
379    Removed,
380    #[serde(rename = "failed")]
381    Failed,
382    #[serde(rename = "uncertain")]
383    Uncertain,
384}
385
386impl SubscriptionState {
387    /// Explicit transition table. A socket write (`Sent`) is not
388    /// an acknowledgement, and only a venue acknowledgement can produce
389    /// `Acknowledged`.
390    #[must_use]
391    pub const fn can_transition_to(self, next: Self) -> bool {
392        matches!(
393            (self, next),
394            (Self::Desired, Self::Sent | Self::Removed)
395                | (
396                    Self::Sent,
397                    Self::Acknowledged | Self::Failed | Self::Uncertain | Self::Removed
398                )
399                | (
400                    Self::Acknowledged,
401                    Self::Active | Self::Removing | Self::Uncertain
402                )
403                | (Self::Active, Self::Removing | Self::Uncertain)
404                | (
405                    Self::Removing,
406                    Self::Removed | Self::Active | Self::Uncertain
407                )
408                | (Self::Removed, Self::Desired)
409                | (Self::Failed, Self::Desired | Self::Removed)
410                | (
411                    Self::Uncertain,
412                    Self::Desired | Self::Acknowledged | Self::Removed
413                )
414        )
415    }
416
417    pub fn transition(self, next: Self) -> Result<Self, InvalidSubscriptionTransition> {
418        if self.can_transition_to(next) {
419            Ok(next)
420        } else {
421            Err(InvalidSubscriptionTransition {
422                from: self,
423                to: next,
424            })
425        }
426    }
427}

429impl Display for SubscriptionState {
430    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
431        let text = match self {
432            Self::Desired => "desired",

434            Self::Acknowledged => "acknowledged",

452/// Connection generation a subscription acknowledgement belongs to. It
453/// advances on every disconnect; an ack carrying an older epoch is stale.
454#[derive(
455    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Default, Serialize, Deserialize,
456)]
457#[serde(transparent)]
458pub struct SubscriptionEpoch(pub u64);
459
460impl SubscriptionEpoch {
461    #[must_use]
462    pub const fn new(value: u64) -> Self {
463        Self(value)
464    }
465
466    #[must_use]
467    pub const fn get(self) -> u64 {
468        self.0
469    }
470
471    #[must_use]
472    pub const fn next(self) -> Self {
473        Self(self.0.saturating_add(1))
474    }
475}
476
477impl Display for SubscriptionEpoch {
478    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
479        write!(formatter, "{}", self.0)
480    }
481}

483#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
484pub enum SubscriptionError {
485    #[error(transparent)]
486    Transition(#[from] InvalidSubscriptionTransition),
487    #[error("stale subscription epoch: expected {expected}, observed {observed}")]
488    StaleEpoch {
489        expected: SubscriptionEpoch,
490        observed: SubscriptionEpoch,
491    },
492}
493
494/// One subscription bound to the connection epoch in which it was declared.
495#[derive(Debug, Clone, PartialEq, Eq)]
496pub struct SubscriptionLease {
497    pub state: SubscriptionState,
498    pub epoch: SubscriptionEpoch,
499}
500
501impl SubscriptionLease {
502    #[must_use]
503    pub const fn desired(epoch: SubscriptionEpoch) -> Self {
504        Self {
505            state: SubscriptionState::Desired,
506            epoch,
507        }
508    }
509
510    /// Record a successful local socket write. This is transport progress
511    /// only: it never sets `Acknowledged` and never consults a stale epoch.
512    pub fn socket_write(&mut self) -> Result<(), SubscriptionError> {
513        self.state = self.state.transition(SubscriptionState::Sent)?;
514        Ok(())
515    }
516
517    /// Accept a venue acknowledgement for the current epoch only.
518    pub fn acknowledge(&mut self, epoch: SubscriptionEpoch) -> Result<(), SubscriptionError> {
519        self.require_epoch(epoch)?;
520        self.state = self.state.transition(SubscriptionState::Acknowledged)?;
521        Ok(())
522    }
523
524    /// Accept observed activity for the current epoch only.
525    pub fn activate(&mut self, epoch: SubscriptionEpoch) -> Result<(), SubscriptionError> {
526        self.require_epoch(epoch)?;
527        self.state = self.state.transition(SubscriptionState::Active)?;
528        Ok(())
529    }
530
531    fn require_epoch(&self, observed: SubscriptionEpoch) -> Result<(), SubscriptionError> {
532        if observed == self.epoch {
533            Ok(())
534        } else {
535            Err(SubscriptionError::StaleEpoch {
536                expected: self.epoch,
537                observed,
538            })
539        }
540    }
541
542    /// A transport disconnect invalidates every assumption tied to the old
543    /// epoch, regardless of the prior state: the venue-side subscription is no
544    /// longer known and must be re-declared in the returned new epoch.
545    pub fn disconnect(&mut self) -> SubscriptionEpoch {
546        self.state = SubscriptionState::Uncertain;
547        self.epoch = self.epoch.next();
548        self.epoch
549    }
550}

670                to: S::Active
671            }
672        );
673        assert_eq!(
674            error.to_string(),
675            "invalid subscription transition: sent -> active"
676        );
677
678        let all = [
679            S::Desired,
680            S::Sent,
681            S::Acknowledged,
682            S::Active,
683            S::Removing,
684            S::Removed,
685            S::Failed,
686            S::Uncertain,
687        ];
688        let allowed: BTreeSet<(S, S)> = all
689            .iter()
690            .flat_map(|from| {
691                all.iter()
692                    .filter(move |to| from.can_transition_to(**to))
693                    .map(move |to| (*from, *to))
694            })
695            .collect();
696        assert_eq!(allowed.len(), 20);
697        assert!(allowed.contains(&(S::Uncertain, S::Acknowledged)));
698        assert!(!allowed.contains(&(S::Removed, S::Sent)));
699    }

701    #[test]
702    fn socket_write_does_not_imply_acknowledgement() {
703        let epoch = SubscriptionEpoch::new(7);
704        let mut lease = SubscriptionLease::desired(epoch);
705        assert_eq!(lease.state, SubscriptionState::Desired);
706
707        lease.socket_write().expect("socket write recorded");
708        assert_eq!(lease.state, SubscriptionState::Sent);
709        assert_ne!(lease.state, SubscriptionState::Acknowledged);
710        assert!(!SubscriptionState::Sent.can_transition_to(SubscriptionState::Active));
711
712        // An ack from another epoch cannot turn the write into a venue ack.
713        assert_eq!(
714            lease.acknowledge(epoch.next()),
715            Err(SubscriptionError::StaleEpoch {
716                expected: epoch,
717                observed: epoch.next(),
718            })
719        );
720        assert_eq!(lease.state, SubscriptionState::Sent);
721
722        lease.acknowledge(epoch).expect("current-epoch ack");
723        assert_eq!(lease.state, SubscriptionState::Acknowledged);
724    }

726    #[test]
727    fn disconnect_invalidates_epoch_assumptions() {
728        let epoch = SubscriptionEpoch::new(4);
729        let mut lease = SubscriptionLease::desired(epoch);
730        lease.socket_write().expect("write");
731        lease.acknowledge(epoch).expect("ack");
732        lease.activate(epoch).expect("activity");
733        assert_eq!(lease.state, SubscriptionState::Active);
734
735        let new_epoch = lease.disconnect();
736        assert_eq!(new_epoch, epoch.next());
737        assert_eq!(lease.state, SubscriptionState::Uncertain);
738        assert_eq!(lease.epoch, new_epoch);
739
740        // Every old-epoch assumption is now rejected, including activity that
741        // may have been in flight when the socket was lost.
742        assert!(matches!(
743            lease.acknowledge(epoch),
744            Err(SubscriptionError::StaleEpoch { .. })
745        ));
746        assert!(matches!(
747            lease.activate(epoch),
748            Err(SubscriptionError::StaleEpoch { .. })
749        ));
750        assert_eq!(lease.state, SubscriptionState::Uncertain);
751
752        // Re-declaration belongs to the new epoch; an old-epoch ack still fails.
753        lease.state = lease
754            .state
755            .transition(SubscriptionState::Desired)
756            .expect("re-declare");
757        lease.socket_write().expect("new-epoch write");
758        assert!(lease.acknowledge(epoch).is_err());
759        lease.acknowledge(new_epoch).expect("new-epoch ack");
760        assert_eq!(lease.state, SubscriptionState::Acknowledged);
761    }
