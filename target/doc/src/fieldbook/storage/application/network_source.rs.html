1//! Generic line-oriented network source for the non-market-data application.
2//!
3//! This source is independent of the market-data collector: it speaks a tiny
4//! local protocol (`HELLO <name>` / `TOKEN <declared token>`) over TCP and then
5//! reuses the same [`FileSource`] line parser, duplicate index and resume
6//! anchor used by the file source. The declared token identifies the stream;
7//! it is not a whole-stream hash, so a resume anchor binds to the token rather
8//! than claiming content completeness.
9//!
10//! No credential, proxy, DNS override or venue protocol is involved. Callers
11//! supply an address and a token; tests use a loopback listener.
12

19/// Errors from opening or handshaking a network source.
20#[derive(Debug, Clone, PartialEq, Eq, Error)]
21pub enum NetworkSourceError {
22    #[error("connecting to network source `{address}`: {message}")]
23    Connect { address: String, message: String },
24    #[error("network handshake with `{address}`: {message}")]
25    Handshake { address: String, message: String },
26    #[error("network source `{address}` declared token `{found}`, expected `{expected}`")]
27    TokenMismatch {
28        address: String,
29        expected: String,
30        found: String,
31    },
32}
33
34fn handshake(
35    stream: &mut std::net::TcpStream,
36    name: &str,
37    address: &str,
38    expected_token: &str,
39) -> Result<(), NetworkSourceError> {
40    use std::io::{Read, Write};
41
42    let hello = format!("HELLO {name}\n");
43    stream
44        .write_all(hello.as_bytes())
45        .map_err(|error| NetworkSourceError::Handshake {
46            address: address.to_owned(),
47            message: error.to_string(),
48        })?;
49    let mut reply = Vec::new();
50    let mut byte = [0_u8; 1];
51    while reply.len() < 4096 {
52        let read = stream
53            .read(&mut byte)
54            .map_err(|error| NetworkSourceError::Handshake {
55                address: address.to_owned(),
56                message: error.to_string(),
57            })?;
58        if read == 0 {
59            return Err(NetworkSourceError::Handshake {
60                address: address.to_owned(),
61                message: "connection closed before token".to_owned(),
62            });
63        }
64        if byte[0] == b'\n' {
65            break;
66        }
67        reply.push(byte[0]);
68    }
69    let text = String::from_utf8(reply).map_err(|error| NetworkSourceError::Handshake {
70        address: address.to_owned(),
71        message: format!("token is not UTF-8: {error}"),
72    })?;
73    let found = text
74        .strip_prefix("TOKEN ")
75        .ok_or_else(|| NetworkSourceError::Handshake {
76            address: address.to_owned(),
77            message: format!("unexpected handshake reply `{text}`"),
78        })?;
79    if found != expected_token {
80        return Err(NetworkSourceError::TokenMismatch {
81            address: address.to_owned(),
82            expected: expected_token.to_owned(),
83            found: found.to_owned(),
84        });
85    }
86    Ok(())
87}

89/// Open a fresh network source positioned at the first line.
90pub fn connect(
91    address: &str,
92    name: &str,
93    token: &str,
94) -> Result<FileSource<BufReader<std::net::TcpStream>>, NetworkSourceError> {
95    let mut stream =
96        std::net::TcpStream::connect(address).map_err(|error| NetworkSourceError::Connect {
97            address: address.to_owned(),
98            message: error.to_string(),
99        })?;
100    handshake(&mut stream, name, address, token)?;
101    let identity = SourceIdentity::register_token(name, token);
102    Ok(FileSource::new(BufReader::new(stream), identity))
103}

105/// Reconnect and resume from a verified anchor. The already-committed prefix
106/// is discarded locally from the new stream; the server always sends the whole
107/// declared stream, so no server-side cursor state is trusted.
108pub fn resume(
109    address: &str,
110    name: &str,
111    token: &str,
112    anchor: &DurableCheckpoint,
113    target_id: &str,
114) -> Result<FileSource<BufReader<std::net::TcpStream>>, NetworkSourceError> {
115    let mut stream =
116        std::net::TcpStream::connect(address).map_err(|error| NetworkSourceError::Connect {
117            address: address.to_owned(),
118            message: error.to_string(),
119        })?;
120    handshake(&mut stream, name, address, token)?;
121    let identity = SourceIdentity::register_token(name, token);
122    FileSource::with_resume(BufReader::new(stream), identity, anchor, target_id).map_err(|error| {
123        NetworkSourceError::Handshake {
124            address: address.to_owned(),
125            message: error.to_string(),
126        }
127    })
