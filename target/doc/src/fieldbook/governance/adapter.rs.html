135/// Machine-readable declaration of a source's real capabilities.

165    /// Capability declaration for the document port, when declared.
166    #[serde(default)]

168    /// Capability declaration for the on-chain port, when declared.
169    #[serde(default)]
170    pub on_chain: Option<OnChainCapability>,
171}

173impl SourceCapabilityManifest {
174    /// Validate declaration shape and fail closed on empty coverage.
175    pub fn validate(&self) -> Result<(), ManifestError> {
176        if self.source_id.is_empty() || self.provider.is_empty() {
177            return Err(ManifestError::MissingIdentity);
178        }
179        if self.asset_classes.is_empty() || self.data_categories.is_empty() {
180            return Err(ManifestError::MissingCoverage);
181        }
182        if self.protocols.is_empty() || self.transports.is_empty() {
183            return Err(ManifestError::MissingTransport);
184        }
185        if self.schema_version == 0 {
186            return Err(ManifestError::InvalidSchemaVersion);
187        }
188        validate_modalities(self)?;
189        Ok(())
190    }
191}

193/// Manifest validation failure.
194#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
195pub enum ManifestError {
196    /// Source or provider identity is empty.
197    #[error("source and provider identity are required")]
198    MissingIdentity,
199    /// Asset classes or data categories are empty.
200    #[error("coverage declaration is required")]
201    MissingCoverage,
202    /// Protocol or transport declaration is empty.
203    #[error("protocol and transport declaration are required")]
204    MissingTransport,
205    /// Schema version must be positive.
206    #[error("schema version must be positive")]
207    InvalidSchemaVersion,
208    /// Two different manifests claimed the same source identity.
209    #[error("conflicting manifest for source {0}")]
210    Conflict(String),
211    /// The manifest did not declare any acquisition modality.
212    #[error("at least one acquisition modality must be declared")]
213    MissingModalities,
214    /// A declared modality has no corresponding capability declaration.
215    #[error("modality {0} has no capability declaration")]
216    MissingModalityCapability(AcquisitionModality),
217    /// Declared protocols or transports cannot realize a declared modality.
218    #[error("modality {modality} is inconsistent with declared protocols or transports: {reason}")]
219    InconsistentModality {
220        /// Modality that could not be realized.
221        modality: AcquisitionModality,
222        /// Description of the inconsistency.
223        reason: &'static str,
224    },
225}

255/// Acquisition contract implemented by every source connector.
256#[async_trait]
257pub trait SourceConnector: Send + Sync {
258    /// Return the connector's capability declaration.
259    fn manifest(&self) -> &SourceCapabilityManifest;
260
261    /// Establish the source connection.
262    async fn connect(&mut self) -> Result<()>;
263
264    /// Discover available instruments or information objects.
265    async fn discover(&mut self) -> Result<Vec<String>>;
266
267    /// Subscribe to an explicit object set.
268    async fn subscribe(&mut self, identifiers: &[String]) -> Result<()>;
269
270    /// Recover from a declared discontinuity.
271    async fn recover(&mut self) -> Result<RecoveryReport>;
272
273    /// Report source health with a reason.
274    async fn health(&mut self) -> Result<SourceHealth>;
275
276    /// Close the source connection.
277    async fn disconnect(&mut self) -> Result<()>;
278}

280/// Registry of declared source capabilities.
281#[derive(Debug, Default)]
282pub struct SourceRegistry {
283    manifests: BTreeMap<String, SourceCapabilityManifest>,
284}
285
286impl SourceRegistry {
287    /// Create an empty registry.
288    pub fn new() -> Self {
289        Self::default()
290    }
291
292    /// Register a validated manifest.
293    pub fn register(&mut self, manifest: SourceCapabilityManifest) -> Result<(), ManifestError> {
294        manifest.validate()?;
295        match self.manifests.get(&manifest.source_id) {
296            Some(existing) if existing != &manifest => {
297                Err(ManifestError::Conflict(manifest.source_id.clone()))
298            }
299            Some(_) => Ok(()),
300            None => {
301                self.manifests.insert(manifest.source_id.clone(), manifest);
302                Ok(())
303            }
304        }
305    }
306
307    /// Return a declared manifest by source identity.
308    pub fn resolve(&self, source_id: &str) -> Option<&SourceCapabilityManifest> {
309        self.manifests.get(source_id)
310    }
311}
