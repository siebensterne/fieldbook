2//! Caller-supplied egress intent and active authorization lifecycle types.
3//!
4//! An [`EgressIntent`] is the *actual request* a caller wants to perform. It is
5//! supplied by the caller; the authorization gate never derives one from an
6//! installed proof. Every field is explicit and there is no wildcard default.
7//!
8//! The lifecycle types in this module implement the proposed model in
9//! `NDC-SEMANTIC-MODEL-V2-PROPOSED.md` section 2.5: a rights-registry revision,
10//! an immutable grant snapshot identity, a short-lived lease with a declared
11//! maximum duration, an active session id, reauthorization checkpoints and
12//! fail-closed refresh. Revocation is **not** instantaneous: the maximum
13//! propagation time is the lease duration, because that is how long a previously
14//! granted session can continue before it must refresh its authority.

45/// Declared ceiling on any single authorization lease.
46///
47/// A request may declare a shorter lease; a longer one is rejected when the
48/// grant snapshot is issued. Revocation propagation is bounded by this value
49/// (15 minutes) unless a faster, tested notification channel exists. There is
50/// no instantaneous-revocation claim in this module.
51pub const MAX_LEASE_DURATION: Duration = Duration::from_secs(15 * 60);
52
53/// Maximum interval between two authorization checkpoints in a receive loop.
54///
55/// A continuously ready socket never postpones a checkpoint: the worker checks
56/// its monotonic deadline at the top of every iteration and again after each
57/// read, before the frame is processed. This duration also bounds the time a
58/// revocation can stay unnoticed when the authority responds promptly.
59pub const CHECKPOINT_INTERVAL: Duration = Duration::from_secs(1);
60
61/// Longest permitted single authority refresh at a checkpoint.
62///
63/// A refresh that consumes more monotonic time than this fails closed even if
64/// it eventually returns a valid state; a hung authority cannot hold a session
65/// open past the bound.
66pub const AUTHORITY_REFRESH_TIMEOUT: Duration = Duration::from_millis(500);
67
68/// Assumed upper bound for scheduling the worker's stop action after a denial.
69pub const MAX_SCHEDULER_LATENCY: Duration = Duration::from_millis(250);
70
71/// Assumed upper bound for closing transport and publishing the final audit
72/// state after the stop action is scheduled.
73pub const MAX_CANCEL_LATENCY: Duration = Duration::from_millis(250);
74
75/// Renew a lease when this much monotonic time remains before its deadline.
76pub const LEASE_RENEWAL_MARGIN: Duration = Duration::from_secs(30);
77
78/// Maximum number of HTTP redirect hops followed under one intent before the
79/// caller must submit a fresh intent.
80pub const MAX_REDIRECTS: usize = 3;
81
82fn bounded_sum(parts: impl IntoIterator<Item = Duration>) -> Duration {
83    parts
84        .into_iter()
85        .try_fold(Duration::ZERO, |sum, part| sum.checked_add(part))
86        .unwrap_or(Duration::MAX)
87}
88
89/// Maximum revocation propagation bound `T_stop`.
90///
91/// `T_stop <= T_checkpoint + T_refresh + T_scheduler + T_cancel`, clipped to the
92/// declared lease duration: a lease that expires earlier stops the session by
93/// itself, so no longer bound is claimed. All four terms are monotonic client
94/// policy, not venue guarantees.
95#[must_use]
96pub fn revocation_propagation_bound() -> Duration {
97    bounded_sum([
98        CHECKPOINT_INTERVAL,
99        AUTHORITY_REFRESH_TIMEOUT,
100        MAX_SCHEDULER_LATENCY,
101        MAX_CANCEL_LATENCY,
102    ])
103    .min(MAX_LEASE_DURATION)
104}

106/// The concrete network authority a request will contact.
107///
108/// Scheme, host and port are recorded separately so a redirect from one
109/// authority to another is a different value and must be reauthorized. An IPv6
110/// literal is carried without brackets; [`Self::authority`] restores them.
111#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
112#[serde(deny_unknown_fields)]
113pub struct EndpointIdentity {
114    pub scheme: String,
115    pub host: String,
116    pub port: u16,
117}
118
119impl EndpointIdentity {
120    /// Parse an endpoint from a URL. A URL without a known port is rejected;
121    /// no port is inferred from the scheme by this constructor.
122    ///
123    /// # Errors
124    /// Returns [`RightsError::InvalidEndpoint`] for a URL that has no host or
125    /// no explicit or well-known port.
126    pub fn from_url(url: &str) -> Result<Self, RightsError> {
127        let parsed = Url::parse(url).map_err(|_| RightsError::InvalidEndpoint)?;
128        let host = parsed
129            .host_str()
130            .ok_or(RightsError::InvalidEndpoint)?
131            .strip_prefix('[')
132            .and_then(|value| value.strip_suffix(']'))
133            .unwrap_or_else(|| parsed.host_str().unwrap_or_default())
134            .to_owned();
135        let port = parsed
136            .port_or_known_default()
137            .ok_or(RightsError::InvalidEndpoint)?;
138        Ok(Self {
139            scheme: parsed.scheme().to_ascii_lowercase(),
140            host,
141            port,
142        })
143    }
144
145    /// Render the host and port as a URI authority, restoring IPv6 brackets.
146    #[must_use]
147    pub fn authority(&self) -> String {
148        if self.host.contains(':') {
149            format!("[{}]:{}", self.host, self.port)
150        } else {
151            format!("{}:{}", self.host, self.port)
152        }
153    }
154
155    /// Whether the endpoint is a loopback test endpoint.
156    #[must_use]
157    pub fn is_loopback(&self) -> bool {
158        self.host.eq_ignore_ascii_case("localhost")
159            || self
160                .host
161                .parse::<std::net::IpAddr>()
162                .is_ok_and(|address| address.is_loopback())
163    }
164}

605/// Immutable identity of the grant snapshot a session was authorized against.
606#[derive(Debug, Clone, PartialEq, Eq)]
607pub struct GrantSnapshot {
608    pub registry_revision: u64,
609    pub snapshot_id: String,
610    pub lease_duration_us: u64,
611    pub issued_at_us: i64,
612    /// Stable identity of the lease issued from this snapshot. It is recorded
613    /// in every lifecycle audit event and does not change on renewal.
614    pub lease_id: String,
615}
616
617impl GrantSnapshot {
618    /// Build a snapshot, rejecting a lease longer than the declared maximum.
619    ///
620    /// # Errors
621    /// Returns [`RightsError::LeaseTooLong`] when `lease_duration_us` exceeds
622    /// [`MAX_LEASE_DURATION`] or is zero.
623    pub fn new(
624        registry_revision: u64,
625        snapshot_id: String,
626        lease_duration_us: u64,
627        issued_at_us: i64,
628    ) -> Result<Self, RightsError> {
629        let ceiling = MAX_LEASE_DURATION.as_micros() as u64;
630        if lease_duration_us == 0 || lease_duration_us > ceiling {
631            return Err(RightsError::LeaseTooLong);
632        }
633        let lease_id = format!("lease-{snapshot_id}-{registry_revision}-{issued_at_us}");
634        Ok(Self {
635            registry_revision,
636            snapshot_id,
637            lease_duration_us,
638            issued_at_us,
639            lease_id,
640        })
641    }
642
643    /// The instant at which this snapshot's lease expires.
644    #[must_use]
645    pub fn expires_at_us(&self) -> i64 {
646        self.issued_at_us
647            .saturating_add(i64::try_from(self.lease_duration_us).unwrap_or(i64::MAX))
648    }
649}

651/// A live authorization session bound to one grant snapshot.
652#[derive(Debug, Clone, PartialEq, Eq)]
653pub struct AuthorizationSession {
654    pub session_id: String,
655    pub snapshot: GrantSnapshot,
656    pub expires_at_us: i64,
657    pub renewals: u64,
658    /// Lease identity and epoch carried by lifecycle audit events. The epoch
659    /// advances once per committed renewal; a failed renewal never advances it.
660    pub lease_id: String,
661    pub lease_epoch: u64,
662    /// Authoritative monotonic lease deadline. `None` marks a legacy proof that
663    /// is checked against the wall-clock expiry only.
664    pub lease_deadline_monotonic_us: Option<u64>,
665}

667/// The authority state observed at a reauthorization checkpoint.
668#[derive(Debug, Clone, PartialEq, Eq, Default)]
669pub struct AuthorityState {
670    pub revision: u64,
671    pub valid_snapshots: BTreeSet<String>,
672}
673
674/// Refreshable authority used by the gate at each checkpoint.
675///
676/// A refresh error denies egress: an authority that cannot be consulted is not
677/// an authority that granted anything.
678pub trait AuthorityRefresh: fmt::Debug + Send + Sync {
679    /// Return the current registry revision and valid snapshot identities.
680    ///
681    /// # Errors
682    /// Returns an error when the authority cannot be consulted; the gate fails
683    /// closed on every error.
684    fn refresh(&self) -> anyhow::Result<AuthorityState>;
685}
686
687/// In-process authority used by production and by deterministic tests.
688#[derive(Debug)]
689pub struct InMemoryAuthority {
690    state: Mutex<AuthorityState>,
691    failing: Mutex<bool>,
692}
693
694impl InMemoryAuthority {
695    /// Create an authority holding one valid snapshot at `revision`.
696    #[must_use]
697    pub fn new(revision: u64, snapshot_id: impl Into<String>) -> Self {
698        Self::new_with_snapshots(revision, [snapshot_id.into()])
699    }
700
701    /// Create an authority holding several valid snapshots at `revision`.
702    ///
703    /// A whole operation plan shares one handle, so revoking one snapshot or
704    /// advancing the revision is observed by every step at the next checkpoint.
705    #[must_use]
706    pub fn new_with_snapshots(revision: u64, snapshots: impl IntoIterator<Item = String>) -> Self {
707        Self {
708            state: Mutex::new(AuthorityState {
709                revision,
710                valid_snapshots: snapshots.into_iter().collect(),
711            }),
712            failing: Mutex::new(false),
713        }
714    }
715
716    /// Current registry revision held by this authority.
717    #[must_use]
718    pub fn revision(&self) -> u64 {
719        self.state().revision
720    }
721
722    /// Snapshot identities currently held as valid by this authority.
723    #[must_use]
724    pub fn snapshot_ids(&self) -> BTreeSet<String> {
725        self.state().valid_snapshots.clone()
726    }
727
728    fn state(&self) -> MutexGuard<'_, AuthorityState> {
729        self.state
730            .lock()
731            .unwrap_or_else(std::sync::PoisonError::into_inner)
732    }
733
734    /// Record a new registry revision, invalidating every previous snapshot.
735    pub fn advance_revision(&self, revision: u64, valid: impl IntoIterator<Item = String>) {
736        let mut state = self.state();
737        state.revision = revision;
738        state.valid_snapshots = valid.into_iter().collect();
739    }
740
741    /// Revoke one grant snapshot without changing the revision.
742    pub fn revoke(&self, snapshot_id: &str) {
743        self.state().valid_snapshots.remove(snapshot_id);
744    }
745
746    /// Make every subsequent refresh fail, modelling an unreachable authority.
747    pub fn set_failing(&self, failing: bool) {
748        *self
749            .failing
750            .lock()
751            .unwrap_or_else(std::sync::PoisonError::into_inner) = failing;
752    }
753}
754
755impl AuthorityRefresh for InMemoryAuthority {
756    fn refresh(&self) -> anyhow::Result<AuthorityState> {
757        if *self
758            .failing
759            .lock()
760            .unwrap_or_else(std::sync::PoisonError::into_inner)
761        {
762            anyhow::bail!("authority could not be refreshed");
763        }
764        Ok(self.state().clone())
765    }
766}

778///

785/// Clock pair used by the authorization lifecycle.
786///
787/// Wall-clock microseconds decide *contract validity*: a rights record's
788/// effective interval is a calendar fact. Monotonic microseconds decide *lease
789/// deadlines*: moving the wall clock backwards or forwards can never extend or
790/// shorten a lease. Keeping the two in one injected trait lets offline tests
791/// advance either timeline independently while production uses
792/// [`SystemClock`].
793pub trait Clock: fmt::Debug + Send + Sync {
794    /// Wall-clock microseconds since the Unix epoch.
795    ///
796    /// # Errors
797    /// Returns an error when the system clock cannot be read.
798    fn wall_us(&self) -> anyhow::Result<i64>;
799
800    /// Monotonic microseconds since an arbitrary process-local origin.
801    fn monotonic_us(&self) -> u64;
802}
803
804/// Production clock: system wall time plus a process-monotonic origin.
805#[derive(Debug, Clone, Copy, Default)]
806pub struct SystemClock;
807
808impl Clock for SystemClock {
809    fn wall_us(&self) -> anyhow::Result<i64> {
810        now_us().context("reading the wall clock for authorization")
811    }
812
813    fn monotonic_us(&self) -> u64 {
814        static ORIGIN: LazyLock<Instant> = LazyLock::new(Instant::now);
815        ORIGIN
816            .elapsed()
817            .as_micros()
818            .min(u64::MAX as u128)
819            .try_into()
820            .unwrap_or(u64::MAX)
821    }
822}

824/// Deterministic clock for abstract-value tests and fault injection.
825///
826/// Both timelines start at caller-declared values and change only when a test
827/// advances them. No real waiting or wall-clock adjustment is involved.
828#[derive(Debug, Default)]
829pub struct AbstractClock {
830    wall_us: AtomicI64,
831    monotonic_us: AtomicU64,
832}
833
834impl AbstractClock {
835    /// Create a clock at explicit abstract wall and monotonic instants.
836    #[must_use]
837    pub fn new(wall_us: i64, monotonic_us: u64) -> Self {
838        Self {
839            wall_us: AtomicI64::new(wall_us),
840            monotonic_us: AtomicU64::new(monotonic_us),
841        }
842    }
843
844    /// Set the abstract wall-clock instant.
845    pub fn set_wall_us(&self, value: i64) {
846        self.wall_us.store(value, Ordering::SeqCst);
847    }
848
849    /// Move the abstract wall clock by `delta`, in either direction.
850    pub fn advance_wall_us(&self, delta: i64) {
851        self.wall_us.fetch_add(delta, Ordering::SeqCst);
852    }
853
854    /// Set the abstract monotonic instant.
855    pub fn set_monotonic_us(&self, value: u64) {
856        self.monotonic_us.store(value, Ordering::SeqCst);
857    }
858
859    /// Move only the monotonic timeline forward.
860    pub fn advance_monotonic_us(&self, delta: u64) {
861        self.monotonic_us.fetch_add(delta, Ordering::SeqCst);
862    }
863}
864
865impl Clock for AbstractClock {
866    fn wall_us(&self) -> anyhow::Result<i64> {
867        Ok(self.wall_us.load(Ordering::SeqCst))
868    }
869
870    fn monotonic_us(&self) -> u64 {
871        self.monotonic_us.load(Ordering::SeqCst)
872    }
873}

875/// One authorization decision's two clock readings.
876#[derive(Debug, Clone, Copy, PartialEq, Eq)]
877pub struct DecisionTime {
878    /// Wall-clock instant used for contract validity and audit recording.
879    pub wall_us: i64,
880    /// Monotonic instant used for lease deadlines. `None` keeps the legacy
881    /// wall-clock expiry check for proofs installed without a monotonic lease.
882    pub monotonic_us: Option<u64>,
883}
884
885impl DecisionTime {
886    /// A decision with only a wall-clock reading (legacy preflight callers).
887    #[must_use]
888    pub const fn wall_only(wall_us: i64) -> Self {
889        Self {
890            wall_us,
891            monotonic_us: None,
892        }
893    }
894
895    /// Sample both timelines from an injected clock.
896    ///
897    /// # Errors
898    /// Returns an error when the wall clock cannot be read.
899    pub fn from_clock(clock: &dyn Clock) -> anyhow::Result<Self> {
900        Ok(Self {
901            wall_us: clock.wall_us()?,
902            monotonic_us: Some(clock.monotonic_us()),
903        })
904    }
905
906    /// Whether a lease is over at this decision instant. A monotonic deadline,
907    /// when present, is authoritative; wall-clock expiry is the compatibility
908    /// fallback and is never allowed to extend a monotonic lease.
909    #[must_use]
910    pub fn lease_expired(&self, wall_expiry_us: i64, monotonic_deadline_us: Option<u64>) -> bool {
911        match (self.monotonic_us, monotonic_deadline_us) {
912            (Some(now), Some(deadline)) => now >= deadline,
913            _ => self.wall_us > wall_expiry_us,
914        }
915    }
916}
