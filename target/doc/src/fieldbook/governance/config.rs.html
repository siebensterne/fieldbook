29    pub snapshot_interval_secs: u64,
30    /// Market discovery reconciliation cadence.
31    pub discovery_interval_secs: u64,
32    /// Refresh a source connection after this long, with jitter.
33    pub reconnect_after_secs: u64,
34    /// Reconnect if no frame (data or heartbeat) arrives for this long.
35    pub connection_idle_timeout_secs: u64,
36    /// Probe a quiet symbol after this long without an event.
37    pub stale_probe_after_secs: u64,
38    /// Minimum spacing between probes for one symbol.
39    pub probe_interval_secs: u64,
40    /// Failed probes before recording a symbol continuity incident and
41    /// issuing another symbol-scoped refresh.
42    pub max_probes: u32,
43    /// Emit final snapshots for live books during graceful shutdown.
44    pub final_snapshots: bool,
45    /// Bounded writer queue depth, in messages.
46    pub writer_queue: usize,
47    /// Flush cadence for partial SBE chunks.
48    pub flush_interval_secs: u64,
49    /// Cap on selected venue market records (0 means no cap). The cap is
50    /// applied between discovery records, never inside one record's outcome
51    /// set, so a multi-outcome record is never silently truncated.
52    pub max_markets: usize,
53    /// Dedicated market-lifecycle control-plane connections. Book shards carry
54    /// only order-book subscriptions; this many supervised lifecycle sockets
55    /// carry lifecycle traffic. An adapter without a separate lifecycle channel
56    /// ignores this value and does not charge it to its connection budget.
57    /// Set to 0 to disable the lifecycle plane.
58    pub lifecycle_replicas: usize,
59    /// Optional run duration; 0 means run until SIGINT/SIGTERM.
60    pub duration_secs: u64,
61    /// Maximum accepted outbound writer backlog before the service fails.
62    pub writer_send_timeout_secs: u64,
63    /// Hard ceiling on `shards * replicas` websocket connections.
64    pub max_total_connections: usize,
65    /// Hard ceiling on the estimated bytes of one initial subscription frame.
66    pub max_initial_subscription_bytes: usize,
67    /// Hard ceiling on estimated resident memory for book state, in MiB.
68    pub max_estimated_book_memory_mb: u64,
69    /// Hard ceiling on estimated snapshot output, in bytes per second.
70    pub max_snapshot_bytes_per_sec: u64,
71    /// Maximum concurrent websocket connection attempts.
72    pub connect_concurrency: usize,
73    /// Most recent accepted catalog checkpoint. `None` disables persistence.
74    pub catalog_path: Option<PathBuf>,
75    /// Catalog bootstrap policy.
76    pub catalog_mode: CatalogMode,
77    /// Maximum accepted catalog age for `warm` and `require`.
78    pub catalog_max_age_secs: u64,
79    /// Accept a complete discovery round that retires every symbol.
80    pub allow_empty_catalog: bool,
81    /// Minimum anchored-symbol fraction required by `/readyz`.
82    pub readiness_min_anchored_fraction: f64,
83    /// Maximum catalog age required by `/readyz`.
84    pub readiness_max_catalog_age_secs: u64,
85    /// Maximum time since a book's last valid update before `/readyz` fails.
86    pub readiness_max_stale_secs: u64,
87}

89/// How a persisted catalog is used during startup.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize, ValueEnum)]
91#[serde(rename_all = "lowercase")]
92pub enum CatalogMode {
93    /// Discover before starting workers; never read a checkpoint.
94    #[default]
95    Off,
96    /// Start from a fresh checkpoint when available, then refresh in the
97    /// background. A missing, stale, or empty checkpoint falls back to
98    /// discovery.
99    Warm,
100    /// Require a valid, fresh checkpoint; fail startup otherwise.
101    Require,
102    /// Require any valid, non-empty checkpoint regardless of age. This is the
103    /// explicit degraded boot mode for a total discovery outage.
104    Stale,
105}

206/// Transport controls shared by every websocket adapter.
207///
208/// These options exist for latency experiments as well as production tuning.
209/// The defaults are conservative: Nagle's algorithm is disabled because
210/// order-book frames are small and latency-sensitive, and a single connection
211/// attempt is made.
212#[derive(Debug, Clone, Serialize, Deserialize)]
213#[serde(default, deny_unknown_fields)]
214pub struct NetworkSettings {
215    /// Set `TCP_NODELAY` on every accepted websocket TCP socket.
216    pub tcp_nodelay: bool,
217    /// Optional local IP to bind before connecting. The address family must
218    /// match the resolved remote address.
219    pub source_ip: Option<IpAddr>,
220    /// Optional fixed local source port, used as a base: connection
221    /// `shard * 256 + replica` binds `base + ordinal`, so parallel
222    /// connections never collide. Cannot be combined with `connect_race > 1`,
223    /// because racing attempts for one connection would share one port.
224    pub source_port: Option<u16>,
225    /// Number of simultaneous TCP/TLS/websocket handshakes. The first
226    /// successful connection is retained and the rest are closed. One means
227    /// no racing.
228    pub connect_race: usize,
229    /// Deadline for one full connection attempt: DNS resolution, TCP connect,
230    /// TLS negotiation, and the websocket upgrade, including every raced
231    /// candidate. A stalled handshake fails like any other transient error
232    /// instead of wedging the source.
233    pub connect_timeout_secs: u64,
234    /// Egress route for collection traffic: `direct` (default) or
235    /// `explicit_proxy`. An explicit proxy is used only for a route the
236    /// account holder is permitted to use; it must be authorized by the
237    /// venue's terms and the account holder's eligibility and applicable
238    /// law. There is never a silent fallback to direct networking.
239    pub egress_mode: String,
240    /// Environment variable holding the proxy URL
241    /// (`http://[user[:password]@]host[:port]` for CONNECT, `socks5://...`,
242    /// or `socks5h://...` for proxy-resolved DNS). Read only when
243    /// `egress_mode = "explicit_proxy"`; never stored in configuration.
244    pub proxy_url_env: String,
245    /// Environment variable holding the proxy username; takes effect only
246    /// together with `proxy_auth_password_env`.
247    pub proxy_auth_user_env: String,
248    /// Environment variable holding the proxy password; takes effect only
249    /// together with `proxy_auth_user_env`.
250    pub proxy_auth_password_env: String,
251    /// SOCKS5 only: when true, the proxy resolves the venue hostname. For
252    /// `socks5h://` this is forced. HTTP CONNECT always uses remote DNS by
253    /// construction of the tunnel target.
254    pub proxy_remote_dns: bool,
255    /// Optional non-secret operator label for evidence manifests, for example
256    /// the name of the approved egress path. Recorded verbatim; credentials
257    /// must never be placed here.
258    pub proxy_label: Option<String>,
259    /// Venue this transport serves. Supplied by the caller at settings load
260    /// time; never read from a file or environment. A standalone transport
261    /// uses a loopback-only test identity; an empty venue denies.
262    #[serde(skip)]
263    pub venue: String,
264    /// Runtime-injected caller intent for provider egress. Never read from the
265    /// config file: a self-attested declaration cannot grant a right.
266    #[serde(skip)]
267    pub declared_intent: Option<crate::egress_intent::EgressIntent>,
268}

292impl NetworkSettings {
293    /// Convert validated settings into transport options.
294    ///
295    /// # Errors
296    ///
297    /// Fails when `explicit_proxy` is configured but the proxy environment is
298    /// missing or invalid; the caller must not proceed. This never falls
299    /// back to direct networking.
300    pub fn connect_options(&self) -> anyhow::Result<crate::ws::ConnectOptions> {
301        let egress = crate::egress::EgressRoute::from_network(self)?;
302        Ok(crate::ws::ConnectOptions {
303            tcp_nodelay: self.tcp_nodelay,
304            source_ip: self.source_ip,
305            source_port: self.source_port,
306            attempts: self.connect_race,
307            connect_timeout: std::time::Duration::from_secs(self.connect_timeout_secs.max(1)),
308            egress,
309            trace: None,
310            venue: self.venue.clone(),
311            // Settings-derived sessions capture a feed. Probes override this
312            // to `source_contact` at their own call site.
313            operation: crate::rights::Operation::RealTimeCapture,
314            intent: self.declared_intent.clone(),
315            denial_transport: None,
316        })
317    }

417impl Settings {
418    /// Built-in defaults with an explicit non-secret venue label, without
419    /// reading configuration files or environment variables. Offline tests
420    /// and replay use this so their behavior does not depend on a shell.
421    #[must_use]
422    pub fn for_venue(venue: &str) -> Self {
423        let file = FileConfig::default();
424        let mut settings = Self {
425            governance: file.governance,
426            collector: file.collector,
427            network: file.network,
428            output: file.output,
429            metrics: file.metrics,
430            journal: file.journal,
431        };
432        settings.network.venue = venue.to_owned();
433        settings
434    }
435
436    pub fn load(path: Option<&Path>) -> anyhow::Result<Self> {
437        let file = match path {
438            Some(path) => {
439                let text = std::fs::read_to_string(path)
440                    .with_context(|| format!("reading config `{}`", path.display()))?;
441                toml::from_str::<FileConfig>(&text)
442                    .with_context(|| format!("parsing config `{}`", path.display()))?
443            }
444            None => FileConfig::default(),
445        };
446        let mut settings = Self {
447            governance: file.governance,
448            collector: file.collector,
449            network: file.network,
450            output: file.output,
451            metrics: file.metrics,
452            journal: file.journal,
453        };
454        settings.apply_environment()?;
455        let base = path.and_then(Path::parent);
456        settings.load_declared_intent(base)?;
457        settings.load_declared_plan(base)?;
458        settings.network.declared_intent = settings.governance.declared_intent.clone();
459        settings.validate()?;
460        Ok(settings)
461    }
462
463    /// Load the caller-declared egress intent from its separate file.
464    ///
465    /// A missing or malformed declaration is an error: the command stays
466    /// fail-closed. No intent value is synthesized from settings or commands.
467    fn load_declared_intent(&mut self, base: Option<&Path>) -> anyhow::Result<()> {
468        let Some(path) = self.governance.egress_intent_file.clone() else {
469            return Ok(());
470        };
471        let path = if path.is_absolute() {
472            path
473        } else {
474            base.map_or(path.clone(), |directory| directory.join(&path))
475        };
476        self.governance.declared_intent = Some(crate::egress_intent::EgressIntent::load(&path)?);
477        Ok(())
478    }
479
480    /// Load the caller-declared operation plan from its separate file. A
481    /// missing or invalid plan is an error: no command starts without a
482    /// complete, declared plan. No step is inferred from a command name.
483    ///
484    /// # Errors
485    /// Returns an error for a missing, malformed or incomplete plan file.
486    pub fn load_declared_plan(&mut self, base: Option<&Path>) -> anyhow::Result<()> {
487        let Some(path) = self.governance.operation_plan_file.clone() else {
488            return Ok(());
489        };
490        let path = if path.is_absolute() {
491            path
492        } else {
493            base.map_or(path.clone(), |directory| directory.join(&path))
494        };
495        self.governance.declared_plan = Some(
496            crate::governance::operation_plan::OperationPlan::load(&path)?,
497        );
498        Ok(())
499    }
500
501    fn apply_environment(&mut self) -> anyhow::Result<()> {
502        if let Ok(value) = std::env::var("COLLECTOR_OUTPUT") {
503            self.output.path = value;
504        }
505        if let Ok(value) = std::env::var("COLLECTOR_JOURNAL_DIR") {
506            self.journal.directory = Some(PathBuf::from(value));
507        }
508        if let Ok(value) = std::env::var("COLLECTOR_METRICS_LISTEN") {
509            self.metrics.listen = Some(
510                value
511                    .parse()
512                    .with_context(|| format!("invalid COLLECTOR_METRICS_LISTEN `{value}`"))?,
513            );
514        }
515        Ok(())
516    }
517
518    /// SHA-256 of the capture-affecting configuration.
519    ///
520    /// The digest covers the collector, network, output, and journal
521    /// sections. Credentials are excluded so the value can be published in a
522    /// journal manifest without exposing secrets.
523    pub fn capture_config_sha256(&self) -> anyhow::Result<String> {
524        let canonical = serde_json::json!({
525            "collector": self.collector,
526            "network": self.network,
527            "output": self.output,
528            "journal": self.journal,
529        });
530        let bytes = serde_json::to_vec(&canonical)
531            .context("serializing capture configuration for hashing")?;
532        Ok(crate::util::sha256_hex(&bytes))
533    }

535    pub fn validate(&self) -> anyhow::Result<()> {
536        anyhow::ensure!(self.collector.shards <= 256, "shards must be <= 256");
537        anyhow::ensure!(self.collector.replicas >= 1, "replicas must be >= 1");
538        anyhow::ensure!(self.collector.replicas <= 4, "replicas must be <= 4");
539        anyhow::ensure!(
540            self.collector.lifecycle_replicas <= 4,
541            "lifecycle_replicas must be <= 4"
542        );
543        anyhow::ensure!(
544            self.collector.max_symbols_per_socket > 0,
545            "max_symbols_per_socket must be > 0"
546        );
547        anyhow::ensure!(
548            self.collector.max_shards >= self.collector.shards.max(1),
549            "max_shards must be >= shards"
550        );
551        anyhow::ensure!(
552            self.collector.snapshot_interval_secs >= 1,
553            "snapshot_interval_secs must be >= 1"
554        );
555        anyhow::ensure!(
556            self.collector.discovery_interval_secs >= 1,
557            "discovery_interval_secs must be >= 1"
558        );
559        anyhow::ensure!(
560            self.collector.reconnect_after_secs >= 10,
561            "reconnect_after_secs must be >= 10"
562        );
563        anyhow::ensure!(
564            self.collector.connection_idle_timeout_secs >= 10,
565            "connection_idle_timeout_secs must be >= 10"
566        );
567        anyhow::ensure!(
568            self.collector.stale_probe_after_secs >= 10,
569            "stale_probe_after_secs must be >= 10"
570        );
571        anyhow::ensure!(
572            self.collector.probe_interval_secs >= 1,
573            "probe_interval_secs must be >= 1"
574        );
575        anyhow::ensure!(self.collector.max_probes >= 1, "max_probes must be >= 1");
576        anyhow::ensure!(
577            self.collector.max_total_connections >= self.collector.replicas.max(1),
578            "max_total_connections must be >= replicas"
579        );
580        anyhow::ensure!(
581            self.collector.max_total_connections <= 4096,
582            "max_total_connections must be <= 4096"
583        );
584        anyhow::ensure!(
585            self.collector.max_initial_subscription_bytes >= 1024,
586            "max_initial_subscription_bytes must be >= 1024"
587        );
588        anyhow::ensure!(
589            self.collector.max_estimated_book_memory_mb >= 1,
590            "max_estimated_book_memory_mb must be >= 1"
591        );
592        anyhow::ensure!(
593            self.collector.max_snapshot_bytes_per_sec >= 1024,
594            "max_snapshot_bytes_per_sec must be >= 1024"
595        );
596        anyhow::ensure!(
597            (1..=256).contains(&self.collector.connect_concurrency),
598            "connect_concurrency must be in [1, 256]"
599        );
600        anyhow::ensure!(
601            self.collector.catalog_mode == CatalogMode::Off
602                || self
603                    .collector
604                    .catalog_path
605                    .as_deref()
606                    .is_some_and(|path| !path.as_os_str().is_empty()),
607            "catalog_mode {:?} requires collector.catalog_path",
608            self.collector.catalog_mode

723/// Validate that the capture and every sidecar resolve to distinct files.
724///
725/// A string comparison is insufficient: `./capture.sbe`, `capture.sbe`, and an
726/// absolute path can all name the same file. Existing paths are canonicalised
727/// (resolving symlinks), nonexistent paths are made absolute lexically, and
728/// existing hard links are compared by device/inode where the platform exposes
729/// them. This prevents a sidecar write from truncating the capture after a run.
730fn validate_output_paths(settings: &Settings) -> anyhow::Result<()> {
731    let mut paths: Vec<(&'static str, PathBuf)> = Vec::new();
732    if settings.output.path != "-" {
733        let output = PathBuf::from(&settings.output.path);
734        ensure_not_directory("output.path", &output)?;
735        paths.push(("output.path", output));
736        if settings.output.rotate_bytes > 0 || settings.output.rotate_secs > 0 {
737            paths.push((
738                "output manifest",
739                rotation_manifest_path(&settings.output.path),
740            ));
741        }
742    }
743    for (name, path) in [
744        ("metadata_path", settings.output.metadata_path.as_deref()),
745        ("report_path", settings.output.report_path.as_deref()),
746        ("catalog_path", settings.collector.catalog_path.as_deref()),
747    ] {
748        let Some(path) = path.filter(|path| !path.as_os_str().is_empty()) else {
749            continue;
750        };
751        anyhow::ensure!(
752            path.as_os_str() != "-",
753            "{name} must be a file path, not stdout"
754        );
755        ensure_not_directory(name, path)?;
756        paths.push((name, path.to_path_buf()));
757    }
758    for left in 0..paths.len() {
759        for right in left + 1..paths.len() {
760            let (left_name, left_path) = &paths[left];
761            let (right_name, right_path) = &paths[right];
762            if paths_alias(left_path, right_path)? {
763                bail!("{left_name} and {right_name} refer to the same file");
764            }
765        }
766    }
767    Ok(())
768}

770/// Manifest path for a rotated capture. Segments are named
771/// `<output>.<sequence>.sbe`, so the manifest sits beside them without
772/// colliding with the unrotated capture path.
773#[must_use]
774pub fn rotation_manifest_path(output_path: &str) -> PathBuf {
775    let path = Path::new(output_path);
776    let name = path
777        .file_name()
778        .and_then(|name| name.to_str())
779        .unwrap_or("output");
780    path.with_file_name(format!("{name}.manifest.json"))
781}

783/// Reject every pre-existing artifact that this run would replace.
784///
785/// This is a startup check, not a configuration-syntax check: discovery,
786/// replay, and latency commands may legitimately share a config while writing
787/// somewhere else. Collection and replay call it before doing any work, so a
788/// failed startup leaves the previous capture and sidecars untouched.
789pub fn preflight_output(settings: &Settings) -> anyhow::Result<()> {
790    if settings.output.overwrite {
791        return Ok(());
792    }
793    let mut artifacts: Vec<(&'static str, PathBuf)> = Vec::new();
794    if settings.output.path != "-" {
795        artifacts.push(("output.path", PathBuf::from(&settings.output.path)));
796        if settings.output.rotate_bytes > 0 || settings.output.rotate_secs > 0 {
797            artifacts.push((
798                "output manifest",
799                rotation_manifest_path(&settings.output.path),
800            ));
801            for segment in rotation_segments(&settings.output.path)? {
802                artifacts.push(("output segment", segment));
803            }
804        }
805    }
806    for (name, path) in [
807        ("metadata_path", settings.output.metadata_path.as_deref()),
808        ("report_path", settings.output.report_path.as_deref()),
809    ] {
810        if let Some(path) = path.filter(|path| !path.as_os_str().is_empty()) {
811            artifacts.push((name, path.to_path_buf()));
812        }
813    }
814    // In warm, require, and stale modes the catalog is an input. Replacing it
815    // after a complete discovery round is the intended behavior, not data
816    // loss. In cold-only mode an existing checkpoint is an output artifact and
817    // follows the same overwrite rule as the capture.
818    if settings.collector.catalog_mode == CatalogMode::Off
819        && let Some(path) = settings
820            .collector
821            .catalog_path
822            .as_deref()
823            .filter(|path| !path.as_os_str().is_empty())
824    {
825        artifacts.push(("catalog_path", path.to_path_buf()));
826    }
827    for (name, path) in artifacts {
828        anyhow::ensure!(
829            !path.exists(),
830            "{name} already exists at `{}`; refusing to replace it. \
831             Set output.overwrite = true to replace existing artifacts",
832            path.display()
833        );
834    }
835    Ok(())
836}

838/// Existing finalized or partial segments for a rotated capture.
839fn rotation_segments(output_path: &str) -> anyhow::Result<Vec<PathBuf>> {
840    let path = Path::new(output_path);
841    let (Some(parent), Some(name)) = (path.parent(), path.file_name().and_then(|n| n.to_str()))
842    else {
843        return Ok(Vec::new());
844    };
845    let parent = if parent.as_os_str().is_empty() {
846        Path::new(".")
847    } else {
848        parent
849    };
850    let prefix = format!("{name}.");
851    let entries = std::fs::read_dir(parent)
852        .with_context(|| format!("scanning rotation directory `{}`", parent.display()))?;
853    Ok(entries
854        .flatten()
855        .filter_map(|entry| {
856            let file_name = entry.file_name();
857            let file_name = file_name.to_str()?;
858            rotation_artifact_name(&prefix, file_name).then(|| entry.path())
859        })
860        .collect())
861}
862
863fn rotation_artifact_name(prefix: &str, name: &str) -> bool {
864    let stem = name.strip_prefix(prefix).and_then(|value| {
865        value
866            .strip_suffix(".sbe")
867            .or_else(|| value.strip_suffix(".sbe.partial"))
868    });
869    stem.is_some_and(|sequence| {
870        sequence.len() == 6 && sequence.bytes().all(|byte| byte.is_ascii_digit())
871    })
872}

874fn ensure_not_directory(name: &str, path: &Path) -> anyhow::Result<()> {
875    if let Ok(metadata) = std::fs::metadata(path) {
876        anyhow::ensure!(
877            !metadata.is_dir(),
878            "{name} points to a directory: `{}`",
879            path.display()
880        );
881    }
882    Ok(())
883}
884
885fn paths_alias(left: &Path, right: &Path) -> anyhow::Result<bool> {
886    if left == right {
887        return Ok(true);
888    }
889    let left_normal = comparison_path(left)?;
890    let right_normal = comparison_path(right)?;
891    if left_normal == right_normal {
892        return Ok(true);
893    }
894    Ok(same_existing_file(left, right))
895}
896
897/// Canonical form for alias detection. Nonexistent files still compare
898/// correctly when their parent directory exists, including through symlinked
899/// directories.
900fn comparison_path(path: &Path) -> anyhow::Result<PathBuf> {
901    if let Ok(canonical) = std::fs::canonicalize(path) {
902        return Ok(canonical);
903    }
904    let absolute = std::path::absolute(path)
905        .with_context(|| format!("making `{}` absolute", path.display()))?;
906    let normalized = normalize_lexically(&absolute);
907    let Some(parent) = normalized.parent() else {
908        return Ok(normalized);
909    };
910    let Some(file_name) = normalized.file_name() else {
911        return Ok(normalized);
912    };
913    match std::fs::canonicalize(parent) {
914        Ok(parent) => Ok(parent.join(file_name)),
915        Err(_) => Ok(normalized),
916    }
917}
918
919fn normalize_lexically(path: &Path) -> PathBuf {
920    let mut normalized = PathBuf::new();
921    for component in path.components() {
922        match component {
923            Component::CurDir => {}
924            Component::ParentDir => {
925                normalized.pop();
926            }
927            other => normalized.push(other.as_os_str()),
928        }
929    }
930    normalized
931}

933#[cfg(unix)]
934fn same_existing_file(left: &Path, right: &Path) -> bool {
935    use std::os::unix::fs::MetadataExt as _;
936
937    match (std::fs::metadata(left), std::fs::metadata(right)) {
938        (Ok(left), Ok(right)) => left.dev() == right.dev() && left.ino() == right.ino(),
939        _ => false,
940    }
941}
942
943#[cfg(not(unix))]
944fn same_existing_file(_left: &Path, _right: &Path) -> bool {
945    false
