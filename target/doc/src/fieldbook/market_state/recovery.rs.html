1//! Typed recovery decisions (M3.1).
2//!
3//! Recovery policy is a pure function of observed conditions. A cause is
4//! evidence, not an instruction: the policy decides which action, if any, a
5//! cause authorizes. Keeping this decision separate from the collector state
6//! machine makes every recovery choice explainable and independently testable,
7//! and prevents a quiet instrument from being silently escalated into a
8//! source-wide reconnect.
9

19/// The action a recovery decision authorizes.
20#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
21#[serde(rename_all = "snake_case")]
22pub enum RecoveryAction {
23    /// Refresh one or more instruments on their existing connection.
24    RefreshInstruments,
25    /// Reconnect the source, which establishes a new connection epoch.
26    ReconnectSource,
27}
28
29/// Why a recovery decision was made. Causes are retained on the decision even
30/// when a stronger cause determines the action, so the audit record explains
31/// all contributing evidence.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
33#[serde(rename_all = "snake_case")]
34pub enum RecoveryCause {
35    /// One instrument went quiet while its source stayed healthy.
36    InstrumentQuiet,
37    /// An expected snapshot did not arrive within the policy window.
38    SnapshotTimeout,
39    /// The attempt budget for this recovery is already spent.
40    RecoveryExhausted,
41    /// The venue heartbeat failed. A disabled heartbeat never produces this
42    /// cause.
43    HeartbeatFailure,
44    /// The transport connection closed or a read failed.
45    TransportClosed,
46    /// A payload was rejected by decoding or validation.
47    DecodeFailure,
48    /// The venue protocol reported a violation such as a sequence gap.
49    ProtocolViolation,
50    /// A periodic refresh was due.
51    ScheduledRefresh,
52    /// An operator explicitly requested recovery.
53    OperatorRequested,
54}
55
56/// Conditions observed by the caller. This snapshot of
57/// facts, not a decision: the policy owns the cause-to-action mapping.
58#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
59pub struct RecoverySignals {
60    pub instrument_quiet: bool,
61    pub snapshot_timeout: bool,
62    pub heartbeat_failed: bool,
63    /// When the venue has no heartbeat mechanism, a missed heartbeat is not
64    /// evidence of failure and must not authorize a reconnect.
65    pub heartbeat_disabled: bool,
66    pub transport_closed: bool,
67    pub decode_failed: bool,
68    pub protocol_violation: bool,
69    pub scheduled_refresh: bool,
70    pub operator_requested: bool,
71    /// Attempts already consumed for this recovery target.
72    pub attempts: u32,
73    /// Maximum attempts permitted by policy. Zero permits none.
74    pub max_attempts: u32,
75}

77/// The auditable outcome of applying policy to observed conditions.
78#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
79pub struct RecoveryDecision {
80    /// Every contributing cause, in stable order. Empty when there is no
81    /// evidence of a recovery need.
82    pub causes: Vec<RecoveryCause>,
83    /// The authorized action, or `None` when no action is justified or the
84    /// attempt budget is exhausted.
85    pub action: Option<RecoveryAction>,
86    /// True when causes exist but the attempt budget forbids another action.
87    pub exhausted: bool,
88}
89
90impl RecoveryDecision {
91    /// Whether this decision authorizes reconnecting the source.
92    #[must_use]
93    pub const fn reconnects_source(&self) -> bool {
94        matches!(self.action, Some(RecoveryAction::ReconnectSource))
95    }
96
97    /// Whether this decision authorizes refreshing instruments in place.
98    #[must_use]
99    pub const fn refreshes_instruments(&self) -> bool {
100        matches!(self.action, Some(RecoveryAction::RefreshInstruments))
101    }
102}

104/// Decide the recovery action for one target from observed conditions.
105///
106/// The function is pure: identical signals always produce an identical
107/// decision. A quiet instrument, a snapshot timeout, and a scheduled refresh
108/// authorize an in-place refresh only. Transport, heartbeat, decode, protocol,
109/// and operator causes authorize a source reconnect. When both classes are
110/// present the stronger action wins, but every cause is retained. Exhausting
111/// the attempt budget records [`RecoveryCause::RecoveryExhausted`] and withholds
112/// the action instead of escalating without limit.
113#[must_use]
114pub fn decide(signals: RecoverySignals) -> RecoveryDecision {
115    let mut causes = BTreeSet::new();
116    if signals.instrument_quiet {
117        causes.insert(RecoveryCause::InstrumentQuiet);
118    }
119    if signals.snapshot_timeout {
120        causes.insert(RecoveryCause::SnapshotTimeout);
121    }
122    if signals.transport_closed {
123        causes.insert(RecoveryCause::TransportClosed);
124    }
125    if signals.decode_failed {
126        causes.insert(RecoveryCause::DecodeFailure);
127    }
128    if signals.protocol_violation {
129        causes.insert(RecoveryCause::ProtocolViolation);
130    }
131    if signals.heartbeat_failed && !signals.heartbeat_disabled {
132        causes.insert(RecoveryCause::HeartbeatFailure);
133    }
134    if signals.scheduled_refresh {
135        causes.insert(RecoveryCause::ScheduledRefresh);
136    }
137    if signals.operator_requested {
138        causes.insert(RecoveryCause::OperatorRequested);
139    }
140
141    let reconnect = causes.iter().any(|cause| {
142        matches!(
143            cause,
144            RecoveryCause::HeartbeatFailure
145                | RecoveryCause::TransportClosed
146                | RecoveryCause::DecodeFailure
147                | RecoveryCause::ProtocolViolation
148                | RecoveryCause::OperatorRequested
149        )
150    });
151    let refresh = causes.iter().any(|cause| {
152        matches!(
153            cause,
154            RecoveryCause::InstrumentQuiet
155                | RecoveryCause::SnapshotTimeout
156                | RecoveryCause::ScheduledRefresh
157        )
158    });
159    let action = if reconnect {
160        Some(RecoveryAction::ReconnectSource)
161    } else if refresh {
162        Some(RecoveryAction::RefreshInstruments)
163    } else {
164        None
165    };
166
167    let exhausted = action.is_some() && signals.attempts >= signals.max_attempts;
168    if exhausted {
169        causes.insert(RecoveryCause::RecoveryExhausted);
170    }
171    RecoveryDecision {
172        causes: causes.into_iter().collect(),
173        action: if exhausted { None } else { action },
174        exhausted,
175    }
176}

186/// A receive fence bound to one exact recovery target.
187#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
188pub struct RecoveryFence {
189    pub target: RecoveryTarget,
190    pub connection_epoch: u64,
191    pub receive_seq: u64,
192}
193
194impl RecoveryFence {
195    /// Whether a frame read at `(epoch, seq)` is strictly newer than the
196    /// fence. Snapshots at or before the fence cannot repair a gap.
197    #[must_use]
198    pub const fn is_before(&self, epoch: u64, seq: u64) -> bool {
199        epoch > self.connection_epoch || (epoch == self.connection_epoch && seq > self.receive_seq)
200    }
201
202    /// Whether an acknowledgement from an older connection cannot fence this
203    /// target.
204    #[must_use]
205    pub const fn rejects_stale_ack(&self, epoch: u64, seq: u64) -> bool {
206        epoch < self.connection_epoch || (epoch == self.connection_epoch && seq < self.receive_seq)
207    }
208}

242/// One queued recovery request. Causes are accumulated, attempts are bounded,
243/// and the outcome records the resulting state.
244#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
245pub struct RecoveryRequest {
246    pub action: RecoveryAction,
247    pub causes: Vec<RecoveryCause>,
248    pub attempt: u32,
249    pub max_attempts: u32,
250    pub deadline_us: Option<i64>,
251    pub outcome: Option<RecoveryOutcome>,
252}
253
254impl RecoveryRequest {
255    fn new(
256        action: RecoveryAction,
257        causes: Vec<RecoveryCause>,
258        max_attempts: u32,
259        deadline_us: Option<i64>,
260    ) -> Self {
261        Self {
262            action,
263            causes,
264            attempt: 0,
265            max_attempts,
266            deadline_us,
267            outcome: None,
268        }
269    }
270
271    fn merge_causes(&mut self, causes: &[RecoveryCause]) {
272        let mut all: BTreeSet<RecoveryCause> = self.causes.iter().copied().collect();
273        all.extend(causes.iter().copied());
274        self.causes = all.into_iter().collect();
275    }
276}

278/// Why a queue operation was refused.
279#[derive(Debug, Clone, PartialEq, Eq)]
280pub enum RecoveryQueueError {
281    /// A request was routed to the queue that does not own its action.
282    ActionMismatch {
283        expected: RecoveryAction,
284        found: RecoveryAction,
285    },
286    /// The queue is at its configured bound.
287    CapacityExceeded {
288        queue: &'static str,
289        capacity: usize,
290    },
291}
292
293impl fmt::Display for RecoveryQueueError {
294    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
295        match self {
296            Self::ActionMismatch { expected, found } => write!(
297                formatter,
298                "recovery action {found:?} cannot enter the {expected:?} queue"
299            ),
300            Self::CapacityExceeded { queue, capacity } => {
301                write!(formatter, "{queue} recovery queue is full ({capacity})")
302            }
303        }
304    }
305}

309/// Bounded, ordered request storage shared by both recovery queues.
310#[derive(Debug, Clone)]
311struct RecoveryQueue<K: Ord> {
312    name: &'static str,
313    capacity: usize,
314    requests: BTreeMap<K, RecoveryRequest>,
315}
316
317impl<K: Ord> RecoveryQueue<K> {
318    fn insert(
319        &mut self,
320        key: K,
321        request: RecoveryRequest,
322        expected: RecoveryAction,
323    ) -> Result<(), RecoveryQueueError> {
324        if request.action != expected {
325            return Err(RecoveryQueueError::ActionMismatch {
326                expected,
327                found: request.action,
328            });
329        }
330        if !self.requests.contains_key(&key) && self.requests.len() >= self.capacity {
331            return Err(RecoveryQueueError::CapacityExceeded {
332                queue: self.name,
333                capacity: self.capacity,
334            });
335        }
336        self.requests.insert(key, request);
337        Ok(())
338    }
339
340    fn get(&self, key: &K) -> Option<&RecoveryRequest> {
341        self.requests.get(key)
342    }
343
344    fn get_mut(&mut self, key: &K) -> Option<&mut RecoveryRequest> {
345        self.requests.get_mut(key)
346    }
347
348    fn remove(&mut self, key: &K) -> Option<RecoveryRequest> {
349        self.requests.remove(key)
350    }
351
352    fn len(&self) -> usize {
353        self.requests.len()
354    }
355
356    fn is_empty(&self) -> bool {
357        self.requests.is_empty()
358    }
359
360    fn drain_due(&self, now_us: i64) -> Vec<(K, RecoveryRequest)>
361    where
362        K: Clone,
363    {
364        // BTreeMap iteration is deterministic by key: (source, instrument,
365        // generation) for instruments and (source, generation) for sources.
366        self.requests
367            .iter()
368            .filter(|(_, request)| {
369                request.outcome.is_none()
370                    && request
371                        .deadline_us
372                        .is_none_or(|deadline| deadline <= now_us)
373            })
374            .map(|(key, request)| (key.clone(), request.clone()))
375            .collect()
376    }
377}

532/// The M3.2 coordinator: two disjoint queues with independent generations.
533#[derive(Debug, Clone)]
534pub struct RecoveryCoordinator {
535    pub instruments: InstrumentRecoveryQueue,
536    pub sources: SourceRecoveryQueue,
537    next_generation: u64,
538    max_instrument_attempts: u32,
539    max_source_attempts: u32,
540}
541
542impl RecoveryCoordinator {
543    #[must_use]
544    pub const fn new(
545        instrument_capacity: usize,
546        source_capacity: usize,
547        max_instrument_attempts: u32,
548        max_source_attempts: u32,
549    ) -> Self {
550        Self {
551            instruments: InstrumentRecoveryQueue::new(instrument_capacity),
552            sources: SourceRecoveryQueue::new(source_capacity),
553            next_generation: 1,
554            max_instrument_attempts,
555            max_source_attempts,
556        }
557    }
558
559    fn allocate_generation(&mut self) -> u64 {
560        let generation = self.next_generation;
561        self.next_generation = self.next_generation.saturating_add(1);
562        generation
563    }
564
565    /// Request an in-place refresh of one instrument on one source.
566    pub fn request_instrument_refresh(
567        &mut self,
568        source: SourceId,
569        instrument: impl Into<String>,
570        causes: &[RecoveryCause],
571        deadline_us: Option<i64>,
572    ) -> Result<InstrumentRecoveryKey, RecoveryQueueError> {
573        let key = InstrumentRecoveryKey {
574            source,
575            instrument: instrument.into(),
576            generation: self.allocate_generation(),
577        };
578        let request = RecoveryRequest::new(
579            RecoveryAction::RefreshInstruments,
580            causes.to_vec(),
581            self.max_instrument_attempts,
582            deadline_us,
583        );
584        self.instruments.insert(key.clone(), request)?;
585        Ok(key)
586    }
587
588    /// Request a reconnect of one source. No instrument identity is involved.
589    pub fn request_source_reconnect(
590        &mut self,
591        source: SourceId,
592        causes: &[RecoveryCause],
593        deadline_us: Option<i64>,
594    ) -> Result<SourceRecoveryKey, RecoveryQueueError> {
595        let key = SourceRecoveryKey {
596            source,
597            generation: self.allocate_generation(),
598        };
599        let request = RecoveryRequest::new(
600            RecoveryAction::ReconnectSource,
601            causes.to_vec(),
602            self.max_source_attempts,
603            deadline_us,
604        );
605        self.sources.insert(key.clone(), request)?;
606        Ok(key)
607    }
608
609    /// Merge repeated causes for one recovery target and generation.
610    pub fn merge_instrument_causes(
611        &mut self,
612        key: &InstrumentRecoveryKey,
613        causes: &[RecoveryCause],
614    ) -> bool {
615        self.instruments.merge_causes(key, causes)
616    }
617
618    pub fn merge_source_causes(
619        &mut self,
620        key: &SourceRecoveryKey,
621        causes: &[RecoveryCause],
622    ) -> bool {
623        self.sources.merge_causes(key, causes)
624    }
625
626    pub fn acknowledge_instrument(
627        &mut self,
628        key: &InstrumentRecoveryKey,
629        connection_epoch: u64,
630        receive_seq: u64,
631    ) -> bool {
632        self.instruments.set_outcome(
633            key,
634            RecoveryOutcome::Acknowledged {
635                connection_epoch,
636                receive_seq,
637            },
638        )
639    }
640
641    pub fn acknowledge_source(
642        &mut self,
643        key: &SourceRecoveryKey,
644        connection_epoch: u64,
645        receive_seq: u64,
646    ) -> bool {
647        self.sources.set_outcome(
648            key,
649            RecoveryOutcome::Acknowledged {
650                connection_epoch,
651                receive_seq,
652            },
653        )
654    }
655
656    pub fn cancel_instrument(&mut self, key: &InstrumentRecoveryKey) -> bool {
657        self.instruments
658            .set_outcome(key, RecoveryOutcome::Cancelled)
659    }
660
661    pub fn cancel_source(&mut self, key: &SourceRecoveryKey) -> bool {
662        self.sources.set_outcome(key, RecoveryOutcome::Cancelled)
663    }
664
665    #[must_use]
666    pub fn stats(&self) -> RecoveryQueueStats {
667        RecoveryQueueStats {
668            instrument_requests: self.instruments.len(),
669            source_requests: self.sources.len(),
670        }
671    }
672
673    /// Deterministic due-work snapshot: instruments first, then sources, each
674    /// in key order. Acknowledged and cancelled requests are not due.
675    #[must_use]
676    pub fn drain_due(&self, now_us: i64) -> DueRecoveryWork {
677        DueRecoveryWork {
678            instruments: self.instruments.drain_due(now_us),
679            sources: self.sources.drain_due(now_us),
680        }
681    }
682
683    /// Drop completed requests so successful retries can allocate fresh
684    /// generations without unbounded queue growth.
685    pub fn reap_completed(&mut self) -> usize {
686        let before = self.stats();
687        self.instruments
688            .inner
689            .requests
690            .retain(|_, request| request.outcome.is_none());
691        self.sources
692            .inner
693            .requests
694            .retain(|_, request| request.outcome.is_none());
695        let after = self.stats();
696        before
697            .instrument_requests
698            .saturating_sub(after.instrument_requests)
699            + before.source_requests.saturating_sub(after.source_requests)
700    }
