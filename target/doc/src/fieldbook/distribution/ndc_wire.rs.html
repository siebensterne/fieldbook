2//! Explicit producer mapping from internal release values to the ratified
3//! canonical NDC v2 Profile 1 wire representation.
4//!
5//! The canonical schema package under workspace authority
6//! `WS-NDC-2026-11-24-003` is the wire authority. The internal
7//! `ReleaseManifest` and its section types are inputs only: every wire
8//! member below is named and validated by this module, never inherited from a
9//! Rust field name. The internal pre-ratification sections carry opaque
10//! references and omit several required canonical members; a strict abstract
11//! evidence supplement supplies the typed values that the internal types do
12//! not hold. A missing, unknown, duplicated or coerced value fails closed with
13//! a typed [`WireError`].
14//!
15//! The critical correctness boundary is byte-derived record ordering.
16//! [`sort_byte_derived`] parses the canonical record bytes, extracts the
17//! ordering member, validates it against the artifact profile, derives the key
18//! and sorts. [`CanonicalRecord`] carries bytes only, so no caller can express
19//! a second position on the public producer input. The
20//! same validated input bytes are returned in sorted order; no record is
21//! re-serialized to obtain an ordering position.
22//!
23//! Non-claim: this module is a producer-side mapping and conformance vector
24//! support. It creates no release, no right, no dataset acceptance and no
25//! cross-execution status. The signing seed used by the conformance executable
26//! is a published test vector, not a production key.

124/// Typed failure of the wire mapping. Every refusal is closed and names its
125/// location so that no partial wire document can be emitted.
126#[derive(Debug, Error)]
127pub enum WireError {
128    /// Input was not valid canonical JSON text.
129    #[error("invalid JSON: {0}")]
130    InvalidJson(String),
131    /// A value could not be serialized as canonical JSON.
132    #[error("canonicalization failed: {0}")]
133    Canonicalization(String),
134    /// An object carried a property outside its closed member set.
135    #[error("unknown field `{0}` at {1}")]
136    UnknownField(String, String),
137    /// A required property was absent.
138    #[error("missing required field `{0}` at {1}")]
139    MissingField(String, String),
140    /// A property appeared more than once.
141    #[error("duplicate field `{0}` at {1}")]
142    DuplicateField(String, String),
143    /// A value had the wrong JSON type for its field.
144    #[error("wrong JSON type at {0}: {1}")]
145    WrongType(String, String),
146    /// A string failed the ratified identifier grammar or was a placeholder.
147    #[error("invalid identifier `{0}` at {1}")]
148    Identifier(String, String),
149    /// A decimal-string value failed the ratified grammar or u64 range.
150    #[error("invalid decimal string `{0}` at {1}")]
151    Decimal(String, String),
152    /// A UTC string failed the canonical nine-fraction-digit grammar.
153    #[error("invalid UTC instant `{0}` at {1}")]
154    UtcInstant(String, String),
155    /// A digest was not lowercase hexadecimal of the required length.
156    #[error("invalid digest at {0}")]
157    Digest(String),
158    /// An artifact path failed the release-relative alphabet rules.
159    #[error("invalid artifact path `{0}`")]
160    ArtifactPath(String),
161    /// Record bytes were not a single canonical UTF-8 JSON line.
162    #[error("non-canonical record at {0}: {1}")]
163    NonCanonicalRecord(String, String),
164    /// A tagged ordering union was malformed or did not match its profile.
165    #[error("ordering error at {0}: {1}")]
166    Ordering(String, String),
167    /// Two records derived the same key with different payloads.
168    #[error("duplicate derived ordering key at {0}")]
169    DuplicateKey(String),
170    /// Coverage arithmetic, evidence or verdict gating failed.
171    #[error("coverage error: {0}")]
172    Coverage(String),
173    /// Source-rights evidence was absent, malformed or inconsistent.
174    #[error("rights evidence error: {0}")]
175    Rights(String),
176    /// Performed operations and typed evidence were not bound one to one.
177    #[error("operation evidence error: {0}")]
178    OperationEvidence(String),
179    /// PROV lineage, revision or invalidation semantics failed.
180    #[error("lineage error: {0}")]
181    Lineage(String),
182    /// The per-event payload schema registry was incomplete or inconsistent.
183    #[error("payload schema registry error: {0}")]
184    PayloadRegistry(String),
185    /// A compression method and version identifier were not a registered pair.
186    #[error("unregistered compression identifier: {0}")]
187    Compression(String),
188    /// A caller-supplied resource bound was exceeded.
189    #[error("resource policy exceeded: {0}")]
190    Policy(String),
191    /// The abstract conformance spec was malformed.
192    #[error("invalid conformance spec: {0}")]
193    Spec(String),
194    /// Signing or verifying failed.
195    #[error("signature error: {0}")]
196    Signature(String),
197    /// A filesystem operation failed.
198    #[error("filesystem failure during {0}: {1}")]
199    Io(String, #[source] std::io::Error),
200}

682
683impl SpecProfile {
684    /// Canonical wire name of the selected profile kind.
685    #[must_use]
686    pub const fn kind(&self) -> &'static str {
687        match self {
688            Self::Sequenced { .. } => "sequenced",
689            Self::Reference => "reference",
690        }
691    }
692
693    /// Convert the abstract profile into the internal producer declaration.
694    pub fn to_internal(&self) -> Result<RecordOrderProfile, WireError> {
695        match self {
696            Self::Sequenced {
697                sequence_domain_id,
698                sequence_epoch,
699            } => {
700                reject_placeholder(sequence_domain_id, "sequence_domain_id")?;
701                reject_placeholder(sequence_epoch, "sequence_epoch")?;
702                Ok(RecordOrderProfile::Sequenced {
703                    sequence_domain: sequence_domain_id.clone(),
704                    sequence_epoch: sequence_epoch.clone(),
705                })
706            }
707            Self::Reference => Ok(RecordOrderProfile::Reference),
708        }
709    }

751/// Enforce a closed object member set with no defaults and no coercions.
752///
753/// The value must be an object; every member must appear in `allowed`, every
754/// name in `required` must be present, and a repeated name is rejected even
755/// though the upstream parser already rejects duplicate JSON names.
756pub fn strict_object<'a>(
757    value: &'a JcsValue,
758    path: &str,
759    allowed: &[&str],
760    required: &[&str],
761) -> Result<BTreeMap<&'a str, &'a JcsValue>, WireError> {
762    let JcsValue::Object(entries) = value else {
763        return Err(WireError::WrongType(path.to_owned(), "object".to_owned()));
764    };
765    let mut fields = BTreeMap::new();
766    for (name, child) in entries {
767        if !allowed.contains(&name.as_str()) {
768            return Err(WireError::UnknownField(name.clone(), path.to_owned()));
769        }
770        if fields.insert(name.as_str(), child).is_some() {
771            return Err(WireError::DuplicateField(name.clone(), path.to_owned()));
772        }
773    }
774    for name in required {
775        if !fields.contains_key(*name) {
776            return Err(WireError::MissingField((*name).to_owned(), path.to_owned()));
777        }
778    }
779    Ok(fields)
780}

810/// Parse a decimal string matching `^(0|[1-9][0-9]*)$` within the u64 range.
811pub fn parse_decimal(value: &str, field: &str) -> Result<u64, WireError> {
812    if value.is_empty()
813        || value.len() > 20
814        || !value.bytes().all(|byte| byte.is_ascii_digit())
815        || (value.len() > 1 && value.starts_with('0'))
816    {

820        .parse::<u64>()
821        .map_err(|_| WireError::Decimal(value.to_owned(), field.to_owned()))
822}

824/// Validate the ratified identifier grammar and reject placeholder literals.
825pub fn validate_identifier(value: &str, field: &str) -> Result<(), WireError> {
826    let mut bytes = value.bytes();
827    let Some(first) = bytes.next() else {
828        return Err(WireError::Identifier(value.to_owned(), field.to_owned()));
829    };
830    if !first.is_ascii_alphanumeric() || value.len() > 128 {
831        return Err(WireError::Identifier(value.to_owned(), field.to_owned()));
832    }
833    if !bytes.all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b':' | b'-'))
834    {
835        return Err(WireError::Identifier(value.to_owned(), field.to_owned()));
836    }
837    reject_placeholder(value, field)
838}

859/// Validate the canonical UTC grammar with exactly nine fractional digits.
860pub fn validate_utc_instant(value: &str, field: &str) -> Result<(), WireError> {
861    let invalid = || WireError::UtcInstant(value.to_owned(), field.to_owned());
862    let bytes = value.as_bytes();
863    if bytes.len() != 30
864        || bytes[4] != b'-'
865        || bytes[7] != b'-'
866        || bytes[10] != b'T'
867        || bytes[13] != b':'
868        || bytes[16] != b':'
869        || bytes[19] != b'.'
870        || bytes[29] != b'Z'
871    {
872        return Err(invalid());
873    }
874    let digits = |range: std::ops::Range<usize>| -> Option<u32> {
875        let slice = value.get(range)?;
876        if !slice.bytes().all(|byte| byte.is_ascii_digit()) {
877            return None;
878        }
879        slice.parse::<u32>().ok()
880    };
881    let year = digits(0..4).ok_or_else(invalid)?;
882    let month = digits(5..7).ok_or_else(invalid)?;
883    let day = digits(8..10).ok_or_else(invalid)?;
884    let hour = digits(11..13).ok_or_else(invalid)?;
885    let minute = digits(14..16).ok_or_else(invalid)?;
886    let second = digits(17..19).ok_or_else(invalid)?;
887    if digits(20..29).is_none() {
888        return Err(invalid());
889    }
890    let leap = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
891    let days = match month {
892        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
893        4 | 6 | 9 | 11 => 30,
894        2 if leap => 29,
895        2 => 28,
896        _ => return Err(invalid()),
897    };
898    if day == 0 || day > days || hour > 23 || minute > 59 || second > 59 {
899        return Err(invalid());
900    }
901    Ok(())
902}

1007/// Parse one canonical record line and derive its ordering key from the parsed
1008/// bytes. No caller-supplied position, index or pre-sorted hint is consulted.
1009pub fn parse_canonical_record(
1010    bytes: &[u8],
1011    policy: &ResourcePolicy,
1012) -> Result<ParsedRecord, WireError> {
1013    if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
1014        return Err(WireError::NonCanonicalRecord(
1015            "root".to_owned(),
1016            "byte-order mark".to_owned(),
1017        ));
1018    }
1019    if u64::try_from(bytes.len()).map_err(|_| WireError::Policy("record length".to_owned()))?
1020        > policy.max_record_bytes
1021    {
1022        return Err(WireError::Policy(format!(
1023            "record length {} exceeds {}",
1024            bytes.len(),
1025            policy.max_record_bytes
1026        )));
1027    }
1028    let source = std::str::from_utf8(bytes)
1029        .map_err(|_| WireError::NonCanonicalRecord("root".to_owned(), "not UTF-8".to_owned()))?;
1030    if source.contains(['\n', '\r']) {
1031        return Err(WireError::NonCanonicalRecord(
1032            "root".to_owned(),
1033            "embedded line terminator".to_owned(),
1034        ));
1035    }
1036    let value =
1037        parse_jcs_value(source).map_err(|error| WireError::InvalidJson(error.to_string()))?;
1038    let canonical = canonicalize_strict(&value, policy)
1039        .map_err(|error| WireError::Canonicalization(error.to_string()))?;
1040    if canonical != bytes {
1041        return Err(WireError::NonCanonicalRecord(
1042            "root".to_owned(),
1043            "bytes differ from RFC 8785 canonical form".to_owned(),
1044        ));
1045    }
1046    let envelope = strict_object(
1047        &value,
1048        "envelope",
1049        &[
1050            "envelope_schema_version",
1051            "event_type",
1052            "ordering",
1053            "payload",
1054        ],
1055        &[
1056            "envelope_schema_version",
1057            "event_type",
1058            "ordering",
1059            "payload",
1060        ],
1061    )?;
1062    let envelope_schema_version =
1063        field_string(&envelope, "envelope_schema_version", "envelope")?.to_owned();
1064    if envelope_schema_version.is_empty() {
1065        return Err(WireError::WrongType(
1066            "envelope.envelope_schema_version".to_owned(),
1067            "non-empty string".to_owned(),
1068        ));
1069    }
1070    let event_type = field_string(&envelope, "event_type", "envelope")?.to_owned();
1071    validate_identifier(&event_type, "event_type")?;
1072    let ordering_value = envelope["ordering"];
1073    let ordering = parse_record_ordering(ordering_value)?;
1074    let derived_key = match ordering {
1075        RecordOrdering::Sequenced {
1076            sequence_domain_id,
1077            sequence_epoch,
1078            sequence_number,
1079        } => WireOrderKey::Sequenced {
1080            sequence_domain_id,
1081            sequence_epoch,
1082            sequence_number,
1083        },
1084        RecordOrdering::Reference {
1085            object_id,
1086            revision,
1087            knowledge_time_ns,
1088            assertion_id,
1089        } => WireOrderKey::Reference {
1090            object_id,
1091            revision,
1092            knowledge_time_ns,
1093            assertion_id,
1094        },
1095    };
1096    let payload = envelope["payload"].clone();
1097    validate_payload(&payload, policy)?;
1098    Ok(ParsedRecord {
1099        bytes: bytes.to_vec(),
1100        envelope_schema_version,
1101        event_type,
1102        derived_key,
1103        payload,
1104    })
1105}

1121fn parse_record_ordering(value: &JcsValue) -> Result<RecordOrdering, WireError> {
1122    let JcsValue::Object(entries) = value else {
1123        return Err(WireError::Ordering(
1124            "ordering".to_owned(),
1125            "tagged object required".to_owned(),
1126        ));
1127    };
1128    let kind = entries.iter().find(|(name, _)| name == "kind");
1129    let Some((_, kind_value)) = kind else {
1130        return Err(WireError::Ordering(
1131            "ordering".to_owned(),
1132            "missing kind".to_owned(),
1133        ));
1134    };
1135    let JcsValue::String(kind_text) = kind_value else {
1136        return Err(WireError::Ordering(
1137            "ordering.kind".to_owned(),
1138            "string required".to_owned(),
1139        ));
1140    };
1141    match kind_text.as_str() {
1142        "sequenced" => {
1143            let fields = strict_object(
1144                value,
1145                "ordering",
1146                &[
1147                    "kind",
1148                    "sequence_domain_id",
1149                    "sequence_epoch",
1150                    "sequence_number",
1151                ],
1152                &[
1153                    "kind",
1154                    "sequence_domain_id",
1155                    "sequence_epoch",
1156                    "sequence_number",
1157                ],
1158            )?;
1159            Ok(RecordOrdering::Sequenced {
1160                sequence_domain_id: field_string(&fields, "sequence_domain_id", "ordering")?
1161                    .to_owned(),
1162                sequence_epoch: field_string(&fields, "sequence_epoch", "ordering")?.to_owned(),
1163                sequence_number: field_decimal(&fields, "sequence_number", "ordering")?,
1164            })
1165        }
1166        "reference" => {
1167            let fields = strict_object(
1168                value,
1169                "ordering",
1170                &[
1171                    "kind",
1172                    "object_id",
1173                    "revision",
1174                    "knowledge_time_ns",
1175                    "assertion_id",
1176                ],
1177                &[
1178                    "kind",
1179                    "object_id",
1180                    "revision",
1181                    "knowledge_time_ns",
1182                    "assertion_id",
1183                ],
1184            )?;
1185            let object_id = field_string(&fields, "object_id", "ordering")?.to_owned();
1186            let assertion_id = field_string(&fields, "assertion_id", "ordering")?.to_owned();
1187            validate_identifier(&object_id, "ordering.object_id")?;
1188            validate_identifier(&assertion_id, "ordering.assertion_id")?;
1189            Ok(RecordOrdering::Reference {
1190                object_id,

1192                knowledge_time_ns: field_decimal(&fields, "knowledge_time_ns", "ordering")?,

1194            })
1195        }
1196        other => Err(WireError::Ordering(
1197            "ordering.kind".to_owned(),
1198            format!("unknown kind {other}"),
1199        )),
1200    }

1203fn validate_payload(value: &JcsValue, policy: &ResourcePolicy) -> Result<(), WireError> {
1204    let payload = strict_object(value, "payload", &["fields"], &["fields"])?;
1205    let JcsValue::Array(fields) = payload["fields"] else {
1206        return Err(WireError::WrongType(
1207            "payload.fields".to_owned(),
1208            "array".to_owned(),
1209        ));
1210    };
1211    if fields.is_empty() {
1212        return Err(WireError::Spec(
1213            "payload.fields must not be empty".to_owned(),
1214        ));
1215    }
1216    if u64::try_from(fields.len()).map_err(|_| WireError::Policy("payload fields".to_owned()))?
1217        > u64::from(policy.max_array_elements)
1218    {
1219        return Err(WireError::Policy(
1220            "payload.fields exceeds array bound".to_owned(),
1221        ));
1222    }
1223    for (index, field) in fields.iter().enumerate() {
1224        let location = format!("payload.fields[{index}]");
1225        let members = strict_object(
1226            field,
1227            &location,
1228            &["field_name", "kind", "value"],
1229            &["field_name", "kind", "value"],
1230        )?;
1231        let name = field_string(&members, "field_name", &location)?;
1232        require_nonempty(name, "field_name")?;
1233        let kind = field_string(&members, "kind", &location)?;
1234        let scalar = field_string(&members, "value", &location)?;
1235        match kind {
1236            "string" => {}
1237            "decimal_string" => {
1238                parse_decimal(scalar, "payload field value")?;
1239            }
1240            "boolean" => {
1241                if scalar != "true" && scalar != "false" {
1242                    return Err(WireError::WrongType(
1243                        format!("{location}.value"),
1244                        "boolean literal".to_owned(),
1245                    ));
1246                }
1247            }
1248            "bytes_hex" => {
1249                if scalar.len() % 2 != 0
1250                    || !scalar
1251                        .bytes()
1252                        .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
1253                {
1254                    return Err(WireError::Digest(format!("{location}.value")));
1255                }
1256            }
1257            other => {
1258                return Err(WireError::WrongType(
1259                    format!("{location}.kind"),
1260                    format!("unknown kind {other}"),
1261                ));
1262            }
1263        }
1264    }
1265    Ok(())
1266}

1268/// Validate a parsed record against its containing artifact profile and
1269/// return the key derived from the parsed canonical bytes.
1270pub fn derive_order_key(
1271    record: &ParsedRecord,
1272    profile: &WireArtifactProfile,
1273) -> Result<WireOrderKey, WireError> {
1274    match (&record.derived_key, profile) {
1275        (
1276            WireOrderKey::Sequenced {
1277                sequence_domain_id,
1278                sequence_epoch,
1279                sequence_number,
1280            },
1281            WireArtifactProfile::Sequenced {
1282                sequence_domain_id: expected_domain,
1283                sequence_epoch: expected_epoch,
1284            },
1285        ) => {
1286            if sequence_domain_id != expected_domain || sequence_epoch != expected_epoch {
1287                return Err(WireError::Ordering(
1288                    record.event_type.clone(),
1289                    "record domain or epoch differs from artifact profile".to_owned(),
1290                ));
1291            }
1292            Ok(WireOrderKey::Sequenced {
1293                sequence_domain_id: sequence_domain_id.clone(),
1294                sequence_epoch: sequence_epoch.clone(),
1295                sequence_number: *sequence_number,
1296            })
1297        }
1298        (WireOrderKey::Reference { .. }, WireArtifactProfile::Reference) => {
1299            Ok(record.derived_key.clone())
1300        }
1301        _ => Err(WireError::Ordering(
1302            record.event_type.clone(),
1303            "record ordering mode differs from artifact profile".to_owned(),
1304        )),
1305    }
1306}

1308/// Parse, validate and sort records by keys derived from their own bytes.
1309///
1310/// [`CanonicalRecord`] carries no order-key member. Duplicate keys with
1311/// differing payloads fail closed. The returned byte vectors are the
1312/// validated input bytes unchanged.
1313pub fn sort_byte_derived(
1314    records: &[CanonicalRecord],
1315    profile: &WireArtifactProfile,
1316    policy: &ResourcePolicy,
1317) -> Result<Vec<Vec<u8>>, WireError> {
1318    if u64::try_from(records.len()).map_err(|_| WireError::Policy("record count".to_owned()))?
1319        > policy.max_records_per_artifact
1320    {
1321        return Err(WireError::Policy("record count exceeds bound".to_owned()));
1322    }
1323    let mut entries: Vec<(WireOrderKey, Vec<u8>, String)> = Vec::with_capacity(records.len());
1324    for (index, record) in records.iter().enumerate() {
1325        let parsed = parse_canonical_record(&record.bytes, policy)
1326            .map_err(|error| WireError::NonCanonicalRecord(index.to_string(), error.to_string()))?;
1327        let key = derive_order_key(&parsed, profile)?;
1328        let payload_bytes = canonicalize_strict(&parsed.payload, policy)
1329            .map_err(|error| WireError::Canonicalization(error.to_string()))?;
1330        let digest = sha256_hex(&payload_bytes);
1331        if let Some(previous) = entries.iter().find(|(old, _, _)| old == &key)
1332            && previous.2 != digest
1333        {
1334            return Err(WireError::DuplicateKey(format!("{key:?}")));
1335        }
1336        entries.push((key, parsed.bytes, digest));
1337    }
1338    entries.sort_by(|left, right| left.0.cmp(&right.0).then_with(|| left.1.cmp(&right.1)));
1339    Ok(entries.into_iter().map(|(_, bytes, _)| bytes).collect())
1340}

1406fn map_producer_activity(
1407    activity: &crate::distribution::ndc_release::ProducerActivity,
1408    evidence: &EvidenceSpec,
1409) -> Result<JcsValue, WireError> {
1410    let mut operations: BTreeMap<&str, &PerformedOperationSpec> = BTreeMap::new();
1411    for operation in &evidence.performed_operations {
1412        validate_identifier(&operation.operation_ref, "operation_ref")?;
1413        if operations
1414            .insert(operation.operation_ref.as_str(), operation)
1415            .is_some()
1416        {
1417            return Err(WireError::OperationEvidence(format!(
1418                "duplicate performed operation {}",
1419                operation.operation_ref
1420            )));
1421        }
1422    }
1423    let mut resolved: BTreeMap<&str, &OperationEvidenceSpec> = BTreeMap::new();
1424    for record in &evidence.operation_evidence {
1425        validate_identifier(&record.operation_ref, "operation_ref")?;
1426        if resolved
1427            .insert(record.operation_ref.as_str(), record)
1428            .is_some()
1429        {
1430            return Err(WireError::OperationEvidence(format!(
1431                "duplicate evidence for {}",
1432                record.operation_ref
1433            )));
1434        }
1435    }
1436    let mut mapped_operations = Vec::with_capacity(activity.performed_operations.len());
1437    let mut seen_operations = BTreeSet::new();
1438    for reference in &activity.performed_operations {
1439        validate_identifier(reference, "performed operation reference")?;
1440        if !seen_operations.insert(reference.as_str()) {
1441            return Err(WireError::OperationEvidence(format!(
1442                "duplicate performed operation reference {reference}"
1443            )));
1444        }
1445        let detail = operations.get(reference.as_str()).ok_or_else(|| {
1446            WireError::OperationEvidence(format!("no structured detail for {reference}"))
1447        })?;
1448        validate_action(&detail.action, "performed operation action")?;
1449        mapped_operations.push(obj(vec![
1450            ("operation_ref", text(detail.operation_ref.clone())),
1451            ("action", text(detail.action.clone())),
1452            ("data_object", map_data_object(&detail.data_object)?),
1453        ]));
1454    }
1455    let mut mapped_evidence = Vec::with_capacity(activity.operation_evidence.len());
1456    let mut seen_evidence = BTreeSet::new();
1457    for reference in &activity.operation_evidence {
1458        validate_identifier(reference, "operation evidence reference")?;
1459        if !seen_evidence.insert(reference.as_str()) {
1460            return Err(WireError::OperationEvidence(format!(
1461                "duplicate operation evidence reference {reference}"
1462            )));
1463        }
1464        let detail = resolved.get(reference.as_str()).ok_or_else(|| {
1465            WireError::OperationEvidence(format!("no typed evidence for {reference}"))
1466        })?;
1467        validate_identifier(&detail.event_id, "event_id")?;
1468        validate_identifier(&detail.run_id, "run_id")?;
1469        parse_decimal(&detail.recorded_at_ns, "recorded_at_ns")?;
1470        require_nonempty(&detail.decision_key, "decision_key")?;
1471        validate_sha256_hex(&detail.digest_sha256, "operation digest_sha256")?;
1472        mapped_evidence.push(obj(vec![
1473            ("operation_ref", text(detail.operation_ref.clone())),
1474            ("event_id", text(detail.event_id.clone())),
1475            ("run_id", text(detail.run_id.clone())),
1476            ("recorded_at_ns", text(detail.recorded_at_ns.clone())),
1477            ("decision_key", text(detail.decision_key.clone())),
1478            ("digest_sha256", text(detail.digest_sha256.clone())),
1479        ]));
1480    }
1481    if seen_operations != seen_evidence {
1482        return Err(WireError::OperationEvidence(
1483            "operations and evidence are not bound one to one by operation_ref".to_owned(),
1484        ));
1485    }
1486    if seen_operations.len() != operations.len() || seen_evidence.len() != resolved.len() {
1487        return Err(WireError::OperationEvidence(
1488            "an operation or evidence detail has no matching declaration".to_owned(),
1489        ));
1490    }
1491    Ok(obj(vec![
1492        ("performed_operations", JcsValue::Array(mapped_operations)),
1493        ("operation_evidence", JcsValue::Array(mapped_evidence)),
1494    ]))
1495}

1700fn map_rights(
1701    rights: &crate::distribution::ndc_release::RightsEvidence,
1702    evidence: &EvidenceSpec,
1703) -> Result<JcsValue, WireError> {
1704    let mut grants: BTreeMap<&str, &PermissionGrantSpec> = BTreeMap::new();
1705    for grant in &evidence.permission_grants {
1706        validate_identifier(&grant.grant_id, "grant_id")?;
1707        if grants.insert(grant.grant_id.as_str(), grant).is_some() {
1708            return Err(WireError::Rights(format!(
1709                "duplicate grant {}",
1710                grant.grant_id
1711            )));
1712        }
1713    }
1714    let mut documents: BTreeMap<&str, &DocumentReferenceSpec> = BTreeMap::new();
1715    for document in &evidence.document_references {
1716        require_nonempty(&document.reference, "document reference")?;
1717        require_nonempty(&document.clause, "document clause")?;
1718        validate_sha256_hex(&document.document_sha256, "document_sha256")?;
1719        require_nonempty(&document.contract_revision, "document contract_revision")?;
1720        require_nonempty(&document.review_authority, "document review_authority")?;
1721        validate_utc_instant(&document.reviewed_on, "document.reviewed_on")?;
1722        if documents
1723            .insert(document.reference.as_str(), document)
1724            .is_some()
1725        {
1726            return Err(WireError::Rights(format!(
1727                "duplicate document reference {}",
1728                document.reference
1729            )));
1730        }
1731    }
1732    let mut attestations: BTreeMap<&str, &RegisteredAttestationSpec> = BTreeMap::new();
1733    for attestation in &evidence.reviewed_attestations {
1734        validate_identifier(&attestation.attestation_id, "attestation_id")?;
1735        validate_identifier(&attestation.grant_id, "attestation grant_id")?;
1736        map_attestation(&attestation.attestation)?;
1737        if attestations
1738            .insert(attestation.attestation_id.as_str(), attestation)
1739            .is_some()
1740        {
1741            return Err(WireError::Rights(format!(
1742                "duplicate attestation {}",
1743                attestation.attestation_id
1744            )));
1745        }
1746    }
1747    if grants.is_empty() || documents.is_empty() || attestations.is_empty() {
1748        return Err(WireError::Rights("rights evidence is absent".to_owned()));
1749    }
1750    let mut mapped_grants = Vec::with_capacity(rights.permission_grants.len());
1751    let mut used_grants = BTreeSet::new();
1752    for reference in &rights.permission_grants {
1753        let grant = grants
1754            .get(reference.as_str())
1755            .ok_or_else(|| WireError::Rights(format!("no typed grant for {reference}")))?;
1756        if !used_grants.insert(reference.as_str()) {
1757            return Err(WireError::Rights(format!(
1758                "duplicate grant reference {reference}"
1759            )));
1760        }
1761        let registered = attestations
1762            .values()
1763            .find(|item| item.grant_id == grant.grant_id)
1764            .ok_or_else(|| {
1765                WireError::Rights(format!("no registered attestation for {reference}"))
1766            })?;
1767        let embedded = map_attestation(&grant.reviewed_attestation)?;
1768        let registered_value = map_attestation(&registered.attestation)?;
1769        if embedded != registered_value {
1770            return Err(WireError::Rights(format!(
1771                "embedded and registered attestations differ for {reference}"
1772            )));
1773        }
1774        mapped_grants.push(map_grant(grant)?);
1775    }
1776    let mut mapped_documents = Vec::with_capacity(rights.document_references.len());
1777    let mut used_documents = BTreeSet::new();
1778    for reference in &rights.document_references {
1779        let document = documents.get(reference.as_str()).ok_or_else(|| {
1780            WireError::Rights(format!("no typed document reference for {reference}"))
1781        })?;
1782        if !used_documents.insert(reference.as_str()) {
1783            return Err(WireError::Rights(format!(
1784                "duplicate document reference {reference}"
1785            )));
1786        }
1787        mapped_documents.push(obj(vec![
1788            ("reference", text(document.reference.clone())),
1789            ("clause", text(document.clause.clone())),
1790            ("document_sha256", text(document.document_sha256.clone())),
1791            (
1792                "contract_revision",
1793                text(document.contract_revision.clone()),
1794            ),
1795            ("review_authority", text(document.review_authority.clone())),
1796            ("reviewed_on", text(document.reviewed_on.clone())),
1797        ]));
1798    }
1799    let mut mapped_attestations = Vec::with_capacity(rights.reviewed_attestations.len());
1800    let mut used_attestations = BTreeSet::new();
1801    for reference in &rights.reviewed_attestations {
1802        let attestation = attestations.get(reference.as_str()).ok_or_else(|| {
1803            WireError::Rights(format!("no registered attestation for {reference}"))
1804        })?;
1805        if !used_attestations.insert(reference.as_str()) {
1806            return Err(WireError::Rights(format!(
1807                "duplicate attestation reference {reference}"
1808            )));
1809        }
1810        mapped_attestations.push(obj(vec![
1811            ("attestation_id", text(attestation.attestation_id.clone())),
1812            ("grant_id", text(attestation.grant_id.clone())),
1813            ("attestation", map_attestation(&attestation.attestation)?),
1814        ]));
1815    }
1816    if used_grants.len() != grants.len()
1817        || used_documents.len() != documents.len()
1818        || used_attestations.len() != attestations.len()
1819    {
1820        return Err(WireError::Rights(
1821            "an evidence detail has no matching reference".to_owned(),
1822        ));
1823    }
1824    Ok(obj(vec![
1825        ("permission_grants", JcsValue::Array(mapped_grants)),
1826        ("document_references", JcsValue::Array(mapped_documents)),
1827        (
1828            "reviewed_attestations",
1829            JcsValue::Array(mapped_attestations),
1830        ),
1831    ]))
1832}

1856    evidence: &EvidenceSpec,
1857    release_as_of_ns: &str,
1858) -> Result<JcsValue, WireError> {
1859    if coverage.dimensions.len() != NINE_DIMENSIONS.len() {
1860        return Err(WireError::Coverage(
1861            "exactly nine dimensions are required".to_owned(),
1862        ));
1863    }
1864    let mut details: BTreeMap<&str, &CoverageDetailSpec> = BTreeMap::new();
1865    for detail in &evidence.coverage_details {
1866        if details
1867            .insert(detail.dimension_id.as_str(), detail)
1868            .is_some()
1869        {
1870            return Err(WireError::Coverage(format!(
1871                "duplicate coverage detail {}",
1872                detail.dimension_id
1873            )));
1874        }
1875    }
1876    let mut mapped = Vec::with_capacity(coverage.dimensions.len());
1877    let mut seen = BTreeSet::new();
1878    for dimension in &coverage.dimensions {
1879        if !NINE_DIMENSIONS.contains(&dimension.dimension_id.as_str()) {
1880            return Err(WireError::Coverage(format!(
1881                "unknown dimension {}",
1882                dimension.dimension_id
1883            )));
1884        }
1885        if !seen.insert(dimension.dimension_id.as_str()) {
1886            return Err(WireError::Coverage(format!(
1887                "duplicate dimension {}",
1888                dimension.dimension_id
1889            )));
1890        }
1891        let detail = details
1892            .get(dimension.dimension_id.as_str())
1893            .ok_or_else(|| {
1894                WireError::Coverage(format!("no detail for {}", dimension.dimension_id))
1895            })?;
1896        if dimension.denominator == 0 {
1897            return Err(WireError::Coverage(format!(
1898                "{} has no declared denominator",
1899                dimension.dimension_id
1900            )));
1901        }
1902        require_nonempty(&detail.denominator_identity, "denominator_identity")?;
1903        require_nonempty(&detail.count_basis, "count_basis")?;
1904        require_nonempty(&dimension.evidence_reference, "evidence_reference")?;
1905        let total = dimension
1906            .numerator
1907            .checked_add(detail.missing_count)
1908            .and_then(|value| value.checked_add(detail.rejected_count))
1909            .ok_or_else(|| WireError::Coverage("coverage count overflow".to_owned()))?;
1910        if total != dimension.denominator {
1911            return Err(WireError::Coverage(format!(
1912                "{} arithmetic does not decompose its denominator",
1913                dimension.dimension_id
1914            )));
1915        }
1916        if detail.as_of_ns != release_as_of_ns {
1917            return Err(WireError::Coverage(format!(
1918                "{} as-of differs from release as-of",
1919                dimension.dimension_id
1920            )));
1921        }
1922        parse_decimal(&detail.as_of_ns, "coverage as_of_ns")?;
1923        if !["PASS", "FAIL", "UNKNOWN"].contains(&dimension.verdict.as_str()) {
1924            return Err(WireError::Coverage("invalid verdict".to_owned()));
1925        }
1926        if dimension.verdict == "PASS"
1927            && (dimension.numerator != dimension.denominator
1928                || detail.missing_count != 0
1929                || detail.rejected_count != 0)
1930        {
1931            return Err(WireError::Coverage(format!(
1932                "{} producer PASS is not admitted by independent evidence",
1933                dimension.dimension_id
1934            )));
1935        }
1936        validate_identifier(&dimension.universe_id, "universe_id")?;
1937        mapped.push(obj(vec![
1938            ("dimension_id", text(dimension.dimension_id.clone())),
1939            ("universe_id", text(dimension.universe_id.clone())),
1940            (
1941                "denominator_identity",
1942                text(detail.denominator_identity.clone()),
1943            ),
1944            ("count_basis", text(detail.count_basis.clone())),
1945            ("numerator", count(dimension.numerator)),
1946            ("denominator", count(dimension.denominator)),
1947            ("missing_count", count(detail.missing_count)),
1948            ("rejected_count", count(detail.rejected_count)),
1949            ("as_of_ns", text(detail.as_of_ns.clone())),
1950            ("verdict", text(dimension.verdict.clone())),
1951            (
1952                "evidence_reference",
1953                text(dimension.evidence_reference.clone()),
1954            ),
1955            ("gaps", map_gaps(&detail.gaps)?),
1956        ]));
1957    }
1958    if seen.len() != NINE_DIMENSIONS.len() || details.len() != seen.len() {
1959        return Err(WireError::Coverage(
1960            "coverage dimension set is incomplete".to_owned(),
1961        ));
1962    }
1963    Ok(obj(vec![("dimensions", JcsValue::Array(mapped))]))
1964}

2513pub fn parse_conformance_spec(text: &str) -> Result<ConformanceSpec, WireError> {
2514    validate_spec_shape(text)?;
2515    let spec: ConformanceSpec =
2516        serde_json::from_str(text).map_err(|error| WireError::InvalidJson(error.to_string()))?;
2517    if spec.spec_version != WIRE_SCHEMA_VERSION {
2518        return Err(WireError::Spec("unregistered spec version".to_owned()));
2519    }
2520    require_nonempty(&spec.note, "spec note")?;
2521    if spec.artifacts.len() < 2 {
2522        return Err(WireError::Spec(
2523            "a conformance spec needs two artifacts".to_owned(),
2524        ));
2525    }
2526    let mut profiles = BTreeSet::new();
2527    let mut compressed = false;
2528    for artifact in &spec.artifacts {
2529        if artifact.records.len() < 2 {
2530            return Err(WireError::Spec(
2531                "each artifact needs multiple records".to_owned(),
2532            ));
2533        }
2534        profiles.insert(artifact.record_order_profile.kind());
2535        if artifact.compression == "gzip" {
2536            compressed = true;
2537        }
2538    }
2539    if profiles.len() != 2 || !compressed {
2540        return Err(WireError::Spec(
2541            "a conformance spec needs both profiles and one gzip artifact".to_owned(),
2542        ));
2543    }
2544    Ok(spec)
2545}

2547/// Prepare one abstract artifact: byte-derived sort, canonical stream,
2548/// registered compression and computed descriptor.
2549pub fn prepare_artifact(
2550    artifact: &SpecArtifact,
2551    policy: &ResourcePolicy,
2552) -> Result<PreparedArtifact, WireError> {
2553    validate_identifier(&artifact.logical_partition_id, "logical_partition_id")?;
2554    validate_artifact_path(&artifact.path)?;
2555    if artifact.encoding != ENCODING_NDJSON {
2556        return Err(WireError::Spec(
2557            "artifact encoding is not registered".to_owned(),
2558        ));
2559    }
2560    let internal_profile = artifact.record_order_profile.to_internal()?;
2561    let profile = map_artifact_profile(&internal_profile)?;
2562    let (compression, version) = match artifact.compression.as_str() {
2563        "none" => (ArtifactCompression::None, COMPRESSION_VERSION_NONE),
2564        "gzip" => (ArtifactCompression::Gzip, COMPRESSION_VERSION_GZIP),
2565        other => return Err(WireError::Compression(other.to_owned())),
2566    };
2567    validate_compression_pair(compression.as_str(), version)?;
2568    let mut records = Vec::with_capacity(artifact.records.len());
2569    for record in &artifact.records {
2570        records.push(CanonicalRecord {
2571            bytes: record.bytes.as_bytes().to_vec(),
2572        });
2573    }
2574    let ordered_records = sort_byte_derived(&records, &profile, policy)?;
2575    let mut event_types = BTreeSet::new();
2576    for bytes in &ordered_records {
2577        let parsed = parse_canonical_record(bytes, policy)?;
2578        event_types.insert(parsed.event_type);
2579    }
2580    let mut uncompressed = Vec::new();
2581    for bytes in &ordered_records {
2582        uncompressed.extend_from_slice(bytes);
2583        uncompressed.push(b'\n');
2584    }
2585    let uncompressed_len = u64::try_from(uncompressed.len())
2586        .map_err(|_| WireError::Policy("uncompressed length".to_owned()))?;
2587    if uncompressed_len > policy.max_uncompressed_bytes_per_artifact {
2588        return Err(WireError::Policy(
2589            "uncompressed artifact bound exceeded".to_owned(),
2590        ));
2591    }
2592    let stored = match compression {
2593        ArtifactCompression::None => uncompressed.clone(),
2594        ArtifactCompression::Gzip => {
2595            let mut encoder = GzEncoder::new(Vec::new(), GzipCompression::default());
2596            encoder
2597                .write_all(&uncompressed)
2598                .map_err(|error| WireError::Io("gzip record stream".to_owned(), error))?;
2599            encoder
2600                .finish()
2601                .map_err(|error| WireError::Io("finalize gzip stream".to_owned(), error))?
2602        }
2603    };
2604    let stored_len =
2605        u64::try_from(stored.len()).map_err(|_| WireError::Policy("stored length".to_owned()))?;
2606    if stored_len > policy.max_stored_bytes_per_artifact {
2607        return Err(WireError::Policy(
2608            "stored artifact bound exceeded".to_owned(),
2609        ));
2610    }
2611    let descriptor = ArtifactDescriptor {
2612        logical_partition_id: artifact.logical_partition_id.clone(),
2613        shard_ordinal: artifact.shard_ordinal,
2614        path: artifact.path.clone(),
2615        record_order_profile: internal_profile,
2616        encoding: artifact.encoding.clone(),
2617        compression: compression.as_str().to_owned(),
2618        compression_version: version.to_owned(),
2619        stored_bytes: stored_len,
2620        stored_sha256: sha256_hex(&stored),
2621        uncompressed_bytes: uncompressed_len,
2622        uncompressed_sha256: sha256_hex(&uncompressed),
2623        record_count: u64::try_from(ordered_records.len())
2624            .map_err(|_| WireError::Policy("record count".to_owned()))?,
2625    };
2626    Ok(PreparedArtifact {
2627        stored_bytes: stored,
2628        uncompressed_bytes: uncompressed,
2629        descriptor,
2630        ordered_records,
2631        event_types,
2632    })
2633}

2636/// order. Declared compression changes neither this digest nor its order.
2637pub fn dataset_digest(artifacts: &[PreparedArtifact]) -> String {

2645/// Sign the exact canonical manifest bytes with an injected key. The domain
2646/// prefix and digest binding are those of the ratified sidecar contract.
2647pub fn sign_manifest(
2648    manifest_bytes: &[u8],
2649    signing: &SigningKey,
2650    verifying: &VerifyingKey,
2651) -> Result<[u8; 64], WireError> {
2652    let digest = Sha256::digest(manifest_bytes);
2653    let mut message = Vec::with_capacity(SIGNING_DOMAIN.len() + digest.len());
2654    message.extend_from_slice(SIGNING_DOMAIN);
2655    message.extend_from_slice(&digest);
2656    let signature = signing.sign(&message);
2657    verifying
2658        .verify(&message, &signature)
2659        .map_err(|_| WireError::Signature("injected key self-check failed".to_owned()))?;
2660    Ok(signature.to_bytes())
2661}

2663/// Build the exactly five-member sidecar with a single trailing line feed.
2664pub fn sidecar_bytes(
2665    key_id: &str,
2666    manifest_sha256: &str,
2667    signature: &[u8; 64],
2668    policy: &ResourcePolicy,
2669) -> Result<Vec<u8>, WireError> {
2670    validate_identifier(key_id, "sidecar key_id")?;
2671    validate_sha256_hex(manifest_sha256, "sidecar manifest_sha256")?;
2672    let sidecar = obj(vec![
2673        (
2674            "schema_version",
2675            JcsValue::Integer(i64::from(SIDECAR_SCHEMA_VERSION)),
2676        ),
2677        ("algorithm", text(SIGNATURE_ALGORITHM)),
2678        ("key_id", text(key_id)),
2679        ("manifest_sha256", text(manifest_sha256)),
2680        ("signature_b64", text(BASE64.encode(signature))),
2681    ]);
2682    let mut bytes = canonicalize_strict(&sidecar, policy)
2683        .map_err(|error| WireError::Canonicalization(error.to_string()))?;
2684    bytes.push(b'\n');
2685    if u64::try_from(bytes.len()).map_err(|_| WireError::Policy("sidecar length".to_owned()))?
2686        > policy.max_sidecar_bytes
2687    {
2688        return Err(WireError::Policy("sidecar bound exceeded".to_owned()));
2689    }
2690    Ok(bytes)
2691}

2693/// Write one complete release directory from validated wire bytes. Files are
2694/// created exclusively; no existing release name is overwritten.
2695pub fn write_release(
2696    out: &Path,
2697    manifest_bytes: &[u8],
2698    signature_bytes: &[u8],
2699    artifacts: &[PreparedArtifact],
2700) -> Result<(), WireError> {
2701    fs::create_dir(out)
2702        .map_err(|error| WireError::Io(format!("create {}", out.display()), error))?;
2703    let manifest_path = out.join(crate::distribution::ndc_release::MANIFEST_FILE_NAME);
2704    let sidecar_path = out.join(SIDECAR_FILE_NAME);
2705    write_new(&manifest_path, manifest_bytes)?;
2706    write_new(&sidecar_path, signature_bytes)?;
2707    let artifact_root = out.join(crate::distribution::ndc_release::ARTIFACT_DIRECTORY);
2708    fs::create_dir(&artifact_root)
2709        .map_err(|error| WireError::Io(format!("create {}", artifact_root.display()), error))?;
2710    for artifact in artifacts {
2711        let path = artifact_root.join(
2712            artifact
2713                .descriptor
2714                .path
2715                .strip_prefix("artifacts/")
2716                .ok_or_else(|| WireError::ArtifactPath(artifact.descriptor.path.clone()))?,
2717        );
2718        write_new(&path, &artifact.stored_bytes)?;
2719    }
2720    Ok(())
2721}
2722
2723fn write_new(path: &Path, bytes: &[u8]) -> Result<(), WireError> {
2724    let mut file = fs::OpenOptions::new()
2725        .write(true)
2726        .create_new(true)
2727        .open(path)
2728        .map_err(|error| WireError::Io(format!("create {}", path.display()), error))?;
2729    file.write_all(bytes)
2730        .map_err(|error| WireError::Io(format!("write {}", path.display()), error))?;
2731    file.sync_all()
2732        .map_err(|error| WireError::Io(format!("sync {}", path.display()), error))?;
2733    Ok(())
2734}

3435/// Exact canonical release bytes produced from one abstract spec.
3436#[derive(Debug, Clone)]
3437pub struct ReleaseBytes {

3439    pub manifest_bytes: Vec<u8>,
3440    /// Five-member sidecar with a single trailing line feed.
3441    pub signature_bytes: Vec<u8>,
3442    /// Dataset digest over ordered uncompressed artifact streams.
3443    pub data_sha256: String,
3444}

3446/// Convert a byte-derived wire key into the internal producer key type.
3447pub(crate) fn wire_order_key_to_internal(key: &WireOrderKey) -> Result<RecordOrderKey, WireError> {
3448    match key {
3449        WireOrderKey::Sequenced {
3450            sequence_domain_id,
3451            sequence_epoch,
3452            sequence_number,
3453        } => Ok(RecordOrderKey::Sequenced {
3454            sequence_domain: sequence_domain_id.clone(),
3455            sequence_epoch: sequence_epoch.clone(),
3456            sequence_number: *sequence_number,
3457        }),
3458        WireOrderKey::Reference {
3459            object_id,
3460            revision,
3461            knowledge_time_ns,
3462            assertion_id,
3463        } => Ok(RecordOrderKey::Reference {
3464            object_id: object_id.clone(),
3465            revision: *revision,
3466            knowledge_time_ns: knowledge_time_ns.to_string(),
3467            assertion_id: assertion_id.clone(),
3468        }),
3469    }

3472/// Convert abstract spec artifacts into production builder inputs and collect
3473/// the event types observed in the validated record lines. The builder
3474/// derives order keys from the canonical bytes; the untrusted spec hints are
3475/// not copied. This is the single bridge from the abstract spec to
3476/// [`crate::distribution::ndc_release::build_release`].
3477pub fn spec_to_build_inputs(
3478    spec: &ConformanceSpec,
3479    policy: &ResourcePolicy,
3480) -> Result<(Vec<ArtifactInput>, BTreeSet<String>), WireError> {
3481    let mut inputs = Vec::with_capacity(spec.artifacts.len());
3482    let mut observed_event_types = BTreeSet::new();
3483    for artifact in &spec.artifacts {
3484        let prepared = prepare_artifact(artifact, policy)?;
3485        let profile = artifact.record_order_profile.to_internal()?;
3486        let mut records = Vec::with_capacity(prepared.ordered_records.len());
3487        for bytes in &prepared.ordered_records {
3488            let parsed = parse_canonical_record(bytes, policy)?;
3489            records.push(CanonicalRecord {
3490                bytes: bytes.clone(),
3491            });
3492            observed_event_types.insert(parsed.event_type);
3493        }
3494        let compression = match artifact.compression.as_str() {
3495            "none" => ArtifactCompression::None,
3496            "gzip" => ArtifactCompression::Gzip,
3497            other => return Err(WireError::Compression(other.to_owned())),
3498        };
3499        inputs.push(ArtifactInput {
3500            logical_partition_id: artifact.logical_partition_id.clone(),
3501            shard_ordinal: artifact.shard_ordinal,
3502            path: artifact.path.clone(),
3503            record_order_profile: profile,
3504            encoding: artifact.encoding.clone(),
3505            compression,
3506            records,
3507        });
3508    }
3509    Ok((inputs, observed_event_types))
3510}

3512/// Map one abstract spec to complete canonical release bytes. Artifacts are
3513/// canonically ordered by `(logical_partition_id, shard_ordinal)` before the
3514/// dataset digest and manifest inventory are computed.
3515pub fn build_conformance_release(
3516    spec: &ConformanceSpec,
3517    policy: &ResourcePolicy,
3518    signing: &SigningKey,
3519    verifying: &VerifyingKey,
3520) -> Result<(ReleaseBytes, Vec<PreparedArtifact>), WireError> {
3521    let mut artifacts = Vec::with_capacity(spec.artifacts.len());
3522    for artifact in &spec.artifacts {
3523        artifacts.push(prepare_artifact(artifact, policy)?);
3524    }
3525    artifacts.sort_by(|left, right| {
3526        let a = &left.descriptor;
3527        let b = &right.descriptor;
3528        (a.logical_partition_id.as_str(), a.shard_ordinal)
3529            .cmp(&(b.logical_partition_id.as_str(), b.shard_ordinal))
3530    });
3531    let data_sha256 = dataset_digest(&artifacts);
3532    let mut manifest = spec.manifest.clone();
3533    manifest.artifacts = artifacts
3534        .iter()
3535        .map(|item| item.descriptor.clone())
3536        .collect();
3537    manifest.digests.data_sha256 = data_sha256.clone();
3538    let mut event_types = BTreeSet::new();
3539    for artifact in &artifacts {
3540        event_types.extend(artifact.event_types.iter().cloned());
3541    }
3542    let wire_manifest = map_manifest(&manifest, &spec.evidence, &data_sha256, &event_types)?;
3543    let manifest_bytes = canonicalize_strict(&wire_manifest, policy)
3544        .map_err(|error| WireError::Canonicalization(error.to_string()))?;
3545    if u64::try_from(manifest_bytes.len())
3546        .map_err(|_| WireError::Policy("manifest length".to_owned()))?
3547        > policy.max_manifest_bytes
3548    {
3549        return Err(WireError::Policy("manifest bound exceeded".to_owned()));
3550    }
3551    let digest = Sha256::digest(&manifest_bytes);
3552    let manifest_sha256 = crate::distribution::ndc_release::hex_encode(&digest);
3553    let signature = sign_manifest(&manifest_bytes, signing, verifying)?;
3554    let signature_bytes = sidecar_bytes(
3555        &manifest.publication_identity.signing_key_id,
3556        &manifest_sha256,
3557        &signature,
3558        policy,
3559    )?;
3560    Ok((
3561        ReleaseBytes {
3562            manifest_bytes,
3563            signature_bytes,
3564            data_sha256,
3565        },
3566        artifacts,
3567    ))
