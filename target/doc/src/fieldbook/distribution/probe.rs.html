1//! `network-probe`: a transport diagnostic for the collector's egress path.
2//!
3//! The probe never authenticates to a venue and never collects market data.
4//! It exercises exactly one real stage sequence — DNS, TCP, optional proxy
5//! negotiation, TLS and the websocket upgrade — and reports them separately,
6//! so "connected" cannot be mistaken for "a feed session is healthy".
7//!
8//! Result levels (see ADR 0002): the probe implements levels 0–2 (config
9//! parses, route is usable, TLS validates and the upgrade completes). Levels
10//! 3–4 require a venue-approved session, live subscription, capture, journal
11//! verification and deterministic replay, which are out of scope here.
12//!
13//! Credentials never appear in the report. The venue URL is operator input;
14//! proxy credentials are read from the environment and are not printed,
15//! logged, or hashed.
16

29/// Outcome taxonomy. `transport_ok` means the websocket handshake completed
30/// (HTTP 101 or a closed-but-reached peer for non-HTTP upgrades); it is not a
31/// statement about feed completeness.
32const RESULT_OK: &str = "transport_ok";
33const RESULT_FAILED: &str = "failed";
34
35/// Machine-readable probe report, written to stdout or `--out`.
36#[derive(Debug, Clone, Serialize)]
37pub struct ProbeReport {
38    pub schema_version: u32,
39    pub tool: &'static str,
40    pub binary_version: &'static str,
41    pub git_commit: &'static str,
42    /// Non-secret route provenance; credentials can never appear.
43    pub route: RouteProvenance,
44    /// The target websocket URL the operator supplied (or the configured
45    /// exchange endpoint).
46    pub target: String,
47    /// Stage timings in attempt order. Missing stages mean the attempt did
48    /// not reach them.
49    pub stages: Vec<ProbeStage>,
50    /// Wall time of the whole attempt, in microseconds.
51    pub total_us: u64,
52    /// `transport_ok` or `failed`.
53    pub result: String,
54    /// Deterministic classifier label for failures; null on success.
55    pub reason: Option<String>,
56}
57
58impl ProbeReport {
59    /// Whether the websocket handshake completed.
60    #[must_use]
61    pub fn is_transport_ok(&self) -> bool {
62        self.result == RESULT_OK
63    }
64}
65
66/// Map a transport failure onto a deterministic reason label.
67///
68/// The match uses the anyhow chain constructed in
69/// `ws.rs`; it is a routing aid for humans and CI, not a parser contract.
70/// Authentication rejections win over every other class.
71#[must_use]
72pub fn classify_reason(error: &anyhow::Error) -> String {
73    if let Some(status) = ws::http_status(error)
74        && matches!(status, 401 | 403)
75    {
76        return "venue_auth".to_owned();
77    }
78    let text = format!("{error:#}");
79    for (needle, reason) in [
80        ("timed out", "timeout"),
81        ("resolving", "dns"),
82        ("proxy host connect", "proxy_tcp_connect"),
83        ("proxy CONNECT refused", "proxy_tunnel"),
84        ("proxy closed", "proxy_tunnel"),
85        ("SOCKS5 proxy negotiation", "proxy_tunnel"),
86        ("Socks5", "proxy_tunnel"),
87        ("websocket handshake", "websocket_upgrade"),
88        ("binding local address", "tcp_bind"),
89        ("TCP connect to", "tcp_connect"),
90        ("creating TCP socket", "tcp_socket"),
91    ] {
92        if text.contains(needle) {
93            return reason.to_owned();
94        }
95    }
96    "unknown".to_owned()
97}

99/// Run the probe against `target` using the loaded settings, resolving the
100/// route from the process environment.
101///
102/// A target must be supplied explicitly with `--target`; no venue endpoint
103/// default is registered after the RRP-003 adapter retirement. A
104/// configuration-level failure is a failed report (`reason: "config"`), not a
105/// panic.
106pub async fn run(settings: &Settings, target: Option<&str>) -> anyhow::Result<ProbeReport> {
107    match EgressRoute::from_network(&settings.network) {
108        Ok(route) => run_with_route(settings, route, target).await,
109        Err(_error) => {
110            let target_report = target.map_or_else(|| default_target(settings), str::to_owned);
111            Ok(failed_config_report(&target_report))
112        }
113    }
114}

116/// Run the probe with an explicit route. Semantically identical to [`run`]
117/// once the route exists; it is the deterministic entry point for tests and
118/// for callers that resolved the route themselves.
119pub async fn run_with_route(
120    settings: &Settings,
121    route: EgressRoute,
122    target: Option<&str>,
123) -> anyhow::Result<ProbeReport> {
124    let Some(target) = target else {
125        anyhow::bail!(
126            "network-probe requires an explicit --target; no venue endpoint default is registered"
127        );
128    };
129    let provenance = route.provenance(&settings.network);
130    // The probe declares a source-contact request from the caller-supplied
131    // intent and the actual target. No permission is inferred from the target.
132    let intent = settings
133        .network
134        .declared_intent
135        .as_ref()
136        .map(|declared| declared.for_source_contact(target))
137        .transpose()?;
138
139    let mut options = ConnectOptions {
140        tcp_nodelay: settings.network.tcp_nodelay,
141        source_ip: settings.network.source_ip,
142        source_port: settings.network.source_port,
143        attempts: 1,
144        connect_timeout: std::time::Duration::from_secs(
145            settings.network.connect_timeout_secs.max(1),
146        ),
147        egress: route,
148        trace: Some(ProbeTrace::begin()),
149        venue: settings.network.venue.clone(),
150        // A network probe only contacts the source; it never captures data.
151        operation: Operation::SourceContact,
152        intent,
153        denial_transport: None,
154    };
155
156    let started = Instant::now();
157    let outcome = ws::connect(target, &[], &options).await;
158    let total_us = started.elapsed().as_micros() as u64;
159    let trace = options.trace.take();
160    let stages = trace.map(ProbeTrace::finish).unwrap_or_default();
161
162    let (result, reason) = match &outcome {
163        Ok(_) => (RESULT_OK.to_owned(), None),
164        Err(error) => (RESULT_FAILED.to_owned(), Some(classify_reason(error))),
165    };
166    let report = ProbeReport {
167        schema_version: 1,
168        tool: "fieldbook network-probe",
169        binary_version: env!("CARGO_PKG_VERSION"),
170        git_commit: option_env!("GIT_COMMIT")
171            .or(option_env!("GITHUB_SHA"))
172            .unwrap_or("unknown"),
173        route: provenance,
174        target: target.to_owned(),
175        stages,
176        total_us,
177        result,
178        reason,
179    };
180    Ok(report)
181}

183fn failed_config_report(target: &str) -> ProbeReport {
184    // The reason label is coarse: configuration failures are
185    // corrected by the operator, not classified for machines.
186    ProbeReport {
187        schema_version: 1,
188        tool: "fieldbook network-probe",
189        binary_version: env!("CARGO_PKG_VERSION"),
190        git_commit: option_env!("GIT_COMMIT")
191            .or(option_env!("GITHUB_SHA"))
192            .unwrap_or("unknown"),
193        route: RouteProvenance {
194            mode: "unresolved".to_owned(),
195            kind: "unresolved".to_owned(),
196            host: None,
197            port: None,
198            remote_dns: None,
199            label: None,
200            fingerprint_sha256: sha256_hex(b"egress-route/v1|unresolved"),
201        },
202        target: target.to_owned(),
203        stages: Vec::new(),
204        total_us: 0,
205        result: RESULT_FAILED.to_owned(),
206        reason: Some("config".to_owned()),
207    }
208}

210/// Diagnostic placeholder used only in a configuration-failure report.
211///
212/// A live probe always requires an explicit target; no venue endpoint is
213/// registered in the retired-adapter tree.
214fn default_target(_settings: &Settings) -> String {
215    "unspecified".to_owned()
216}

218/// Serialize a report to JSON text. Serialization of this fixed schema
219/// cannot fail; failure collapses to a minimal marker instead of panicking.
220#[must_use]
221pub fn to_json_string_lossy(report: &ProbeReport) -> String {
222    match serde_json::to_string_pretty(report) {
223        Ok(text) => text,
224        Err(_) => "{\"result\":\"failed\"}".to_owned(),
225    }
