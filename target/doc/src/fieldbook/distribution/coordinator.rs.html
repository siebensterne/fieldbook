41/// Shard identifier reserved for a dedicated market-lifecycle control plane.
42/// Book shards are validated to `<= 256`, so `u16::MAX` never collides with a
43/// book source. Exported for the report's source list; consumers should use
44/// `source.shard == LIFECYCLE_SHARD` to distinguish the control plane.
45pub const LIFECYCLE_SHARD: u16 = u16::MAX;

47/// Debounce for lifecycle-triggered discovery reconciliation, so a burst of
48/// notifications cannot spawn overlapping REST rounds.
49const FORCED_RECONCILE_COOLDOWN_US: i64 = 5_000_000;

51/// Owns the shared REST-discovery in-flight state and spawns reconciliation
52/// rounds off the event loop. Used by the periodic ticker and by the lifecycle
53/// control plane: a lifecycle event for an untracked market, or a lifecycle
54/// socket reconnecting after missed notifications, requests an out-of-cycle
55/// round.
56struct DiscoveryPlane {
57    exchange: Arc<dyn Exchange>,
58    max_markets: usize,
59    tx: mpsc::Sender<anyhow::Result<DiscoveryResult>>,
60    in_flight: bool,
61    last_forced_us: i64,
62}
63
64impl DiscoveryPlane {
65    fn new(
66        exchange: Arc<dyn Exchange>,
67        max_markets: usize,
68        tx: mpsc::Sender<anyhow::Result<DiscoveryResult>>,
69    ) -> Self {
70        Self {
71            exchange,
72            max_markets,
73            tx,
74            in_flight: false,
75            last_forced_us: 0,
76        }
77    }
78
79    /// Fire a discovery round unless one is already in flight or the last
80    /// forced round started less than `min_gap_us` ago. Returns true when a
81    /// round was spawned.
82    fn request(&mut self, min_gap_us: i64) -> bool {
83        if self.in_flight {
84            return false;
85        }
86        let now = now_us().unwrap_or(i64::MAX);
87        if now - self.last_forced_us < min_gap_us {
88            return false;
89        }
90        self.in_flight = true;
91        self.last_forced_us = now;
92        spawn_discovery_round(&self.exchange, self.max_markets, &self.tx);
93        true
94    }
95
96    fn mark_finished(&mut self) {
97        self.in_flight = false;
98    }
99}

101fn spawn_discovery_round(
102    exchange: &Arc<dyn Exchange>,
103    max_markets: usize,
104    result_tx: &mpsc::Sender<anyhow::Result<DiscoveryResult>>,
105) {
106    let exchange = Arc::clone(exchange);
107    let result_tx = result_tx.clone();
108    // The release profile uses panic=abort, so catch_unwind cannot provide
109    // recovery there. A panic is process-fatal and systemd restarts from a
110    // clean state; ordinary discovery failures use Result.
111    tokio::spawn(async move {
112        let result = exchange.discover_markets_with_stats(max_markets).await;
113        let _ = result_tx.send(result).await;
114    });
115}

136/// Start the raw-frame journal when a directory is configured.
137///
138/// The manifest records the source commit injected at build time and a digest
139/// of the capture-affecting configuration. Credentials never enter either
140/// value.
141fn start_journal(
142    settings: &Settings,
143    venue: &str,
144    append_latency: Option<Arc<Histogram>>,
145) -> anyhow::Result<Option<JournalHandle>> {
146    let Some(directory) = settings.journal.directory.clone() else {
147        return Ok(None);
148    };
149    let config = JournalConfig {
150        directory,
151        queue_capacity: settings.journal.queue_capacity,
152        enqueue_timeout: Duration::from_millis(settings.journal.enqueue_timeout_ms),
153        flush_interval: Duration::from_millis(settings.journal.flush_interval_ms),
154        sync_interval: Duration::from_millis(settings.journal.sync_interval_ms),
155        segment_max_bytes: settings.journal.segment_max_bytes,
156        segment_max_seconds: settings.journal.segment_max_secs,
157        sync_mode: match settings.journal.sync_mode {
158            JournalSyncMode::Interval => SyncMode::Interval,
159            JournalSyncMode::Always => SyncMode::Always,
160        },
161    };
162    let code_commit = option_env!("GIT_COMMIT")
163        .or(option_env!("GITHUB_SHA"))
164        .unwrap_or("unknown");
165    let config_sha256 = settings.capture_config_sha256()?;
166    // Settings validation has already run at load time; a failure here means
167    // the proxy environment changed after validation. Stop rather than record
168    // provenance for a route that was not validated.
169    let route = Some(
170        crate::egress::EgressRoute::from_network(&settings.network)
171            .context("resolving egress route for journal provenance")?
172            .provenance(&settings.network),
173    );
174    let meta = JournalMeta::new(venue, code_commit, &config_sha256, route);
175    let started = match append_latency {
176        Some(histogram) => JournalHandle::start_with_timing(config, meta, histogram),
177        None => JournalHandle::start(config, meta),
178    };
179    started.map(Some).context("starting raw-frame journal")
180}

193/// Startup subscription set and its provenance.
194#[derive(Debug)]
195struct Bootstrap {
196    markets: Vec<MarketInfo>,
197    discovery: Option<DiscoveryStats>,
198    catalog: CatalogReport,
199    /// Catalog generation time to persist with the startup snapshot. `None`

204fn catalog_mode_name(mode: CatalogMode) -> &'static str {
205    match mode {
206        CatalogMode::Off => "off",
207        CatalogMode::Warm => "warm",
208        CatalogMode::Require => "require",
209        CatalogMode::Stale => "stale",
210    }
211}

213/// Load the startup set according to the catalog policy.
214///
215/// A checkpoint is trusted only after its schema, counts, market fields, and
216/// content checksum verify. `warm` uses a fresh checkpoint and falls back to
217/// discovery; `require` demands a fresh checkpoint; `stale` accepts
218/// an old checkpoint so a total discovery outage does not block a restart.
219async fn bootstrap_markets(
220    settings: &Settings,
221    exchange: &Arc<dyn Exchange>,
222    now_us: i64,
223) -> anyhow::Result<Bootstrap> {
224    let mode = settings.collector.catalog_mode;
225    let mut report = CatalogReport {
226        mode: catalog_mode_name(mode).to_owned(),
227        path: settings
228            .collector
229            .catalog_path
230            .as_ref()
231            .map(|path| path.display().to_string()),
232        source: "cold".to_owned(),
233        generated_at_us: None,
234        age_us: None,
235        checksum: None,
236        selected_markets: 0,
237        outcome_symbols: 0,
238    };
239    if mode != CatalogMode::Off {
240        let path = settings
241            .collector
242            .catalog_path
243            .as_deref()
244            .context("catalog mode requires collector.catalog_path")?;
245        match Catalog::load(path) {
246            Ok(catalog) => {
247                let age_us = catalog.age_us(now_us);
248                let max_age_us = i64::try_from(
249                    settings
250                        .collector
251                        .catalog_max_age_secs
252                        .saturating_mul(1_000_000),
253                )
254                .unwrap_or(i64::MAX);
255                let fresh = age_us <= max_age_us;
256                let nonempty =
257                    !catalog.markets.is_empty() || settings.collector.allow_empty_catalog;
258                let exchange_matches = catalog.exchange == exchange.name();
259                let usable = nonempty && exchange_matches && (fresh || mode == CatalogMode::Stale);
260                if usable {
261                    report.source = if fresh { "warm" } else { "warm-stale" }.to_owned();
262                    report.generated_at_us = Some(catalog.generated_at_us);
263                    report.age_us = Some(age_us);
264                    report.checksum = Some(catalog.checksum.clone());
265                    report.selected_markets = catalog.selected_markets;
266                    report.outcome_symbols = catalog.outcome_symbols;
267                    info!(
268                        path = %path.display(),
269                        source = %report.source,
270                        age_secs = age_us / 1_000_000,
271                        selected_markets = catalog.selected_markets,
272                        outcome_symbols = catalog.outcome_symbols,
273                        "using catalog checkpoint"
274                    );
275                    return Ok(Bootstrap {
276                        markets: catalog.markets,
277                        discovery: None,
278                        catalog: report,
279                        catalog_generated_at_us: None,
280                    });
281                }
282                let reason = if !exchange_matches {
283                    "belongs to another exchange"
284                } else if !nonempty {
285                    "contains no markets"
286                } else {
287                    "is older than catalog_max_age_secs"
288                };
289                if matches!(mode, CatalogMode::Require | CatalogMode::Stale) {
290                    bail!(
291                        "catalog checkpoint `{}` {reason}; refusing to start",
292                        path.display()
293                    );
294                }
295                warn!(
296                    path = %path.display(),
297                    reason,
298                    "catalog checkpoint unusable; falling back to discovery"
299                );
300            }
301            Err(error) => {
302                if matches!(mode, CatalogMode::Require | CatalogMode::Stale) {
303                    return Err(error).with_context(|| {
304                        format!("loading required catalog checkpoint `{}`", path.display())
305                    });
306                }
307                warn!(
308                    path = %path.display(),
309                    error = %format!("{error:#}"),
310                    "catalog checkpoint unreadable; falling back to discovery"
311                );
312            }
313        }
314    }
315
316    let discovery =
317        discover_with_retry_stats(exchange, settings.collector.max_markets, exchange.name())
318            .await?;
319    if discovery.markets.is_empty() && !settings.collector.allow_empty_catalog {
320        bail!(
321            "{} returned no live, non-MVE markets; refusing to start an empty collector",
322            exchange.name()
323        );
324    }
325    let stats = discovery.stats;
326    let finished_at_us = stats.finished_at_us;
327    report.source = "cold".to_owned();
328    report.generated_at_us = Some(finished_at_us);
329    report.age_us = Some(0);
330    report.checksum = Some(crate::catalog::checksum_for(
331        exchange.name(),
332        stats.finished_at_us,
333        &discovery.markets,
334    )?);
335    report.selected_markets = usize::try_from(stats.selected_markets).unwrap_or(usize::MAX);
336    report.outcome_symbols = usize::try_from(stats.outcome_symbols).unwrap_or(usize::MAX);
337    Ok(Bootstrap {
338        markets: discovery.markets,
339        discovery: Some(stats),
340        catalog: report,
341        catalog_generated_at_us: Some(finished_at_us),
342    })
343}

345/// Run a live collection until SIGINT/SIGTERM, the configured duration, or a
346/// fatal source/writer error.
347pub async fn run(
348    settings: Settings,
349    authorization: SessionAuthorization,
350) -> anyhow::Result<RunReport> {
351    let duration_secs = settings.collector.duration_secs;
352    run_with_shutdown(settings, authorization, async move {
353        let signal = wait_for_shutdown_signal();
354        tokio::pin!(signal);
355        if duration_secs == 0 {
356            signal.await
357        } else {
358            tokio::select! {
359                reason = &mut signal => reason,
360                _ = tokio::time::sleep(Duration::from_secs(duration_secs)) => {
361                    format!("duration elapsed ({duration_secs}s)")
362                }
363            }
364        }
365    })
366    .await
367}

369/// Run a live collection until the supplied future resolves.
370///
371/// `shutdown` returns the human-readable shutdown reason recorded in the run
372/// report. This is the same loop used by [`run`]; separating the trigger keeps
373/// the live orchestration testable without relying on process signals or a
374/// fixed wall-clock duration. `collector.duration_secs` is interpreted by
375/// [`run`], not by this function.
376pub async fn run_with_shutdown<F>(
377    settings: Settings,
378    authorization: SessionAuthorization,
379    shutdown: F,
380) -> anyhow::Result<RunReport>
381where
382    F: std::future::Future<Output = String>,
383{
384    settings.validate()?;
385    // Refuse to replace any previous capture or sidecar before discovery or
386    // writer creation can touch the filesystem.
387    crate::config::preflight_output(&settings)?;
388    let started_at_us = now_us().context("reading wall clock at startup")?;
389    let exchange = build_exchange(&settings, true)?;
390    let bootstrap = bootstrap_markets(&settings, &exchange, started_at_us).await?;
391    // A qualified venue adapter may narrow discovery to the set the session
392    // will subscribe. Metadata, desired subscriptions and coverage must all
393    // use that same set or the certificate demands anchors that were never
394    // requested.
395    let initial_markets = exchange.subscribed_universe(bootstrap.markets);
396    if initial_markets.is_empty() {
397        bail!(
398            "{} returned no live, non-MVE markets; refusing to start an empty collector",
399            exchange.name()
400        );
401    }
402    let capacity =
403        capacity::plan(&settings, &initial_markets).context("planning collector capacity")?;
404    let shard_count = capacity.shards;
405    let replica_count = settings.collector.replicas;
406    let core_config = CoreConfig::from_settings(&settings, shard_count, exchange.capabilities())?;
407    let mut core = CollectorCore::new(core_config)?;
408    // One set of stage histograms is shared with the writer, journal, and
409    // final report. No worker computes or averages its own percentile.
410    let journal_latency = settings
411        .journal
412        .directory
413        .as_ref()
414        .map(|_| Arc::new(Histogram::default()));
415    let timings = StageTimings {
416        journal_append: journal_latency.clone(),
417        ..StageTimings::default()
418    };
419    // The initial discovery cannot produce removal snapshots, but applying it
420    // here establishes shard assignments before workers are spawned.
421    core.apply_discovery(started_at_us, initial_markets.clone())
422        .context("applying initial market discovery")?;
423
424    let persistence = PersistenceHandle::start(
425        sidecar_path(&settings.output.metadata_path).map(std::path::Path::to_path_buf),
426        settings.collector.catalog_path.clone(),
427        exchange.name(),
428    )
429    .context("starting metadata/catalog persistence worker")?;
430    let persistence_timeout =
431        Duration::from_secs(settings.collector.writer_send_timeout_secs.max(1));
432    persistence
433        .submit(
434            Arc::new(core.market_infos()),
435            started_at_us,
436            bootstrap.catalog_generated_at_us,
437        )
438        .context("queueing initial metadata/catalog snapshot")?;
439    let writer = WriterHandle::start_with_timings(
440        crate::writer::WriterOptions::from_settings(&settings),
441        Arc::new(timings.writer.clone()),
442    )?;
443    let send_timeout = persistence_timeout;
444    let registry = MetricsRegistry::new(
445        exchange.name(),
446        ReadinessPolicy {
447            min_anchored_fraction: settings.collector.readiness_min_anchored_fraction,
448            max_catalog_age_secs: settings.collector.readiness_max_catalog_age_secs,
449            max_stale_secs: settings.collector.readiness_max_stale_secs,
450        },
451    );
452    let stale_after_us = i64::try_from(
453        settings
454            .collector
455            .readiness_max_stale_secs
456            .saturating_mul(1_000_000),
457    )
458    .unwrap_or(i64::MAX);
459    let mut runtime = RuntimeState {
460        catalog: CatalogHealth {
461            mode: bootstrap.catalog.mode.clone(),
462            source: bootstrap.catalog.source.clone(),
463            generated_at_us: bootstrap.catalog.generated_at_us,
464            last_discovery_finished_at_us: bootstrap
465                .discovery
466                .as_ref()
467                .map(|round| round.finished_at_us)
468                .or(bootstrap.catalog.generated_at_us),
469            last_discovery_error: None,
470        },
471        discovery: bootstrap.discovery.clone(),
472        discovery_failures: 0,
473        fatal_error: None,
474        lifecycle_configured: 0,
475    };
476    let (metrics_shutdown, metrics_handle) = start_metrics(&settings, &registry).await?;
477    let journal = start_journal(&settings, exchange.name(), journal_latency)?;
478
479    let factory = Arc::clone(&exchange).session_factory();
480    let mut worker_config = WorkerConfig::from_settings(&settings.collector);
481    worker_config.journal = journal.clone();
482    // One authorization lifecycle shared by every source worker. All sockets
483    // observe the same gate, authority handle and checkpoint schedule.
484    worker_config.authorization = Some(authorization);
485    let (worker_event_tx, mut worker_event_rx) = mpsc::channel::<WorkerEvent>(8192);
486    let (shutdown_tx, _) = broadcast::channel::<()>(1);
487    let mut commands: BTreeMap<SourceId, mpsc::Sender<Control>> = BTreeMap::new();
488    let mut handles: BTreeMap<SourceId, JoinHandle<()>> = BTreeMap::new();
489    let mut active_sources = BTreeSet::new();
490    // Last successfully applied subscription set per source. Kept across
491    // disconnect transitions so the shutdown drain still observes every
492    // subscription that was ever in force.
493    let mut acknowledged: BTreeMap<SourceId, BTreeSet<String>> = BTreeMap::new();
494    for shard in 0..shard_count {
495        for replica in 0..replica_count {
496            let source = SourceId::new(shard as u16, replica as u16);
497            let (command_tx, command_rx) = mpsc::channel::<Control>(256);
498            commands.insert(source, command_tx);
499            active_sources.insert(source);
500            let handle = tokio::spawn(run_worker(
501                Arc::clone(&factory),
502                source,
503                command_rx,
504                worker_event_tx.clone(),
505                worker_config.clone(),
506                shutdown_tx.subscribe(),
507            ));
508            handles.insert(source, handle);
509        }
510    }
511    // Dedicated lifecycle control plane. One supervised socket per configured
512    // lifecycle replica, carrying `market_lifecycle_v2` only; book shards
513    // never see the lifecycle channel. A venue without a separate lifecycle
514    // channel (or missing credentials) returns no factory and no sockets are
515    // spawned.
516    let lifecycle_factory = exchange.clone().lifecycle_session_factory();
517    let lifecycle_replicas = if lifecycle_factory.is_some() {
518        settings.collector.lifecycle_replicas
519    } else {
520        0
521    };
522    let mut active_lifecycle = BTreeSet::new();
523    if let Some(lifecycle_factory) = lifecycle_factory {
524        for index in 0..lifecycle_replicas {
525            let Ok(index) = u16::try_from(index) else {
526                break;
527            };
528            let source = SourceId::new(LIFECYCLE_SHARD, index);
529            // The command channel keeps the worker's command drain alive; no
530            // book-plane control is ever dispatched to this source.
531            let (command_tx, command_rx) = mpsc::channel::<Control>(8);
532            commands.insert(source, command_tx);
533            let handle = tokio::spawn(run_worker(
534                Arc::clone(&lifecycle_factory),
535                source,
536                command_rx,
537                worker_event_tx.clone(),
538                worker_config.clone(),
539                shutdown_tx.subscribe(),
540            ));
541            handles.insert(source, handle);
542            active_lifecycle.insert(source);
543        }
544    }
545    if !active_lifecycle.is_empty() {
546        info!(
547            lifecycle_connections = active_lifecycle.len(),
548            "dedicated lifecycle control plane started"
549        );
550    }
551    runtime.lifecycle_configured = active_lifecycle.len() as u64;
552    drop(worker_event_tx);
553
554    dispatch_controls(&mut core, &commands, send_timeout).await?;
555    update_metrics(
556        &registry,
557        &core,
558        &writer,
559        &persistence,
560        &runtime,
561        stale_after_us,
562    );
563    info!(
564        exchange = exchange.name(),
565        environment = exchange.environment(),
566        symbols = core.symbol_count(),
567        shards = shard_count,
568        replicas = replica_count,
569        output = %settings.output.path,
570        metadata = ?settings.output.metadata_path,
571        report = ?settings.output.report_path,
572        journal = ?settings.journal.directory,
573        capacity = %capacity.summary(),
574        "collector started"
575    );
576
577    let mut ticker = interval(Duration::from_secs(1));
578    ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
579    let mut discovery_ticker = interval(settings.discovery_interval());
580    discovery_ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
581    // The initial discovery already ran; consume the immediate interval tick.
582    discovery_ticker.tick().await;
583    let mut flush_ticker = interval(Duration::from_secs(
584        settings.collector.flush_interval_secs.max(1),
585    ));
586    flush_ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
587    flush_ticker.tick().await;
588    let (discovery_tx, mut discovery_rx) = mpsc::channel::<anyhow::Result<DiscoveryResult>>(1);
589    let mut discovery = DiscoveryPlane::new(
590        Arc::clone(&exchange),
591        settings.collector.max_markets,
592        discovery_tx,
593    );
594
595    let mut shutdown = std::pin::pin!(shutdown);
596
597    let mut shutdown_reason = String::new();
598    let mut fatal: Option<String> = None;
599    loop {
600        tokio::select! {
601            reason = &mut shutdown => {
602                shutdown_reason = reason;
603                break;
604            }
605            event = worker_event_rx.recv() => {
606                match event {
607                    Some(event) => {
608                        if let Err(error) = process_worker_event(
609                            event,
610                            &mut core,
611                            &writer,
612                            &commands,
613                            &mut active_sources,
614                            &mut acknowledged,
615                            &mut discovery,
616                            &timings.core_processing,
617                            send_timeout,
618                        ).await {
619                            fatal = Some(format!("{error:#}"));
620                            break;
621                        }
622                        // Book-plane collapse is fatal even if the lifecycle
623                        // control plane is still connected.
624                        if active_sources.is_empty() {
625                            fatal = Some("all websocket sources stopped".to_owned());
626                            break;
627                        }
628                    }
629                    None => {
630                        fatal = Some("worker event channel closed unexpectedly".to_owned());
631                        break;
632                    }
633                }
634            }
635            _ = ticker.tick() => {
636                // Every serialized timestamp comes from the wall clock, so
637                // generated snapshots share one timeline with event receive
638                // times. The tick cadence itself is the Tokio timer.
639                let now = now_us().unwrap_or(started_at_us);
640                let outputs = core.tick(now);
641                if let Err(error) = emit_outputs(&writer, outputs, send_timeout).await {
642                    fatal = Some(format!("{error:#}"));
643                    break;
644                }
645                if let Err(error) = dispatch_controls(&mut core, &commands, send_timeout).await {
646                    fatal = Some(format!("{error:#}"));
647                    break;
648                }
649                update_metrics(
650                    &registry,
651                    &core,
652                    &writer,
653                    &persistence,
654                    &runtime,
655                    stale_after_us,
656                );
657            }
658            _ = discovery_ticker.tick() => {
659                // Run REST discovery off the event loop. Paginating thousands
660                // of markets while holding this loop would stop draining
661                // websocket events and can cause avoidable gaps.
662                let _ = discovery.request(0);
663            }
664            result = discovery_rx.recv() => {
665                discovery.mark_finished();
666                match result {
667                    Some(Ok(round))
668                        if round.markets.is_empty()
669                            && core.symbol_count() > 0
670                            && !settings.collector.allow_empty_catalog =>
671                    {
672                        runtime.discovery_failures =
673                            runtime.discovery_failures.saturating_add(1);
674                        runtime.catalog.last_discovery_error = Some(
675                            "complete discovery round returned no markets; \
676                             the current subscription set was retained"
677                                .to_owned(),
678                        );
679                        warn!(
680                            current = core.symbol_count(),
681                            "discovery returned no markets; keeping the current subscription set"
682                        );
683                        update_metrics(
684                            &registry,
685                            &core,
686                            &writer,
687                            &persistence,
688                            &runtime,
689                            stale_after_us,
690                        );
691                    }
692                    Some(Ok(round)) => {
693                        let DiscoveryResult { markets, stats } = round;
694                        let markets = exchange.subscribed_universe(markets);
695                        let round_started_us = stats.started_at_us;
696                        let finished_at_us = stats.finished_at_us;
697                        let now = now_us().unwrap_or(started_at_us);
698                        let admission = core.projected_shard_loads(&markets)
699                            .and_then(|loads| capacity::plan_fixed_topology(&settings, &markets, &loads));
700                        match admission
701                            .and_then(|_| core.apply_discovery_with_epoch(now, markets, Some(round_started_us)))
702                        {
703                            Ok(outputs) => {
704                                if let Err(error) =
705                                    emit_outputs(&writer, outputs, send_timeout).await
706                                {
707                                    fatal = Some(format!("{error:#}"));
708                                    break;
709                                }
710                                if let Err(error) =
711                                    dispatch_controls(&mut core, &commands, send_timeout).await
712                                {
713                                    fatal = Some(format!("{error:#}"));
714                                    break;
715                                }
716                                runtime.catalog.last_discovery_finished_at_us =
717                                    Some(finished_at_us);
718                                runtime.catalog.last_discovery_error = None;
719                                runtime.discovery = Some(stats);
720                                if let Err(error) = persistence.submit(
721                                    Arc::new(core.market_infos()),
722                                    now,
723                                    Some(finished_at_us),
724                                ) {
725                                    runtime.catalog.last_discovery_error =
726                                        Some(format!("{error:#}"));
727                                    warn!(
728                                        error = %format!("{error:#}"),
729                                        "failed to queue metadata/catalog snapshot"
730                                    );
731                                }
732                                update_metrics(
733                                    &registry,
734                                    &core,
735                                    &writer,
736                                    &persistence,
737                                    &runtime,
738                                    stale_after_us,
739                                );
740                            }
741                            Err(error) => {
742                                runtime.discovery_failures =
743                                    runtime.discovery_failures.saturating_add(1);
744                                runtime.catalog.last_discovery_error =
745                                    Some(format!("{error:#}"));
746                                warn!(
747                                    error = %format!("{error:#}"),
748                                    "invalid discovery result; retaining current subscriptions"
749                                );
750                                update_metrics(
751                                    &registry,
752                                    &core,
753                                    &writer,
754                                    &persistence,
755                                    &runtime,
756                                    stale_after_us,
757                                );
758                            }
759                        }
760                    }
761                    Some(Err(error)) => {
762                        runtime.discovery_failures =
763                            runtime.discovery_failures.saturating_add(1);
764                        runtime.catalog.last_discovery_error = Some(format!("{error:#}"));
765                        warn!(error = %format!("{error:#}"), "market discovery failed; retaining current subscriptions");
766                        update_metrics(
767                            &registry,
768                            &core,
769                            &writer,
770                            &persistence,
771                            &runtime,
772                            stale_after_us,
773                        );
774                    }
775                    None => {
776                        warn!("market discovery result channel closed; continuing");
777                    }
778                }
779            }
780            _ = flush_ticker.tick() => {
781                if let Err(error) = writer.flush(send_timeout).await {
782                    fatal = Some(format!("{error:#}"));
783                    break;
784                }
785                update_metrics(
786                    &registry,
787                    &core,
788                    &writer,
789                    &persistence,
790                    &runtime,
791                    stale_after_us,
792                );
793            }
794        }
795    }
796
797    if let Some(error) = fatal {
798        shutdown_reason = format!("fatal: {error}");
799        error!(error, "collector stopping after fatal error");
800    }
801    info!(reason = %shutdown_reason, "shutting down collector");
802    let mut shutdown_error: Option<anyhow::Error> = None;
803
804    // Stop accepting new work first. The broadcast wakes workers blocked in a
805    // handshake or backoff; the per-source command is best-effort because a
806    // full queue must never delay process shutdown.
807    let _ = shutdown_tx.send(());
808    for command in commands.values() {
809        let _ = command.try_send(Control::Shutdown);
810    }
811
812    // Apply every event already accepted by the channel before the final
813    // snapshot. Workers stop producing after the broadcast, so this reaches a
814    // fixed point. Status transitions are deferred: a worker disconnect would
815    // otherwise invalidate the books that the final snapshot must capture.
816    let mut deferred_statuses = Vec::new();
817    let drain_deadline = tokio::time::Instant::now() + send_timeout;
818    'drain: while !active_sources.is_empty() || !active_lifecycle.is_empty() {
819        match timeout_at(drain_deadline, worker_event_rx.recv()).await {
820            Ok(Some(WorkerEvent::Stopped { source })) => {
821                active_sources.remove(&source);
822                active_lifecycle.remove(&source);
823            }
824            Ok(Some(WorkerEvent::Status(status))) => deferred_statuses.push(status),
825            Ok(Some(WorkerEvent::Subscribed { source, symbols })) => {
826                acknowledged.insert(source, symbols.into_iter().collect());
827            }
828            Ok(Some(WorkerEvent::Events { source, events })) => {
829                for event in events {
830                    let core_started = Instant::now();
831                    let outputs = core.handle_event(source, event);
832                    timings.core_processing.record(core_started.elapsed());
833                    if let Err(error) = emit_outputs(&writer, outputs, send_timeout).await {
834                        shutdown_error.get_or_insert(error);
835                        break 'drain;
836                    }
837                }
838            }
839            Ok(Some(WorkerEvent::Fatal { source, error })) => {
840                warn!(%source, error, "source failed during shutdown");
841            }
842            Ok(Some(WorkerEvent::RecoveryReady {
843                source,
844                targets,
845                generation,
846                connection_epoch,
847                receive_seq,
848            })) => {
849                core.handle_recovery_ready(
850                    source,
851                    &targets,
852                    generation,
853                    connection_epoch,
854                    receive_seq,
855                );
856            }
857            Ok(Some(WorkerEvent::AuthorizationHalted {
858                source,
859                reason,
860                lease_id,
861                lease_epoch,
862                buffered_frames,
863                uncommitted_events,
864            })) => {
865                warn!(
866                    %source, %reason, %lease_id, lease_epoch, buffered_frames, uncommitted_events,
867                    "authorization halted a source during shutdown"
868                );
869            }
870            Ok(None) | Err(_) => break,
871        }
872    }
873
874    let ended_at_us = now_us().unwrap_or(started_at_us);
875    core.begin_shutdown();
876    if settings.collector.final_snapshots {
877        let outputs = core.finalize(ended_at_us);
878        if let Err(error) = emit_outputs(&writer, outputs, send_timeout).await {
879            shutdown_error.get_or_insert(error);
880        }
881    }
882    // Capture coverage before the deferred disconnect transitions invalidate
883    // the final books. The transitions still update the per-source report.
884    let final_coverage = core.coverage(ended_at_us, stale_after_us);
885    for status in deferred_statuses {
886        let _ = core.handle_status(status, ended_at_us);
887    }
888    for handle in handles.into_values() {
889        handle.abort();
890    }
891    if let Err(error) = writer.flush(send_timeout).await {
892        shutdown_error.get_or_insert(error);
893    }
894    let stats_before_shutdown = writer.stats();
895    let writer_stats = match writer.shutdown_with_stats(send_timeout).await {
896        Ok(stats) => stats,
897        Err(error) => {
898            shutdown_error.get_or_insert(error);
899            stats_before_shutdown
900        }
901    };
902    // Workers have been aborted and no handle records after shutdown is
903    // requested. Finalizing here makes every accepted frame durable before the
904    // report claims a complete run.
905    let journal_status = match journal.as_ref() {
906        Some(handle) => {
907            if let Err(error) = handle.shutdown(send_timeout) {
908                shutdown_error.get_or_insert(error);
909            }
910            Some(handle.status())
911        }
912        None => None,
913    };
914    let mut journal_consistency_error = None;
915    let journal_verification = journal_status.as_ref().and_then(|status| {
916        match crate::journal::verify_run(std::path::Path::new(&status.run_directory)) {
917            Ok(mut verification) => {
918                if let Err(error) = crate::journal::validate_report_status(status, &verification) {
919                    let message = format!("{error:#}");
920                    verification.complete = false;
921                    verification
922                        .errors
923                        .push(format!("journal report/manifest mismatch: {message}"));
924                    journal_consistency_error = Some(message);
925                    shutdown_error.get_or_insert(error);
926                }
927                Some(verification)
928            }
929            Err(error) => {
930                journal_consistency_error = Some(format!("{error:#}"));
931                shutdown_error.get_or_insert(error);
932                None
933            }
934        }
935    });
936    stop_metrics(metrics_shutdown, metrics_handle).await;
937    let core_stats = core.stats();
938    let sources: Vec<_> = core.statuses().values().cloned().collect();
939    // Union across shards and replicas: every symbol at least one source
940    // successfully subscribed. Native identifiers; `coverage` maps them to
941    // SBE symbols through the metadata sidecar.
942    let acknowledged_symbols: Vec<String> = acknowledged
943        .values()
944        .flat_map(|symbols| symbols.iter().cloned())
945        .collect::<BTreeSet<_>>()
946        .into_iter()
947        .collect();
948    let mut report = RunReport::new(
949        "live",
950        exchange.name(),
951        exchange.environment(),
952        started_at_us,
953        ended_at_us,
954        &shutdown_reason,
955        &settings.output.path,
956        shard_count,
957        replica_count,
958        core_stats.discovered_symbols as usize,
959        core.symbol_count(),
960        core_stats,
961        writer_stats,
962        sources,
963        ReportConfig::from_settings(&settings, shard_count),
964    );
965    report.capacity = Some(capacity);
966    report.discovery = runtime.discovery.clone();
967    report.catalog = Some(bootstrap.catalog);
968    report.coverage = Some(final_coverage);
969    report.acknowledged_symbols = acknowledged_symbols;
970    report.journal = journal_status;
971    report.journal_verification = journal_verification;
972    report.journal_consistency_error = journal_consistency_error;
973    report.stage_timing = Some(timings.report());
974
975    // The final snapshot is persisted through the same latest-wins worker as
976    // every intermediate one, then the worker is drained and joined. A warm
977    // checkpoint that no discovery round refreshed keeps its original
978    // generation time, so readiness cannot mistake a restart for a refresh.
979    let catalog_generated_at_us = runtime.discovery.as_ref().map(|round| round.finished_at_us);
980    if let Err(error) = persistence.submit_final(
981        Arc::new(core.market_infos()),
982        ended_at_us,
983        catalog_generated_at_us,
984        send_timeout,
985    ) {
986        warn!(
987            error = %format!("{error:#}"),
988            "failed to persist final metadata/catalog snapshot"
989        );
990        shutdown_error.get_or_insert(error);
991    }
992    let persistence_status = persistence.status();
993    report.persistence = Some(persistence_status);
994    if let Err(error) = persistence.shutdown(send_timeout) {
995        warn!(
996            error = %format!("{error:#}"),
997            "metadata/catalog persistence worker did not stop cleanly"
998        );
999        shutdown_error.get_or_insert(error);
1000    }
1001    if crate::writer::WriterOptions::from_settings(&settings).rotation_enabled() {
1002        report.output_manifest = Some(
1003            crate::config::rotation_manifest_path(&settings.output.path)
1004                .display()
1005                .to_string(),
1006        );
1007    }
1008    if let Some(path) = sidecar_path(&settings.output.report_path) {
1009        report.write(path).context("writing run report")?;
1010    }
1011    if let Some(error) = shutdown_error {
1012        return Err(error);
1013    }
1014    Ok(report)
1015}

1017async fn discover_with_retry_stats(
1018    exchange: &Arc<dyn Exchange>,
1019    max_markets: usize,
1020    name: &str,
1021) -> anyhow::Result<DiscoveryResult> {
1022    let mut last_error: Option<anyhow::Error> = None;
1023    for attempt in 0..5u32 {
1024        match exchange.discover_markets_with_stats(max_markets).await {
1025            Ok(mut result) => {
1026                result.stats.retries = result.stats.retries.saturating_add(u64::from(attempt));
1027                return Ok(result);
1028            }
1029            Err(error) => {
1030                warn!(
1031                    exchange = name,
1032                    attempt = attempt + 1,
1033                    error = %format!("{error:#}"),
1034                    "initial market discovery failed"
1035                );
1036                last_error = Some(error);
1037                if attempt < 4 {
1038                    let delay = 500u64.saturating_mul(1u64 << attempt);
1039                    tokio::time::sleep(Duration::from_millis(delay)).await;
1040                }
1041            }
1042        }
1043    }
1044    Err(last_error.unwrap_or_else(|| anyhow::anyhow!("initial market discovery failed")))
1045}

1047async fn start_metrics(
1048    settings: &Settings,
1049    registry: &MetricsRegistry,
1050) -> anyhow::Result<(
1051    Option<broadcast::Sender<()>>,
1052    Option<JoinHandle<anyhow::Result<()>>>,
1053)> {
1054    let Some(address) = settings.metrics.listen else {
1055        return Ok((None, None));
1056    };
1057    let listener = tokio::net::TcpListener::bind(address)
1058        .await
1059        .with_context(|| format!("binding metrics listener {address}"))?;
1060    let (shutdown_tx, shutdown_rx) = broadcast::channel(1);
1061    let registry = registry.clone();
1062    let handle = tokio::spawn(async move { run_server_on(listener, registry, shutdown_rx).await });
1063    Ok((Some(shutdown_tx), Some(handle)))
1064}

1066async fn stop_metrics(
1067    shutdown: Option<broadcast::Sender<()>>,
1068    handle: Option<JoinHandle<anyhow::Result<()>>>,
1069) {
1070    if let Some(shutdown) = shutdown {
1071        let _ = shutdown.send(());
1072    }

1076            Ok(Ok(Err(error))) => warn!(error = %format!("{error:#}"), "metrics server failed"),
1077            Ok(Err(error)) => warn!(error = %error, "metrics server task failed"),
1078            Err(_) => warn!("metrics server did not stop within two seconds"),
1079        }
1080    }
1081}

1083#[expect(
1084    clippy::too_many_arguments,
1085    reason = "one worker event is dispatched against the core, writer, metrics and clock"
1086)]
1087async fn process_worker_event(
1088    event: WorkerEvent,
1089    core: &mut CollectorCore,
1090    writer: &WriterHandle,
1091    commands: &BTreeMap<SourceId, mpsc::Sender<Control>>,
1092    active_sources: &mut BTreeSet<SourceId>,
1093    acknowledged: &mut BTreeMap<SourceId, BTreeSet<String>>,
1094    discovery: &mut DiscoveryPlane,
1095    core_timing: &Histogram,
1096    send_timeout: Duration,
1097) -> anyhow::Result<()> {
1098    match event {
1099        WorkerEvent::Events { source, events } => {
1100            for event in events {
1101                let core_started = Instant::now();
1102                let outputs = core.handle_event(source, event);
1103                core_timing.record(core_started.elapsed());
1104                emit_outputs(writer, outputs, send_timeout).await?;
1105            }
1106            // The lifecycle control plane saw a market the universe does not
1107            // track; pull the authoritative catalog rather than waiting for
1108            // the next ticker.
1109            if core.take_lifecycle_reconcile_request() {
1110                discovery.request(FORCED_RECONCILE_COOLDOWN_US);
1111            }
1112            dispatch_controls(core, commands, send_timeout).await?;
1113        }
1114        WorkerEvent::Status(status) => {
1115            let lifecycle_reconnect =
1116                status.source.shard == LIFECYCLE_SHARD && status.connected && status.reconnects > 0;
1117            let outputs = core.handle_status(status, now_us().unwrap_or(0));
1118            emit_outputs(writer, outputs, send_timeout).await?;
1119            if lifecycle_reconnect {
1120                // The event stream is incomplete after a disconnect; the
1121                // REST catalog is the source of truth for what changed
1122                // while the control socket was down.
1123                discovery.request(0);
1124            }
1125            dispatch_controls(core, commands, send_timeout).await?;
1126        }
1127        WorkerEvent::Subscribed { source, symbols } => {
1128            acknowledged.insert(source, symbols.into_iter().collect());
1129        }
1130        WorkerEvent::Fatal { source, error } => {
1131            bail!("source {source} reported a fatal error: {error}");
1132        }
1133        WorkerEvent::RecoveryReady {
1134            source,
1135            targets,
1136            generation,
1137            connection_epoch,
1138            receive_seq,
1139        } => {
1140            core.handle_recovery_ready(source, &targets, generation, connection_epoch, receive_seq);
1141        }
1142        WorkerEvent::Stopped { source } => {
1143            active_sources.remove(&source);
1144        }
1145        WorkerEvent::AuthorizationHalted {
1146            source,
1147            reason,
1148            lease_id,
1149            lease_epoch,
1150            buffered_frames,
1151            uncommitted_events,
1152        } => {
1153            // Fail closed: no queued record from this source is committed after
1154            // an authorization denial. The gate's final audit state is the
1155            // authoritative account; counts here reconcile the in-flight burst.
1156            active_sources.remove(&source);
1157            bail!(
1158                "source {source} authorization halted ({reason}) lease={lease_id} epoch={lease_epoch}: \
1159                 {buffered_frames} buffered frame(s) and {uncommitted_events} uncommitted event(s) not committed"
1160            );
1161        }
1162    }
1163    Ok(())
1164}

1166async fn emit_outputs(
1167    writer: &WriterHandle,
1168    outputs: Vec<OutputMessage>,
1169    timeout: Duration,
1170) -> anyhow::Result<()> {
1171    for output in outputs {
1172        writer.send(output, timeout).await?;
1173    }
1174    Ok(())
1175}

1177async fn dispatch_controls(
1178    core: &mut CollectorCore,
1179    commands: &BTreeMap<SourceId, mpsc::Sender<Control>>,
1180    timeout_duration: Duration,
1181) -> anyhow::Result<()> {
1182    for (source, control) in core.drain_controls() {
1183        let Some(sender) = commands.get(&source) else {
1184            bail!("no command channel for source {source}");
1185        };
1186        match timeout(timeout_duration, sender.send(control)).await {
1187            Ok(Ok(())) => {}
1188            Ok(Err(error)) => {
1189                bail!("source {source} command channel closed: {error}");
1190            }
1191            Err(_) => {
1192                bail!(
1193                    "source {source} command channel was backpressured for {} seconds",
1194                    timeout_duration.as_secs()
1195                );
1196            }
1197        }
1198    }
1199    Ok(())
1200}

1202fn update_metrics(
1203    registry: &MetricsRegistry,
1204    core: &CollectorCore,
1205    writer: &WriterHandle,
1206    persistence: &PersistenceHandle,
1207    runtime: &RuntimeState,
1208    stale_after_us: i64,
1209) {
1210    let now = now_us().unwrap_or(0);
1211    let lifecycle_connected = core
1212        .statuses()
1213        .values()
1214        .filter(|status| status.source.shard == LIFECYCLE_SHARD && status.connected)
1215        .count() as u64;
1216    let status = RuntimeStatus {
1217        coverage: core.coverage(now, stale_after_us),
1218        catalog: runtime.catalog.clone(),
1219        discovery: runtime.discovery.clone(),
1220        discovery_failures: runtime.discovery_failures,
1221        persistence: persistence.status(),
1222        writer_healthy: writer.healthy(),
1223        fatal_error: runtime.fatal_error.clone(),
1224        lifecycle: crate::metrics::LifecycleHealth {
1225            configured: runtime.lifecycle_configured,
1226            connected: lifecycle_connected,
1227        },
1228    };
1229    registry.update(core.stats(), writer.stats(), core.statuses(), status);
1230}

1232async fn wait_for_shutdown_signal() -> String {
1233    #[cfg(unix)]
1234    {
1235        let mut terminate = match tokio::signal::unix::signal(
1236            tokio::signal::unix::SignalKind::terminate(),
1237        ) {
1238            Ok(signal) => signal,
1239            Err(error) => {
1240                warn!(error = %error, "could not install SIGTERM handler; only SIGINT will be handled");
1241                let _ = tokio::signal::ctrl_c().await;
1242                return "SIGINT".to_owned();
1243            }
1244        };
1245        tokio::select! {
1246            _ = tokio::signal::ctrl_c() => "SIGINT".to_owned(),
1247            _ = terminate.recv() => "SIGTERM".to_owned(),
1248        }
1249    }
1250    #[cfg(not(unix))]
1251    {
1252        let _ = tokio::signal::ctrl_c().await;
1253        "SIGINT".to_owned()
1254    }
1255}
