9#[derive(Debug)]
10pub struct Report {
11    pub invocation: Vec<String>,
12    pub finished_at_ms: u128,
13    pub steps: Vec<Outcome>,
14    pub failed: Vec<String>,
15}
16
17impl Report {
18    pub fn new(invocation: &[String]) -> Self {
19        let finished_at_ms = SystemTime::now()
20            .duration_since(UNIX_EPOCH)
21            .map_or(0, |duration| duration.as_millis());
22        Self {
23            invocation: invocation.to_vec(),
24            finished_at_ms,
25            steps: Vec::new(),
26            failed: Vec::new(),
27        }
28    }
29
30    pub fn push(&mut self, outcome: Outcome) {
31        if outcome.status == "failed" {
32            self.failed.push(outcome.command.clone());
33        }
34        self.steps.push(outcome);
35    }
36}

38pub fn write(root: &Path, report: &Report) -> Result<PathBuf, String> {
39    let dir = root.join("target/xtask");
40    fs::create_dir_all(&dir).map_err(|error| format!("creating {}: {error}", dir.display()))?;
41    let path = dir.join("last-run.json");
42    let mut out = String::from("{\n  \"finished_at_ms\": ");
43    out.push_str(&report.finished_at_ms.to_string());
44    out.push_str(",\n  \"invocation\": [");
45    for (index, arg) in report.invocation.iter().enumerate() {
46        if index > 0 {
47            out.push(',');
48        }
49        out.push('\n');
50        out.push_str("    ");
51        out.push_str(&quote(arg));
52    }
53    out.push_str("\n  ],\n  \"steps\": [");
54    for (index, step) in report.steps.iter().enumerate() {
55        if index > 0 {
56            out.push(',');
57        }
58        out.push_str("\n    {");
59        out.push_str("\n      \"command\": ");
60        out.push_str(&quote(&step.command));
61        out.push_str(",\n      \"status\": ");
62        out.push_str(&quote(&step.status));
63        out.push_str(",\n      \"exit_code\": ");
64        out.push_str(
65            &step
66                .exit_code
67                .map_or_else(|| "null".to_owned(), |code| code.to_string()),
68        );
69        out.push_str(",\n      \"duration_ms\": ");
70        out.push_str(&step.duration_ms.to_string());
71        out.push_str(",\n      \"output_tail\": ");
72        out.push_str(&quote(&step.output_tail));
73        out.push_str("\n    }");
74    }
75    out.push_str("\n  ],\n  \"failed\": [");
76    for (index, failed) in report.failed.iter().enumerate() {
77        if index > 0 {
78            out.push(',');
79        }
80        out.push('\n');
81        out.push_str("    ");
82        out.push_str(&quote(failed));
83    }
84    out.push_str("\n  ]\n}\n");
85    fs::write(&path, out).map_err(|error| format!("writing {}: {error}", path.display()))?;
86    Ok(path)
87}

89fn quote(value: &str) -> String {
90    let mut out = String::with_capacity(value.len() + 2);
91    out.push('"');
92    for character in value.chars() {
93        match character {
94            '"' => out.push_str("\\\""),
95            '\\' => out.push_str("\\\\"),
96            '\n' => out.push_str("\\n"),
97            '\r' => out.push_str("\\r"),
98            '\t' => out.push_str("\\t"),
99            control if control.is_control() => {
100                out.push_str(&format!("\\u{:04x}", control as u32));
101            }
102            other => out.push(other),
103        }
104    }
105    out.push('"');
106    out
107}
